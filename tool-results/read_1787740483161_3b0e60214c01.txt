     1→'use client'
     2→
     3→import { useMemo, useRef, useState } from 'react'
     4→import { useLocale, useTranslations } from 'next-intl'
     5→import { AnimatePresence, motion, useReducedMotion } from 'framer-motion'
     6→import { z } from 'zod'
     7→import {
     8→  Globe, Workflow, Boxes, Check, ArrowLeft, ArrowRight,
     9→  Send, RotateCw, AlertCircle,
    10→} from 'lucide-react'
    11→import {
    12→  Slider,
    13→} from '@/components/ui/slider'
    14→import { Input } from '@/components/ui/input'
    15→import { Label } from '@/components/ui/label'
    16→import { Button } from '@/components/ui/button'
    17→import { cn } from '@/lib/utils'
    18→import { SectionHeading } from '@/components/shared/section-heading'
    19→import { playSuccess } from '@/lib/sound'
    20→import { toast } from 'sonner'
    21→import { RingGauge } from './ring-gauge'
    22→import {
    23→  computeEstimate, formatMoney,
    24→  type CalculatorInput, type ServiceType, type IntegrationKey,
    25→  type AutomationLevel, type LanguageOption, type ThreeDOption, type Locale,
    26→} from '@/lib/calculator'
    27→
    28→type Step = 0 | 1 | 2
    29→
    30→const SERVICES: { id: ServiceType; icon: typeof Globe }[] = [
    31→  { id: 'website', icon: Globe },
    32→  { id: 'automation', icon: Workflow },
    33→  { id: 'full', icon: Boxes },
    34→]
    35→
    36→const INTEGRATIONS: IntegrationKey[] = ['crm', 'invoicing', 'email', 'telegram', 'sheets', 'ai']
    37→
    38→const leadSchema = z.object({
    39→  name: z.string().trim().min(2),
    40→  email: z.string().email(),
    41→  whatsapp: z.string().trim().optional(),
    42→})
    43→
    44→type LeadForm = z.infer<typeof leadSchema>
    45→
    46→const INITIAL_INPUT: CalculatorInput = {
    47→  service: 'website',
    48→  pages: 6,
    49→  languages: 'bilingual',
    50→  threeD: 'no',
    51→  integrations: [],
    52→  automationLevel: 'essential',
    53→}
    54→
    55→export function Calculator() {
    56→  const t = useTranslations('calculator')
    57→  // useLocale() returns a broad `string`; narrow to the routing union
    58→  // (unknown values fall back to `ar`, the site default) so formatMoney's
    59→  // tightened `Locale` param type-checks.
    60→  const localeRaw = useLocale()
    61→  const locale: Locale = localeRaw === 'en' ? 'en' : 'ar'
    62→  const reduced = useReducedMotion()
    63→
    64→  const [step, setStep] = useState<Step>(0)
    65→  const [dir, setDir] = useState<1 | -1>(1)
    66→  const [input, setInput] = useState<CalculatorInput>(INITIAL_INPUT)
    67→  const [form, setForm] = useState<LeadForm>({ name: '', email: '', whatsapp: '' })
    68→  const [errors, setErrors] = useState<{ name?: string; email?: string }>({})
    69→  const [submitting, setSubmitting] = useState(false)
    70→  const [done, setDone] = useState(false)
    71→  const [reference, setReference] = useState<string | null>(null)
    72→  // FIX(2-c/18): honeypot trap — bots autofill hidden "companyWebsite"
    73→  // fields; humans never see it. The value rides along in the JSON body
    74→  // and the API silently discards bot submissions with a fake success.
    75→  const honeypotRef = useRef<HTMLInputElement>(null)
    76→
    77→  const result = useMemo(() => computeEstimate(input), [input])
    78→
    79→  // Breakdown line labels — resolved once for type-safety (guide §4.6).
    80→  const breakdownLabels = {
    81→    base: t('result.base'),
    82→    pages: t('result.pages'),
    83→    bilingual: t('result.bilingual'),
    84→    threeD: t('result.threeD'),
    85→    integrations: t('result.integrations'),
    86→    advanced: t('result.advanced'),
    87→  } as const
    88→
    89→  const goNext = () => { setDir(1); setStep((s) => Math.min(2, s + 1) as Step) }
    90→  const goBack = () => { setDir(-1); setStep((s) => Math.max(0, s - 1) as Step) }
    91→
    92→  const toggleIntegration = (key: IntegrationKey) => {
    93→    setInput((p) => ({
    94→      ...p,
    95→      integrations: p.integrations.includes(key)
    96→        ? p.integrations.filter((k) => k !== key)
    97→        : [...p.integrations, key],
    98→    }))
    99→  }
   100→
   101→  const onSubmit = async (e: React.FormEvent) => {
   102→    e.preventDefault()
   103→    const parsed = leadSchema.safeParse(form)
   104→    if (!parsed.success) {
   105→      const fe: { name?: string; email?: string } = {}
   106→      for (const issue of parsed.error.issues) {
   107→        const path = issue.path[0]
   108→        if (path === 'name') fe.name = t('errors.name')
   109→        if (path === 'email') fe.email = t('errors.email')
   110→      }
   111→      setErrors(fe)
   112→      return
   113→    }
   114→    setErrors({})
   115→    setSubmitting(true)
   116→    // Phase 3: real storage — the server recomputes the estimate from the
   117→    // wizard options and returns a reference (guide §2.9). Client numbers
   118→    // are never sent: only the option set travels.
   119→    try {
   120→      const res = await fetch('/api/leads', {
   121→        method: 'POST',
   122→        headers: {
   123→          'Content-Type': 'application/json',
   124→          'x-elyra-locale': locale,
   125→        },
   126→        body: JSON.stringify({
   127→          source: 'calculator',
   128→          companyWebsite: honeypotRef.current?.value ?? '',
   129→          name: parsed.data.name,
   130→          email: parsed.data.email,
   131→          whatsapp: parsed.data.whatsapp || undefined,
   132→          service: input.service,
   133→          pages: input.pages,
   134→          languages: input.languages,
   135→          threeD: input.threeD,
   136→          integrations: input.integrations,
   137→          automationLevel: input.automationLevel,
   138→        }),
   139→      })
   140→
   141→      if (res.status === 201) {
   142→        const data = (await res.json()) as { reference?: string }
   143→        setReference(data.reference ?? null)
   144→        setDone(true)
   145→        playSuccess() // Phase 2 sensory feedback — fires on REAL success only
   146→        return
   147→      }
   148→
   149→      // Server rejected: surface translated server-side messages.
   150→      const data = (await res.json().catch(() => null)) as
   151→        | { message?: string; fields?: Record<string, string> }
   152→        | null
   153→      if (res.status === 400 && data?.fields) {
   154→        const fe: { name?: string; email?: string } = {}
   155→        if (data.fields.name) fe.name = data.fields.name
   156→        if (data.fields.email) fe.email = data.fields.email
   157→        setErrors(fe)
   158→      }
   159→      toast.error(t('form.errorTitle'), {
   160→        description: data?.message ?? t('form.errorNetwork'),
   161→      })
   162→    } catch {
   163→      // Network failure — data stays in the form for a retry.
   164→      toast.error(t('form.errorTitle'), { description: t('form.errorNetwork') })
   165→    } finally {
   166→      setSubmitting(false)
   167→    }
   168→  }
   169→
   170→  const slideVariants = reduced
   171→    ? undefined
   172→    : {
   173→        enter: (d: number) => ({ opacity: 0, x: d * 30 }),
   174→        center: { opacity: 1, x: 0 },
   175→        exit: (d: number) => ({ opacity: 0, x: -d * 30 }),
   176→      }
   177→
   178→  return (
   179→    <section id="calculator" className="bg-background py-20 sm:py-28" aria-labelledby="calc-title" aria-label={t('kicker')}>
   180→      <div className="elyra-container max-w-5xl">
   181→        <SectionHeading
   182→          kicker={t('kicker')}
   183→          title={t('title')}
   184→          subtitle={t('subtitle')}
   185→          titleId="calc-title"
   186→        />
   187→
   188→        {/* Progress */}
   189→        <div className="mt-10 flex items-center justify-between gap-4">
   190→          <p className="text-sm text-muted-foreground">
   191→            {t('step', { current: step + 1, total: 3 })}
   192→          </p>
   193→          <div className="flex flex-1 gap-2">
   194→            {[0, 1, 2].map((s) => (
   195→              <div
   196→                key={s}
   197→                className="h-1.5 flex-1 overflow-hidden rounded-full bg-muted"
   198→                aria-hidden="true"
   199→              >
   200→                <motion.div
   201→                  className="h-full rounded-full bg-primary"
   202→                  initial={false}
   203→                  animate={{ width: step >= s ? '100%' : '0%' }}
   204→                  transition={{ duration: 0.4 }}
   205→                />
   206→              </div>
   207→            ))}
   208→          </div>
   209→        </div>
   210→
   211→        <div className="relative mt-8 overflow-hidden rounded-3xl border border-border bg-card p-6 sm:p-10">
   212→          <AnimatePresence mode="wait" custom={dir}>
   213→            <motion.div
   214→              key={step}
   215→              custom={dir}
   216→              variants={slideVariants}
   217→              initial="enter"
   218→              animate="center"
   219→              exit="exit"
   220→              transition={{ duration: 0.4, ease: [0.16, 1, 0.3, 1] }}
   221→            >
   222→              {/* STEP 0 — Service type */}
   223→              {step === 0 ? (
   224→                <div>
   225→                  <h3 className="text-lg font-semibold">{t('steps.service')}</h3>
   226→                  <div className="mt-6 grid gap-4 sm:grid-cols-3">
   227→                    {SERVICES.map(({ id, icon: Icon }) => {
   228→                      const active = input.service === id
   229→                      const isFull = id === 'full'
   230→                      return (
   231→                        <button
   232→                          key={id}
   233→                          type="button"
   234→                          data-cursor="magnet"
   235→                          onClick={() => setInput((p) => ({ ...p, service: id }))}
   236→                          aria-pressed={active}
   237→                          className={cn(
   238→                            'group relative overflow-hidden rounded-2xl border p-5 text-start transition-all',
   239→                            active
   240→                              ? 'border-primary bg-primary/5 shadow-[0_0_0_1px_var(--color-primary)]'
   241→                              : 'border-border hover:border-primary/40 hover:bg-foreground/[0.02]'
   242→                          )}
   243→                        >
   244→                          {isFull ? (
   245→                            <span className="absolute end-3 top-3 rounded-full bg-primary px-2 py-0.5 text-[10px] font-semibold text-primary-foreground">
   246→                              {t('services.full.popular')}
   247→                            </span>
   248→                          ) : null}
   249→                          <div className={cn(
   250→                            'flex size-11 items-center justify-center rounded-xl transition-colors',
   251→                            active ? 'bg-primary text-primary-foreground' : 'bg-primary/10 text-primary'
   252→                          )}>
   253→                            <Icon className="size-5" aria-hidden="true" />
   254→                          </div>
   255→                          <p className="mt-3 font-semibold">{t(`services.${id}.title`)}</p>
   256→                          <p className="mt-1 text-sm text-muted-foreground">{t(`services.${id}.desc`)}</p>
   257→                          {active ? (
   258→                            <span className="absolute -end-2 -top-2 flex size-6 items-center justify-center rounded-full bg-primary text-primary-foreground">
   259→                              <Check className="size-3.5" aria-hidden="true" />
   260→                            </span>
   261→                          ) : null}
   262→                        </button>
   263→                      )
   264→                    })}
   265→                  </div>
   266→                </div>
   267→              ) : null}
   268→
   269→              {/* STEP 1 — Features */}
   270→              {step === 1 ? (
   271→                <div className="space-y-8">
   272→                  <div>
   273→                    <div className="flex items-center justify-between">
   274→                      <h3 className="text-lg font-semibold">{t('pages')}</h3>
   275→                      <span className="text-sm font-medium text-primary tabular-nums">
   276→                        {t('pagesValue', { count: input.pages })}
   277→                      </span>
   278→                    </div>
   279→                    <div className="mt-4 px-1">
   280→                      <Slider
   281→                        value={[input.pages]}
   282→                        onValueChange={(v) => setInput((p) => ({ ...p, pages: v[0] ?? p.pages }))}
   283→                        min={1}
   284→                        max={20}
   285→                        step={1}
   286→                        aria-label={t('pages')}
   287→                      />
   288→                      <div className="mt-2 flex justify-between text-xs text-muted-foreground">
   289→                        <span>1</span><span>20</span>
   290→                      </div>
   291→                    </div>
   292→                  </div>
   293→
   294→                  <div>
   295→                    <h3 className="text-lg font-semibold">{t('languages')}</h3>
   296→                    <div className="mt-3 grid gap-3 sm:grid-cols-2">
   297→                      {(['single', 'bilingual'] as LanguageOption[]).map((opt) => {
   298→                        const active = input.languages === opt
   299→                        return (
   300→                          <button
   301→                            key={opt}
   302→                            type="button"
   303→                            onClick={() => setInput((p) => ({ ...p, languages: opt }))}
   304→                            aria-pressed={active}
   305→                            className={cn(
   306→                              'rounded-xl border px-4 py-3 text-start transition-colors',
   307→                              active ? 'border-primary bg-primary/5' : 'border-border hover:bg-foreground/[0.02]'
   308→                            )}
   309→                          >
   310→                            <span className="flex items-center gap-2">
   311→                              <span className={cn('flex size-5 items-center justify-center rounded-full border', active ? 'border-primary bg-primary text-primary-foreground' : 'border-border')}>
   312→                                {active ? <Check className="size-3" aria-hidden="true" /> : null}
   313→                              </span>
   314→                              <span className="text-sm font-medium">{t(`languagesOptions.${opt}`)}</span>
   315→                            </span>
   316→                          </button>
   317→                        )
   318→                      })}
   319→                    </div>
   320→                  </div>
   321→
   322→                  <div>
   323→                    <h3 className="text-lg font-semibold">{t('threeD')}</h3>
   324→                    <div className="mt-3 grid gap-3 sm:grid-cols-2">
   325→                      {(['yes', 'no'] as ThreeDOption[]).map((opt) => {
   326→                        const active = input.threeD === opt
   327→                        const disabled = input.service === 'automation'
   328→                        return (
   329→                          <button
   330→                            key={opt}
   331→                            type="button"
   332→                            disabled={disabled}
   333→                            onClick={() => setInput((p) => ({ ...p, threeD: opt }))}
   334→                            aria-pressed={active}
   335→                            className={cn(
   336→                              'rounded-xl border px-4 py-3 text-start transition-colors',
   337→                              disabled && 'opacity-50',
   338→                              active ? 'border-primary bg-primary/5' : 'border-border hover:bg-foreground/[0.02]'
   339→                            )}
   340→                          >
   341→                            <span className="flex items-center gap-2">
   342→                              <span className={cn('flex size-5 items-center justify-center rounded-full border', active ? 'border-primary bg-primary text-primary-foreground' : 'border-border')}>
   343→                                {active ? <Check className="size-3" aria-hidden="true" /> : null}
   344→                              </span>
   345→                              <span className="text-sm font-medium">{t(`threeDOptions.${opt}`)}</span>
   346→                            </span>
   347→                          </button>
   348→                        )
   349→                      })}
   350→                    </div>
   351→                  </div>
   352→
   353→                  <div>
   354→                    <h3 className="text-lg font-semibold">{t('integrations')}</h3>
   355→                    <div className="mt-3 flex flex-wrap gap-2">
   356→                      {INTEGRATIONS.map((key) => {
   357→                        const active = input.integrations.includes(key)
   358→                        return (
   359→                          <button
   360→                            key={key}
   361→                            type="button"
   362→                            onClick={() => toggleIntegration(key)}
   363→                            aria-pressed={active}
   364→                            className={cn(
   365→                              'inline-flex items-center gap-1.5 rounded-full border px-3 py-1.5 text-sm transition-colors',
   366→                              active ? 'border-primary bg-primary/10 text-primary' : 'border-border hover:bg-foreground/[0.02]'
   367→                            )}
   368→                          >
   369→                            <span className={cn('flex size-4 items-center justify-center rounded-full border', active ? 'border-primary bg-primary text-primary-foreground' : 'border-border')}>
   370→                              {active ? <Check className="size-2.5" aria-hidden="true" /> : null}
   371→                            </span>
   372→                            {t(`integrationsOptions.${key}`)}
   373→                          </button>
   374→                        )
   375→                      })}
   376→                    </div>
   377→                  </div>
   378→
   379→                  <div>
   380→                    <h3 className="text-lg font-semibold">{t('automationLevel')}</h3>
   381→                    <div className="mt-3 grid gap-3 sm:grid-cols-2">
   382→                      {(['essential', 'advanced'] as AutomationLevel[]).map((opt) => {
   383→                        const active = input.automationLevel === opt
   384→                        const disabled = input.service === 'website'
   385→                        return (
   386→                          <button
   387→                            key={opt}
   388→                            type="button"
   389→                            disabled={disabled}
   390→                            onClick={() => setInput((p) => ({ ...p, automationLevel: opt }))}
   391→                            aria-pressed={active}
   392→                            className={cn(
   393→                              'rounded-xl border px-4 py-3 text-start transition-colors',
   394→                              disabled && 'opacity-50',
   395→                              active ? 'border-primary bg-primary/5' : 'border-border hover:bg-foreground/[0.02]'
   396→                            )}
   397→                          >
   398→                            <span className="flex items-center gap-2">
   399→                              <span className={cn('flex size-5 items-center justify-center rounded-full border', active ? 'border-primary bg-primary text-primary-foreground' : 'border-border')}>
   400→                                {active ? <Check className="size-3" aria-hidden="true" /> : null}
   401→                              </span>
   402→                              <span className="text-sm font-medium">{t(`automationLevelOptions.${opt}`)}</span>
   403→                            </span>
   404→                          </button>
   405→                        )
   406→                      })}
   407→                    </div>
   408→                  </div>
   409→                </div>
   410→              ) : null}
   411→
   412→              {/* STEP 2 — Result */}
   413→              {step === 2 ? (
   414→                <div>
   415→                  {done ? (
   416→                    <div className="flex flex-col items-center py-8 text-center">
   417→                      <div className="flex size-16 items-center justify-center rounded-full bg-g-green/15 text-g-green">
   418→                        <Check className="size-8" aria-hidden="true" />
   419→                      </div>
   420→                      <h3 className="mt-5 text-2xl font-semibold">{t('form.successTitle')}</h3>
   421→                      <p className="mt-2 max-w-md text-muted-foreground">{t('form.successDesc')}</p>
   422→                      {reference ? (
   423→                        <p className="mt-3 rounded-full border border-g-green/30 bg-g-green/5 px-4 py-1.5 font-mono text-sm font-semibold tracking-wide text-g-green">
   424→                          {t('form.successReference', { reference })}
   425→                        </p>
   426→                      ) : null}
   427→                      <button
   428→                        type="button"
   429→                        onClick={() => { setDone(false); setReference(null); setStep(0); setInput(INITIAL_INPUT); setForm({ name: '', email: '', whatsapp: '' }) }}
   430→                        className="mt-6 inline-flex h-11 items-center gap-2 rounded-full border border-border px-5 text-sm font-medium hover:bg-foreground/5"
   431→                      >
   432→                        <RotateCw className="size-4" aria-hidden="true" />
   433→                        {t('form.successRestart')}
   434→                      </button>
   435→                    </div>
   436→                  ) : (
   437→                    <div className="grid gap-8 lg:grid-cols-2">
   438→                      <div>
   439→                        <div className="rounded-2xl border border-border bg-gradient-to-br from-primary/5 to-transparent p-6">
   440→                          {/* WS-4: animated SVG ring gauges */}
   441→                          <div className="mb-6 flex items-center justify-center gap-12">
   442→                            <RingGauge
   443→                              fraction={Math.min(1, result.max / 20000)}
   444→                              value={result.max}
   445→                              formatValue={(n) => formatMoney(Math.round(n), locale)}
   446→                              label={t('result.budget')}
   447→                              color="#0071E3"
   448→                              isRtl={locale === 'ar'}
   449→                            />
   450→                            <RingGauge
   451→                              fraction={Math.min(1, result.weeksMax / 12)}
   452→                              value={result.weeksMax}
   453→                              formatValue={(n) => `${Math.round(n)}`}
   454→                              label={t('result.duration')}
   455→                              color="#34A853"
   456→                              isRtl={locale === 'ar'}
   457→                            />
   458→                          </div>
   459→                          <p className="text-xs font-medium uppercase tracking-wider text-muted-foreground">
   460→                            {t('result.budget')}
   461→                          </p>
   462→                          <p className="mt-2 text-3xl font-bold tracking-tight text-primary sm:text-4xl">
   463→                            {formatMoney(result.min, locale)} – {formatMoney(result.max, locale)}
   464→                          </p>
   465→                          <p className="mt-4 text-xs font-medium uppercase tracking-wider text-muted-foreground">
   466→                            {t('result.duration')}
   467→                          </p>
   468→                          <p className="mt-1 text-xl font-semibold">
   469→                            {t('result.weeks', { min: result.weeksMin, max: result.weeksMax })}
   470→                          </p>
   471→                        </div>
   472→
   473→                        <div className="mt-4">
   474→                          <p className="text-sm font-semibold">{t('result.breakdownTitle')}</p>
   475→                          <ul className="mt-2 space-y-1.5 text-sm">
   476→                            {result.breakdown.map((line) => (
   477→                              <li key={line.labelKey} className="flex items-center justify-between gap-2">
   478→                                <span className="text-muted-foreground">{breakdownLabels[line.labelKey as keyof typeof breakdownLabels]}</span>
   479→                                <span className="tabular-nums">
   480→                                  {formatMoney(line.min, locale)} – {formatMoney(line.max, locale)}
   481→                                </span>
   482→                              </li>
   483→                            ))}
   484→                          </ul>
   485→                        </div>
   486→                        <p className="mt-4 flex items-start gap-1.5 text-xs text-muted-foreground">
   487→                          <AlertCircle className="mt-0.5 size-3.5 shrink-0" aria-hidden="true" />
   488→                          {t('result.disclaimer')}
   489→                        </p>
   490→                      </div>
   491→
   492→                      <form onSubmit={onSubmit} className="rounded-2xl border border-border p-6" noValidate>
   493→                        <h3 className="text-lg font-semibold">{t('form.title')}</h3>
   494→                        {/* Honeypot — bots fill it, humans never see it (API silently discards) */}
   495→                        <input
   496→                          ref={honeypotRef}
   497→                          type="text"
   498→                          name="companyWebsite"
   499→                          tabIndex={-1}
   500→                          autoComplete="off"
   501→                          aria-hidden="true"
   502→                          className="pointer-events-none absolute -left-[9999px] h-px w-px overflow-hidden"
   503→                        />
   504→                        <div className="mt-4 space-y-4">
   505→                          <div>
   506→                            <Label htmlFor="calc-name" className="text-sm">{t('form.name')}</Label>
   507→                            <Input
