   103→const JSON_CLS = {
   104→  key: 'text-g-blue',
   105→  string: 'text-g-green',
   106→  number: 'text-g-yellow',
   107→  punct: 'text-white/40',
   108→} as const
   109→
   110→interface JsonToken {
   111→  text: string
   112→  cls: string
   113→}
   114→
   115→function isJsonObject(value: JsonValue): value is { [key: string]: JsonValue } {
   116→  return typeof value === 'object'
   117→}
   118→
   119→function scalarToken(value: Exclude<JsonValue, { [key: string]: JsonValue }>): JsonToken {
   120→  if (typeof value === 'string') return { text: JSON.stringify(value), cls: JSON_CLS.string }
   121→  if (typeof value === 'number') return { text: String(value), cls: JSON_CLS.number }
   122→  return { text: String(value), cls: JSON_CLS.number } // boolean
   123→}
   124→
   125→/** Pretty-prints a JsonValue into syntax-colored display lines (2-space indent). */
   126→function jsonLines(value: JsonValue, indent = 0): JsonToken[][] {
   127→  if (!isJsonObject(value)) return [[scalarToken(value)]]
   128→  const pad = '  '.repeat(indent)
   129→  const lines: JsonToken[][] = [[{ text: '{', cls: JSON_CLS.punct }]]
   130→  const entries = Object.entries(value)
   131→  entries.forEach(([key, val], i) => {
   132→    const comma = i < entries.length - 1
   133→    const keyTok: JsonToken = { text: `${pad}  ${JSON.stringify(key)}`, cls: JSON_CLS.key }
   134→    if (isJsonObject(val)) {
   135→      lines.push([keyTok, { text: ': ', cls: JSON_CLS.punct }])
   136→      const nested = jsonLines(val, indent + 1)
   137→      if (comma) {
   138→        const lastLine = nested[nested.length - 1]
   139→        const lastTok = lastLine?.[lastLine.length - 1]
   140→        if (lastTok) lastTok.text += ','
   141→      }
   142→      nested.forEach((line) => lines.push(line))
   143→    } else {
   144→      const tok = scalarToken(val)
   145→      lines.push([
   146→        keyTok,
   147→        { text: ': ', cls: JSON_CLS.punct },
   148→        { text: tok.text + (comma ? ',' : ''), cls: tok.cls },
   149→      ])
   150→    }
   151→  })
   152→  lines.push([{ text: `${pad}}`, cls: JSON_CLS.punct }])
   153→  return lines
   154→}

   156→/** UI-3: wall-clock timestamp for log lines — HH:MM:SS.mmm, LTR-safe. */
   157→function logTimestamp(date: Date): string {
   158→  const p2 = (n: number) => String(n).padStart(2, '0')
   159→  const p3 = (n: number) => String(n).padStart(3, '0')
   160→  return `${p2(date.getHours())}:${p2(date.getMinutes())}:${p2(date.getSeconds())}.${p3(date.getMilliseconds())}`
   161→}

   180→export function AutomationSimulator({
   181→  scenario: initialScenario = 'newOrder',

   183→}: SimulatorProps) {
   184→  const t = useTranslations('simulator')
   185→  const isRtl = useIsRtl()
   186→  const reduced = useReducedMotion()

   190→
   191→  const [scenario, setScenario] = useState<ScenarioId>(initialScenario)

   194→
   195→  // node center positions in % (RTL mirrors the order so the flow reads naturally)
   196→  const positions = useMemo(() => {
   197→    const ltr = [8, 29, 50, 71, 92]
   198→    return isRtl ? [...ltr].reverse() : ltr
   199→  }, [isRtl])
   200→
   201→  const [status, setStatus] = useState<'idle' | 'running' | 'completed'>('idle')
   202→  const [currentStep, setCurrentStep] = useState(-1)
   203→  const [completed, setCompleted] = useState<number[]>([])
   204→  const [counter, setCounter] = useState(0) // live ms counter for current step
   205→  const [logLines, setLogLines] = useState<LogLine[]>([]) // UI-3: execution log
   206→  const [payloadFlash, setPayloadFlash] = useState(false) // UI-3: "payload arrived" highlight
   207→
   208→  const timeouts = useRef<number[]>([])
   209→  const rafRef = useRef<number>(0)
   210→  const logSeq = useRef(0) // UI-3: stable id sequence for log lines
   211→  const logScrollRef = useRef<HTMLDivElement | null>(null)
   212→
   213→  const clearAll = useCallback(() => {
   214→    timeouts.current.forEach((id) => window.clearTimeout(id))
   215→    timeouts.current = []
   216→    cancelAnimationFrame(rafRef.current)
   217→  }, [])
   218→
   219→  const schedule = useCallback((fn: () => void, ms: number) => {
   220→    const id = window.setTimeout(fn, ms)
   221→    timeouts.current.push(id)
   222→  }, [])
   223→
   224→  useEffect(() => () => clearAll(), [clearAll])
   225→
   226→  const reset = useCallback(() => {
   227→    clearAll()
   228→    setStatus('idle')
   229→    setCurrentStep(-1)
   230→    setCompleted([])
   231→    setCounter(0)
   232→    setLogLines([]) // UI-3: log clears on scenario change
   233→    setPayloadFlash(false)
   234→  }, [clearAll])
   235→
   236→  // Reset whenever scenario changes — deferred to rAF so the setStates
   237→  // inside reset() aren't called synchronously within the effect body.
   238→  useEffect(() => {
   239→    let cancelled = false
   240→    const id = requestAnimationFrame(() => {
   241→      if (!cancelled) reset()
   242→    })
   243→    return () => {
   244→      cancelled = true
   245→      cancelAnimationFrame(id)
   246→    }
   247→  }, [scenario, reset])
   248→
   249→  // UI-3: keep the newest log line in view (instant jump — functional
   250→  // scrolling, not an animation, so it stays under reduced motion).
   251→  useEffect(() => {
   252→    const el = logScrollRef.current
   253→    if (el) el.scrollTop = el.scrollHeight
   254→  }, [logLines])
   255→
   256→  const totalMs = useMemo(() => {
   257→    return steps.reduce((acc, step) => {
   258→      const ms = Number(t.raw(`scenarios.${scenario}.steps.${step.id}.ms`))
   259→      return acc + (Number.isFinite(ms) ? ms : 0)
   260→    }, 0)
   261→  }, [steps, scenario, t])
   262→
   263→  const run = useCallback(() => {
   264→    clearAll()
   265→    setCompleted([])
   266→    setCounter(0)
   267→    setStatus('running')
   268→    setCurrentStep(0)
   269→
   270→    // UI-3: fresh log — the "webhook received" entry line comes first.
   271→    logSeq.current += 1
   272→    setLogLines([{
   273→      id: logSeq.current,
   274→      time: logTimestamp(new Date()),
   275→      kind: 'entry',
   276→      text: FLOW_ENTRY[scenario],
   277→    }])
   278→    // UI-3: payload panel flashes "arrived" (~1s, cleaned with the rest).
   279→    setPayloadFlash(true)
   280→    schedule(() => setPayloadFlash(false), 1000)
   281→
   282→    const STEP_DISPLAY = reduced ? 250 : 850
   283→    const TRANSITION = reduced ? 80 : 320
   284→
   285→    steps.forEach((step, i) => {
   286→      const startAt = i * (STEP_DISPLAY + TRANSITION)
   287→      schedule(() => {
   288→        setCurrentStep(i)
   289→        // animate counter 0 → step.ms over STEP_DISPLAY
   290→        const stepMs = t.raw(`scenarios.${scenario}.steps.${step.id}.ms`) as number
   291→        const target = Number(stepMs)
   292→        if (reduced) {
   293→          setCounter(target)
   294→        } else {
   295→          const start = performance.now()
   296→          const tick = (now: number) => {
   297→            const p = Math.min(1, (now - start) / STEP_DISPLAY)
   298→            const eased = 1 - Math.pow(1 - p, 3)
   299→            setCounter(Math.round(eased * target))
   300→            if (p < 1) rafRef.current = requestAnimationFrame(tick)
   301→          }
   302→          rafRef.current = requestAnimationFrame(tick)
   303→        }
   304→      }, startAt)
   305→
   306→      schedule(() => {
   307→        setCompleted((c) => (c.includes(i) ? c : [...c, i]))
   308→        // UI-3: timestamped log line appended at completion time.
   309→        const ms = Number(t.raw(`scenarios.${scenario}.steps.${step.id}.ms`))
   310→        logSeq.current += 1
   311→        setLogLines((prev) => [...prev, {
   312→          id: logSeq.current,
   313→          time: logTimestamp(new Date()),
   314→          kind: 'step',
   315→          text: t(`scenarios.${scenario}.steps.${step.id}.title`),
   316→          ms: Number.isFinite(ms) ? ms : 0,
   317→        }])
   318→      }, startAt + STEP_DISPLAY)
   319→    })
   320→
   321→    schedule(() => {
   322→      setStatus('completed')
   323→      setCurrentStep(-1)
   324→      setCounter(0)
   325→      playSuccess() // Phase 2 sensory feedback (no-op while muted)
   326→      // UI-3: final summary line for the terminal.
   327→      logSeq.current += 1
   328→      setLogLines((prev) => [...prev, {
   329→        id: logSeq.current,
   330→        time: logTimestamp(new Date()),
   331→        kind: 'done',
   332→        text: 'flow complete',
   333→        ms: totalMs,
   334→      }])
   335→    }, steps.length * (STEP_DISPLAY + TRANSITION))
   336→  }, [clearAll, reduced, steps, scenario, t, schedule, totalMs])
   337→
   338→  const secondsLabel = (totalMs / 1000).toFixed(2)
   339→
   340→  // current step def
   341→  const activeStep = currentStep >= 0 ? steps[currentStep] : null
   342→  const activeStepTitle = activeStep ? t(`scenarios.${scenario}.steps.${activeStep.id}.title`) : null
   343→  const activeStepDesc = activeStep ? t(`scenarios.${scenario}.steps.${activeStep.id}.desc`) : null
   344→  const activeStepMs = activeStep ? Number(t.raw(`scenarios.${scenario}.steps.${activeStep.id}.ms`)) : 0
   345→
   346→  // UI-3: syntax-highlighted payload lines for the active scenario.
   347→  const payloadLines = useMemo(() => jsonLines(PAYLOADS[scenario]), [scenario])
   348→
   349→  // UI-3: stats chip values — live elapsed while running (finished steps +
   350→  // in-flight counter; during the inter-step transition the active step is
   351→  // already in `completed`, so the counter is never double-counted),
   352→  // expected/final total otherwise.
   353→  const completedMsSum = useMemo(() => completed.reduce((acc, i) => {
   354→    const step = steps[i]
   355→    if (!step) return acc
   356→    const ms = Number(t.raw(`scenarios.${scenario}.steps.${step.id}.ms`))
   357→    return acc + (Number.isFinite(ms) ? ms : 0)
   358→  }, 0), [completed, steps, scenario, t])
   359→  const inFlightMs = currentStep >= 0 && !completed.includes(currentStep) ? counter : 0
   360→  const totalDisplay = status === 'running' ? completedMsSum + inFlightMs : totalMs
   361→
   362→  const statusDotCls = status === 'running'
   363→    ? 'bg-primary'
   364→    : status === 'completed'
   365→      ? 'bg-g-green'
   366→      : 'bg-white/40'
   367→
   368→  return (
   369→    <section className="bg-elyra-deep py-20 text-elyra-on-dark sm:py-28" aria-labelledby="sim-title">
   370→      <div className="elyra-container max-w-container">
   371→        <SectionHeading
   372→          kicker={t('kicker')}
   373→          title={t('title')}
   374→          subtitle={t('subtitle')}
   375→          variant="on-dark"
   376→          titleId="sim-title"
   377→        />
   378→
   379→        {showScenarioPicker ? (
   380→          // FIX(2-c/14): plain toggle-button group — the previous
   381→          // role="tablist"/"tab" markup had no tabpanels, no aria-controls
   382→          // and no roving tabindex, which is an incomplete (broken) tabs
   383→          // pattern. These are scenario switches, not tabs.
   384→          <div className="mt-8 flex flex-wrap justify-center gap-2">
   385→            {(['newOrder', 'paymentReminder', 'weeklyReport'] as ScenarioId[]).map((s) => (
   386→              <button
   387→                key={s}
   388→                type="button"
   389→                data-cursor="magnet"
   390→                aria-pressed={scenario === s}
   391→                onClick={() => setScenario(s)}
   392→                className={cn(
   393→                  'inline-flex h-10 items-center rounded-full px-4 text-sm font-medium transition-colors',
   394→                  scenario === s
   395→                    ? 'bg-primary text-primary-foreground'
   396→                    : 'border border-white/15 bg-white/5 text-white/80 hover:bg-white/10'
   397→                )}
   398→              >
   399→                {t(`scenarios.${s}.label`)}
   400→              </button>
   401→            ))}
   402→          </div>
   403→        ) : null}
   404→
   405→        {/* Stage — Phase 5 WS-7: data-cursor="inspect" so the magnetic
   406→            cursor shows the localized 'Inspect element' chip over the
   407→            n8n nodes panel (technical context, not just a magnet snap).
   408→            LOW-11: the min-w-[680px] stage overflows below ~712px, so the
   409→            wrapper is a labelled, focusable region (keyboard-scrollable

   412→          className="mt-10 overflow-x-auto scroll-dark no-scrollbar rounded-2xl focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring/50"
   413→          data-cursor="inspect"

   417→        >
   418→          <div
   419→            className="elyra-dotgrid relative min-w-[680px]"

   421→          >
   422→            {/* SVG edges */}
   423→            <svg

   454→            {/* Static stage skin + (motion-allowed only) keyframes. The

   457→            <style>{`

   467→                font-family: var(--font-mono);
   468→              }
   469→            `}</style>
   470→            {!reduced ? (

   472→                @keyframes elyra-flow { to { stroke-dashoffset: -24 } }

   498→
   499→            {/* UI-3: packet dots travel the active edge (below the nodes,

   501→            {!reduced
   502→              ? steps.slice(0, -1).map((_, i) => {

   507→                  return (

   512→                      style={{ '--from': `${from}%`, '--to': `${to}%` } as CSSProperties}
   513→                    />
