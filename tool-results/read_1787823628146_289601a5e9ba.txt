   102→const JSON_CLS = {
   103→  key: 'text-g-blue',
   104→  string: 'text-g-green',
   105→  number: 'text-g-yellow',
   106→  punct: 'text-white/40',

   109→interface JsonToken {
   110→  text: string
   111→  cls: string
   112→}
   113→
   114→function isJsonObject(value: JsonValue): value is { [key: string]: JsonValue } {
   115→  return typeof value === 'object'
   116→}
   117→
   118→function scalarToken(value: Exclude<JsonValue, { [key: string]: JsonValue }>): JsonToken {
   119→  if (typeof value === 'string') return { text: JSON.stringify(value), cls: JSON_CLS.string }
   120→  if (typeof value === 'number') return { text: String(value), cls: JSON_CLS.number }
   121→  return { text: String(value), cls: JSON_CLS.number } // boolean
   122→}
   123→
   124→/** Pretty-prints a JsonValue into syntax-colored display lines (2-space indent). */
   125→function jsonLines(value: JsonValue, indent = 0): JsonToken[][] {
   126→  if (!isJsonObject(value)) return [[scalarToken(value)]]
   127→  const pad = '  '.repeat(indent)
   128→  const lines: JsonToken[][] = [[{ text: '{', cls: JSON_CLS.punct }]]
   129→  const entries = Object.entries(value)
   130→  entries.forEach(([key, val], i) => {
   131→    const comma = i < entries.length - 1
   132→    const keyTok: JsonToken = { text: `${pad}  ${JSON.stringify(key)}`, cls: JSON_CLS.key }
   133→    if (isJsonObject(val)) {
   134→      lines.push([keyTok, { text: ': ', cls: JSON_CLS.punct }])
   135→      const nested = jsonLines(val, indent + 1)
   136→      if (comma) {
   137→        const lastLine = nested[nested.length - 1]
   138→        const lastTok = lastLine?.[lastLine.length - 1]
   139→        if (lastTok) lastTok.text += ','
   140→      }
   141→      nested.forEach((line) => lines.push(line))
   142→    } else {
   143→      const tok = scalarToken(val)
   144→      lines.push([
   145→        keyTok,
   146→        { text: ': ', cls: JSON_CLS.punct },
   147→        { text: tok.text + (comma ? ',' : ''), cls: tok.cls },
   148→      ])
   149→    }
   150→  })
   151→  lines.push([{ text: `${pad}}`, cls: JSON_CLS.punct }])
   152→  return lines
   153→}

   155→/** UI-3: wall-clock timestamp for log lines — HH:MM:SS.mmm, LTR-safe. */
   156→function logTimestamp(date: Date): string {
   157→  const p2 = (n: number) => String(n).padStart(2, '0')
   158→  const p3 = (n: number) => String(n).padStart(3, '0')
   159→  return `${p2(date.getHours())}:${p2(date.getMinutes())}:${p2(date.getSeconds())}.${p3(date.getMilliseconds())}`
   160→}

   311→          text: t(`scenarios.${scenario}.steps.${step.id}.title`),

   358→  const statusDotCls = status === 'running'
