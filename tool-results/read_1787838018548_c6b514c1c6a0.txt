    45→type UnknownRecord = Record<string, unknown>
    46→
    47→function isRecord(value: unknown): value is UnknownRecord {
    48→  return typeof value === 'object' && value !== null && !Array.isArray(value)
    49→}
    50→
    51→function asString(value: unknown): string {
    52→  return typeof value === 'string' ? value : ''
    53→}
    54→
    55→function asStringArray(value: unknown): string[] {
    56→  return Array.isArray(value) ? value.filter((x): x is string => typeof x === 'string') : []
    57→}
    58→
    59→interface Kpi {
    60→  label: string
    61→  value: string
    62→  delta: string
    63→}
    64→
    65→interface SheetRow {
    66→  ref: string
    67→  party: string
    68→  amount: string
    69→  status: string
    70→}
    71→
    72→function asKpis(value: unknown): Kpi[] {
    73→  if (!Array.isArray(value)) return []
    74→  return value
    75→    .filter(isRecord)
    76→    .map((r) => ({ label: asString(r.label), value: asString(r.value), delta: asString(r.delta) }))
    77→    .filter((k) => k.label !== '' || k.value !== '')
    78→}
    79→
    80→function asRows(value: unknown): SheetRow[] {
    81→  if (!Array.isArray(value)) return []
    82→  return value
    83→    .filter(isRecord)
    84→    .map((r) => ({
    85→      ref: asString(r.ref),
    86→      party: asString(r.party),
    87→      amount: asString(r.amount),
    88→      status: asString(r.status),
    89→    }))
    90→    .filter((r) => r.ref !== '' || r.party !== '')
    91→}

    93→/** Narrow a next-intl t.raw() catalog value into MockContent. */
    94→export function toMockContent(raw: unknown): MockContent | undefined {
    95→  if (!isRecord(raw)) return undefined
    96→  const cards = Array.isArray(raw.cards)
    97→    ? raw.cards
    98→        .filter(isRecord)
    99→        .map((c) => ({
   100→          name: asString(c.name),
   101→          price: asString(c.price),
   102→          old: asString(c.old) || undefined,
   103→        }))
   104→        .filter((c) => c.name !== '' || c.price !== '')
   105→    : undefined
   106→  return {
   107→    brand: asString(raw.brand) || undefined,
   108→    kicker: asString(raw.kicker) || undefined,
   109→    title: asString(raw.title) || undefined,
   110→    sub: asString(raw.sub) || undefined,
   111→    cta: asString(raw.cta) || undefined,
   112→    cards: cards && cards.length > 0 ? cards : undefined,
   113→  }
   114→}

   120→const FALLBACK_RGB: [number, number, number] = [0, 113, 227] // #0071E3 (brand blue)
   121→
   122→function hexToRgb(hex: string): [number, number, number] {
   123→  const m = /^#?([0-9a-f]{6})$/i.exec(hex.trim())
   124→  const digits = m?.[1]
   125→  if (!digits) return FALLBACK_RGB
   126→  const n = Number.parseInt(digits, 16)
   127→  return [(n >> 16) & 255, (n >> 8) & 255, n & 255]
   128→}
   129→
   130→function rgba(hex: string, alpha: number): string {
   131→  const [r, g, b] = hexToRgb(hex)
   132→  return `rgba(${r},${g},${b},${alpha})`
   133→}
   134→
   135→/** Mix toward black by `amount` (0..1). */
   136→function shadeColor(hex: string, amount: number): string {
   137→  const [r, g, b] = hexToRgb(hex)
   138→  const f = (c: number) => Math.round(c * (1 - amount))
   139→  return `rgb(${f(r)},${f(g)},${f(b)})`
   140→}
   141→
   142→/** Mix toward white by `amount` (0..1). */
   143→function tintColor(hex: string, amount: number): string {
   144→  const [r, g, b] = hexToRgb(hex)
   145→  const f = (c: number) => Math.round(c + (255 - c) * amount)
   146→  return `rgb(${f(r)},${f(g)},${f(b)})`
   147→}
   148→
   149→function luminance(hex: string): number {
   150→  const ch = (c: number) => {
   151→    const s = c / 255
   152→    return s <= 0.03928 ? s / 12.92 : ((s + 0.055) / 1.055) ** 2.4
   153→  }
   154→  const [r, g, b] = hexToRgb(hex)
   155→  return 0.2126 * ch(r) + 0.7152 * ch(g) + 0.0722 * ch(b)
   156→}
   157→
   158→const isLightColor = (hex: string) => luminance(hex) > 0.55
   159→
   160→/** Readable ink on top of a solid accent fill (white on dark accents,
   161→ *  darkened accent on light ones like the gold #FBBC05). */
   162→const onAccentColor = (hex: string) => (isLightColor(hex) ? shadeColor(hex, 0.6) : '#ffffff')
   163→
   164→/** Readable ink on top of an accent TINT (announcement bar, chips). */
   165→const accentInkColor = (hex: string) => shadeColor(hex, isLightColor(hex) ? 0.5 : 0.15)

   167→/** "−18%"-style discount derived from the price pair (null → no badge). */
   168→function discountPct(price: string, old: string | undefined): number | null {
   169→  if (!old) return null
   170→  const p = Number.parseInt(price.replace(/[^0-9]/g, ''), 10)
   171→  const o = Number.parseInt(old.replace(/[^0-9]/g, ''), 10)
   172→  if (!Number.isFinite(p) || !Number.isFinite(o) || o <= 0 || p >= o) return null
   173→  return Math.round((1 - p / o) * 100)
   174→}

   198→function smoothLinePath(pts: ReadonlyArray<readonly [number, number]>): string {
   199→  const first = pts[0]
   200→  if (pts.length < 2 || !first) return ''
   201→  let d = `M ${first[0]} ${first[1]}`
   202→  for (let i = 1; i < pts.length - 1; i++) {
   203→    const cur = pts[i]
   204→    const next = pts[i + 1]
   205→    if (!cur || !next) break
   206→    d += ` Q ${cur[0]} ${cur[1]} ${(cur[0] + next[0]) / 2} ${(cur[1] + next[1]) / 2}`
   207→  }
   208→  const last = pts[pts.length - 1]
   209→  if (!last) return d
   210→  d += ` L ${last[0]} ${last[1]}`
   211→  return d
   212→}

   962→/* --------------------------------------------------------------------------
   963→   Scene dispatcher — purely decorative content (aria-hidden); the slider
   964→   itself carries its own accessible name/controls.
   965→   -------------------------------------------------------------------------- */
   966→
   967→function Scene({ variant, accent, mock }: { variant: SceneVariant; accent: string; mock?: MockContent }) {
   968→  return (
   969→    <div className="size-full" aria-hidden="true">
   970→      {variant === 'site-new' && <SiteNewScene accent={accent} mock={mock} />}
   971→      {variant === 'site-old' && <SiteOldScene />}
   972→      {variant === 'dashboard-new' && <DashNewScene accent={accent} brand={mock?.brand} />}
   973→      {variant === 'dashboard-old' && <DashOldScene />}
   974→    </div>
   975→  )
   976→}
   977→
   978→interface BeforeAfterProps {
   979→  variant: SceneVariant
   980→  accent?: string
   981→  className?: string
   982→  /** aria label prefix — REQUIRED (FIX 2-c/17): every call site passes a
   983→   *  localized project title; no untranslated default literal. */
   984→  label: string
   985→  /** Per-project mock content for the "after" scene (see MockContent). */
   986→  mock?: MockContent
   987→}
   988→
   989→export function BeforeAfter({
   990→  variant,
   991→  accent = '#0071E3',
   992→  className,
   993→  label,
   994→  mock,
   995→}: BeforeAfterProps) {
   996→  const t = useTranslations('workSection')
   997→  const tc = useTranslations('common') // WS-2: cursor context label
   998→  const reduced = usePrefersReducedMotion()
   999→  const isRtl = useIsRtl()
  1000→  const containerRef = useRef<HTMLDivElement>(null)
  1001→  const [pos, setPos] = useState(50) // 0-100
  1002→  const dragging = useRef(false)
  1003→
  1004→  const beforeVariant: SceneVariant = variant.includes('site') ? 'site-old' : 'dashboard-old'
  1005→  const afterVariant = variant
  1006→
  1007→  const setFromClientX = useCallback((clientX: number) => {
  1008→    const el = containerRef.current
  1009→    if (!el) return
  1010→    const rect = el.getBoundingClientRect()
  1011→    // pos is measured from the START edge (left in LTR, right in RTL) so the
  1012→    // reveal direction follows the reading direction (audit P1-13).
  1013→    const x = isRtl ? rect.right - clientX : clientX - rect.left
  1014→    const pct = (x / rect.width) * 100
  1015→    setPos(Math.max(2, Math.min(98, pct)))
  1016→  }, [isRtl])
  1017→
  1018→  const onPointerDown = (e: React.PointerEvent) => {
  1019→    dragging.current = true
  1020→    ;(e.currentTarget as HTMLElement).setPointerCapture?.(e.pointerId)
  1021→    setFromClientX(e.clientX)
  1022→  }
  1023→  const onPointerMove = (e: React.PointerEvent) => {
  1024→    if (!dragging.current) return
  1025→    setFromClientX(e.clientX)
  1026→  }
  1027→  const onPointerUp = () => { dragging.current = false }
  1028→
  1029→  const onKeyDown = (e: React.KeyboardEvent) => {
  1030→    const step = e.shiftKey ? 10 : 5
  1031→    if (e.key === 'ArrowLeft') {
  1032→      e.preventDefault()
  1033→      setPos((p) => Math.max(2, p - (isRtl ? -step : step)))
  1034→    } else if (e.key === 'ArrowRight') {
  1035→      e.preventDefault()
  1036→      setPos((p) => Math.min(98, p + (isRtl ? -step : step)))
  1037→    } else if (e.key === 'Home') {
  1038→      e.preventDefault()
  1039→      setPos(2)
  1040→    } else if (e.key === 'End') {
  1041→      e.preventDefault()
  1042→      setPos(98)
  1043→    }
  1044→  }
  1045→
  1046→  // Clip is mirrored in RTL: pos counts from the START edge (right), so the
  1047→  // "after" layer reveals from the LEFT — matching RTL reading order where
  1048→  // "before" sits on the right and "after" on the left (audit P1-13).
  1049→  const clipAfter = isRtl ? `inset(0 ${pos}% 0 0)` : `inset(0 0 0 ${pos}%)`
  1050→
  1051→  return (
  1052→    <div
  1053→      ref={containerRef}
  1054→      data-cursor="drag"
  1055→      data-cursor-label={tc('cursor.drag')}
  1056→      className={cn(
  1057→        'relative aspect-[16/10] w-full select-none overflow-hidden rounded-2xl border border-border bg-card touch-none',
  1058→        className
  1059→      )}
  1060→      onPointerDown={onPointerDown}
  1061→      onPointerMove={onPointerMove}
  1062→      onPointerUp={onPointerUp}
  1063→      onPointerCancel={onPointerUp}
  1064→    >
  1065→      {/* scene-local keyframes (frozen by the global reduced-motion switch) */}
  1066→      <style>{SCENE_KEYFRAMES}</style>
  1067→
  1068→      {/* before layer (bottom) */}
  1069→      <div className="absolute inset-0">
  1070→        <Scene variant={beforeVariant} accent={accent} />
  1071→        <span className="absolute start-2 top-2 rounded-md bg-black/50 px-1.5 py-0.5 text-[10px] font-medium text-white backdrop-blur-sm">
  1072→          {t('before')}
  1073→        </span>
  1074→      </div>
  1075→
  1076→      {/* after layer (top, clipped) */}
  1077→      <div
  1078→        className="absolute inset-0"
  1079→        style={{ clipPath: clipAfter, WebkitClipPath: clipAfter }}
  1080→      >
  1081→        <Scene variant={afterVariant} accent={accent} mock={mock} />
  1082→        <span className="absolute end-2 top-2 rounded-md bg-primary px-1.5 py-0.5 text-[10px] font-medium text-primary-foreground backdrop-blur-sm">
  1083→          {t('after')}
  1084→        </span>
  1085→      </div>
  1086→
  1087→      {/* handle — anchored to the START edge (right in RTL) */}
  1088→      <div
  1089→        className="absolute inset-y-0 z-10 w-0.5 bg-white/80"
  1090→        style={isRtl ? { right: `${pos}%` } : { left: `${pos}%` }}
  1091→        aria-hidden="true"
  1092→      >
  1093→        <div className="absolute top-1/2 left-1/2 flex size-9 -translate-x-1/2 -translate-y-1/2 items-center justify-center rounded-full border border-white/30 bg-white/90 text-elyra-dark shadow-lg">
  1094→          <MoveHorizontal className="size-4" />
  1095→        </div>
  1096→      </div>
  1097→
  1098→      {/* accessible slider control (invisible but focusable) — bounds
  1099→          match the actual clamp range (FIX 2-c/17: was 0–100 while the
  1100→          pointer/keyboard logic clamps to 2–98). */}
  1101→      <div
  1102→        role="slider"
  1103→        tabIndex={0}
  1104→        aria-valuemin={2}
  1105→        aria-valuemax={98}
  1106→        aria-valuenow={Math.round(pos)}
  1107→        aria-label={`${label}: ${t('dragHint')}`}
  1108→        aria-valuetext={`${Math.round(pos)}%`}
  1109→        onKeyDown={onKeyDown}
  1110→        className="absolute inset-y-0 z-20 w-2 cursor-ew-resize focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring focus-visible:ring-offset-2"
  1111→        style={{
  1112→          ...(isRtl ? { right: `calc(${pos}% - 4px)` } : { left: `calc(${pos}% - 4px)` }),
  1113→          touchAction: 'none',
  1114→        }}
  1115→      />
  1116→
  1117→      {/* hint — CSS keyframes cycle (.ba-hint), framer-free (§4.3) */}
  1118→      {reduced ? null : (
  1119→        <div className="ba-hint pointer-events-none absolute bottom-2 left-1/2 -translate-x-1/2 rounded-full bg-black/50 px-2.5 py-1 text-[10px] text-white/90 backdrop-blur-sm">
  1120→          {t('dragHint')}
  1121→        </div>
  1122→      )}
  1123→    </div>
  1124→  )
  1125→}
