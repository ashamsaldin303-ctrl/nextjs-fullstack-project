     1→'use client'
     2→
     3→import { useMemo, useRef, useState } from 'react'
     4→import { useLocale, useTranslations } from 'next-intl'
     5→import { AnimatePresence, motion, useReducedMotion } from 'framer-motion'
     6→import { z } from 'zod'
     7→import {
     8→  Globe, Workflow, Boxes, Check, ArrowLeft, ArrowRight,
     9→  Send, RotateCw, AlertCircle,
    10→} from 'lucide-react'
    11→import {
    12→  Slider,
    13→} from '@/components/ui/slider'
    14→import { Input } from '@/components/ui/input'
    15→import { Label } from '@/components/ui/label'
    16→import { Button } from '@/components/ui/button'
    17→import { cn } from '@/lib/utils'
    18→import { SectionHeading } from '@/components/shared/section-heading'
    19→import { leadEmailSchema, leadNameSchema, leadWhatsappSchema } from '@/lib/lead-fields'
    20→import { playSuccess } from '@/lib/sound'
    21→import { toast } from 'sonner'
    22→import { RingGauge } from './ring-gauge'
    23→import {
    24→  computeEstimate, formatMoney,
    25→  type CalculatorInput, type ServiceType, type IntegrationKey,
    26→  type AutomationLevel, type LanguageOption, type ThreeDOption, type Locale,
    27→} from '@/lib/calculator'
    28→
    29→type Step = 0 | 1 | 2
    30→
    31→const SERVICES: { id: ServiceType; icon: typeof Globe }[] = [
    32→  { id: 'website', icon: Globe },
    33→  { id: 'automation', icon: Workflow },
    34→  { id: 'full', icon: Boxes },
    35→]
    36→
    37→const INTEGRATIONS: IntegrationKey[] = ['crm', 'invoicing', 'email', 'telegram', 'sheets', 'ai']
    38→
    39→// Shared lead-field schemas (R2-MED-1 / R6-LOW-3): the SAME rules the
    40→// API enforces — name 2–100, email ≤254, whatsapp 5–30 + phone pattern —
    41→// so a client-side rejection can never diverge from a server-side one.
    42→const leadSchema = z.object({
    43→  name: leadNameSchema,
    44→  email: leadEmailSchema,
    45→  whatsapp: leadWhatsappSchema,
    46→})
    47→
    48→type LeadForm = z.infer<typeof leadSchema>
    49→
    50→const INITIAL_INPUT: CalculatorInput = {
    51→  service: 'website',
    52→  pages: 6,
    53→  languages: 'bilingual',
    54→  threeD: 'no',
    55→  integrations: [],
    56→  automationLevel: 'essential',
    57→}
    58→
    59→export function Calculator() {
    60→  const t = useTranslations('calculator')
    61→  // Whatsapp has no local calculator.errors key — reuse the API's own
    62→  // translated field copy (apiErrors.fields.whatsapp) so client and
    63→  // server rejections read identically (same rule, same message).
    64→  const tApiFields = useTranslations('apiErrors.fields')
    65→  // useLocale() returns a broad `string`; narrow to the routing union
    66→  // (unknown values fall back to `ar`, the site default) so formatMoney's
    67→  // tightened `Locale` param type-checks.
    68→  const localeRaw = useLocale()
    69→  const locale: Locale = localeRaw === 'en' ? 'en' : 'ar'
    70→  const reduced = useReducedMotion()
    71→
    72→  const [step, setStep] = useState<Step>(0)
    73→  const [dir, setDir] = useState<1 | -1>(1)
    74→  const [input, setInput] = useState<CalculatorInput>(INITIAL_INPUT)
    75→  const [form, setForm] = useState<LeadForm>({ name: '', email: '', whatsapp: '' })
    76→  const [errors, setErrors] = useState<{ name?: string; email?: string; whatsapp?: string }>({})
    77→  const [submitting, setSubmitting] = useState(false)
    78→  const [done, setDone] = useState(false)
    79→  const [reference, setReference] = useState<string | null>(null)
    80→  // FIX(2-c/18): honeypot trap — bots autofill hidden "companyWebsite"
    81→  // fields; humans never see it. The value rides along in the JSON body
    82→  // and the API silently discards bot submissions with a fake success.
    83→  const honeypotRef = useRef<HTMLInputElement>(null)
    84→
    85→  const result = useMemo(() => computeEstimate(input), [input])
    86→
    87→  // Breakdown line labels — resolved once for type-safety (guide §4.6).
    88→  const breakdownLabels = {
    89→    base: t('result.base'),
    90→    pages: t('result.pages'),
    91→    bilingual: t('result.bilingual'),
    92→    threeD: t('result.threeD'),
    93→    integrations: t('result.integrations'),
    94→    advanced: t('result.advanced'),
    95→  } as const
    96→
    97→  const goNext = () => { setDir(1); setStep((s) => Math.min(2, s + 1) as Step) }
    98→  const goBack = () => { setDir(-1); setStep((s) => Math.max(0, s - 1) as Step) }
    99→
   100→  const toggleIntegration = (key: IntegrationKey) => {
   101→    setInput((p) => ({
   102→      ...p,
   103→      integrations: p.integrations.includes(key)
   104→        ? p.integrations.filter((k) => k !== key)
   105→        : [...p.integrations, key],
   106→    }))
   107→  }
   108→
   109→  const onSubmit = async (e: React.FormEvent) => {
   110→    e.preventDefault()
   111→    const parsed = leadSchema.safeParse(form)
   112→    if (!parsed.success) {
   113→      const fe: { name?: string; email?: string; whatsapp?: string } = {}
   114→      for (const issue of parsed.error.issues) {
   115→        const path = issue.path[0]
   116→        if (path === 'name') fe.name = t('errors.name')
   117→        if (path === 'email') fe.email = t('errors.email')
   118→        if (path === 'whatsapp') fe.whatsapp = tApiFields('whatsapp')
   119→      }
   120→      setErrors(fe)
   121→      return
   122→    }
   123→    setErrors({})
   124→    setSubmitting(true)
   125→    // Phase 3: real storage — the server recomputes the estimate from the
   126→    // wizard options and returns a reference (guide §2.9). Client numbers
   127→    // are never sent: only the option set travels.
   128→    try {
   129→      const res = await fetch('/api/leads', {
   130→        method: 'POST',
   131→        headers: {
   132→          'Content-Type': 'application/json',
   133→          'x-elyra-locale': locale,
   134→        },
   135→        body: JSON.stringify({
   136→          source: 'calculator',
   137→          companyWebsite: honeypotRef.current?.value ?? '',
   138→          name: parsed.data.name,
   139→          email: parsed.data.email,
   140→          whatsapp: parsed.data.whatsapp || undefined,
   141→          service: input.service,
   142→          pages: input.pages,
   143→          languages: input.languages,
   144→          threeD: input.threeD,
   145→          integrations: input.integrations,
   146→          automationLevel: input.automationLevel,
   147→        }),
   148→      })
   149→
   150→      if (res.status === 201) {
   151→        const data = (await res.json()) as { reference?: string }
   152→        setReference(data.reference ?? null)
   153→        setDone(true)
   154→        playSuccess() // Phase 2 sensory feedback — fires on REAL success only
   155→        return
   156→      }
   157→
   158→      // Server rejected: surface translated server-side messages.
   159→      const data = (await res.json().catch(() => null)) as
   160→        | { message?: string; fields?: Record<string, string> }
   161→        | null
   162→      if (res.status === 400 && data?.fields) {
   163→        const fe: { name?: string; email?: string; whatsapp?: string } = {}
   164→        if (data.fields.name) fe.name = data.fields.name
   165→        if (data.fields.email) fe.email = data.fields.email
   166→        if (data.fields.whatsapp) fe.whatsapp = data.fields.whatsapp
   167→        setErrors(fe)
   168→      }
   169→      toast.error(t('form.errorTitle'), {
   170→        description: data?.message ?? t('form.errorNetwork'),
   171→      })
   172→    } catch {
   173→      // Network failure — data stays in the form for a retry.
   174→      toast.error(t('form.errorTitle'), { description: t('form.errorNetwork') })
   175→    } finally {
   176→      setSubmitting(false)
   177→    }
   178→  }
   179→
   180→  const slideVariants = reduced
   181→    ? undefined
   182→    : {
   183→        enter: (d: number) => ({ opacity: 0, x: d * 30 }),
   184→        center: { opacity: 1, x: 0 },
   185→        exit: (d: number) => ({ opacity: 0, x: -d * 30 }),
   186→      }
   187→
   188→  return (
   189→    <section id="calculator" className="bg-background py-20 sm:py-28" aria-labelledby="calc-title" aria-label={t('kicker')}>
   190→      <div className="elyra-container max-w-5xl">
   191→        <SectionHeading
   192→          kicker={t('kicker')}
   193→          title={t('title')}
   194→          subtitle={t('subtitle')}
   195→          titleId="calc-title"
   196→        />
   197→
   198→        {/* Progress */}
   199→        <div className="mt-10 flex items-center justify-between gap-4">
   200→          <p className="text-sm text-muted-foreground">
   201→            {t('step', { current: step + 1, total: 3 })}
   202→          </p>
   203→          <div className="flex flex-1 gap-2">
   204→            {[0, 1, 2].map((s) => (
   205→              <div
   206→                key={s}
   207→                className="h-1.5 flex-1 overflow-hidden rounded-full bg-muted"
   208→                aria-hidden="true"
   209→              >
   210→                <motion.div
   211→                  className="h-full rounded-full bg-primary"
   212→                  initial={false}
   213→                  animate={{ width: step >= s ? '100%' : '0%' }}
   214→                  transition={{ duration: 0.4 }}
   215→                />
   216→              </div>
   217→            ))}
   218→          </div>
   219→        </div>
   220→
   221→        <div className="relative mt-8 overflow-hidden rounded-3xl border border-border bg-card p-6 sm:p-10">
   222→          <AnimatePresence mode="wait" custom={dir}>
   223→            <motion.div
   224→              key={step}
   225→              custom={dir}
   226→              variants={slideVariants}
   227→              initial="enter"
   228→              animate="center"
   229→              exit="exit"
   230→              transition={{ duration: 0.4, ease: [0.16, 1, 0.3, 1] }}
   231→            >
   232→              {/* STEP 0 — Service type */}
   233→              {step === 0 ? (
   234→                <div>
   235→                  <h3 className="text-lg font-semibold">{t('steps.service')}</h3>
   236→                  <div className="mt-6 grid gap-4 sm:grid-cols-3">
   237→                    {SERVICES.map(({ id, icon: Icon }) => {
   238→                      const active = input.service === id
   239→                      const isFull = id === 'full'
   240→                      return (
   241→                        <button
   242→                          key={id}
   243→                          type="button"
   244→                          data-cursor="magnet"
   245→                          onClick={() => setInput((p) => ({ ...p, service: id }))}
   246→                          aria-pressed={active}
   247→                          className={cn(
   248→                            'group relative overflow-hidden rounded-2xl border p-5 text-start transition-all',
   249→                            active
   250→                              ? 'border-primary bg-primary/5 shadow-[0_0_0_1px_var(--color-primary)]'
   251→                              : 'border-border hover:border-primary/40 hover:bg-foreground/[0.02]'
   252→                          )}
   253→                        >
   254→                          {isFull ? (
   255→                            <span className="absolute end-3 top-3 rounded-full bg-primary px-2 py-0.5 text-[10px] font-semibold text-primary-foreground">
   256→                              {t('services.full.popular')}
   257→                            </span>
   258→                          ) : null}
   259→                          <div className={cn(
   260→                            'flex size-11 items-center justify-center rounded-xl transition-colors',
   261→                            active ? 'bg-primary text-primary-foreground' : 'bg-primary/10 text-primary'
   262→                          )}>
   263→                            <Icon className="size-5" aria-hidden="true" />
   264→                          </div>
   265→                          <p className="mt-3 font-semibold">{t(`services.${id}.title`)}</p>
   266→                          <p className="mt-1 text-sm text-muted-foreground">{t(`services.${id}.desc`)}</p>
   267→                          {active ? (
   268→                            <span className="absolute -end-2 -top-2 flex size-6 items-center justify-center rounded-full bg-primary text-primary-foreground">
   269→                              <Check className="size-3.5" aria-hidden="true" />
   270→                            </span>
   271→                          ) : null}
   272→                        </button>
   273→                      )
   274→                    })}
   275→                  </div>
   276→                </div>
   277→              ) : null}
   278→
   279→              {/* STEP 1 — Features */}
   280→              {step === 1 ? (
   281→                <div className="space-y-8">
   282→                  <div>
   283→                    <div className="flex items-center justify-between">
   284→                      <h3 className="text-lg font-semibold">{t('pages')}</h3>
   285→                      <span className="text-sm font-medium text-primary tabular-nums">
   286→                        {t('pagesValue', { count: input.pages })}
   287→                      </span>
   288→                    </div>
   289→                    <div className="mt-4 px-1">
   290→                      <Slider
   291→                        value={[input.pages]}
   292→                        onValueChange={(v) => setInput((p) => ({ ...p, pages: v[0] ?? p.pages }))}
   293→                        min={1}
   294→                        max={20}
   295→                        step={1}
   296→                        aria-label={t('pages')}
   297→                      />
   298→                      <div className="mt-2 flex justify-between text-xs text-muted-foreground">
   299→                        <span>1</span><span>20</span>
   300→                      </div>
   301→                    </div>
   302→                  </div>
   303→
   304→                  <div>
   305→                    <h3 className="text-lg font-semibold">{t('languages')}</h3>
   306→                    <div className="mt-3 grid gap-3 sm:grid-cols-2">
   307→                      {(['single', 'bilingual'] as LanguageOption[]).map((opt) => {
   308→                        const active = input.languages === opt
   309→                        return (
   310→                          <button
   311→                            key={opt}
   312→                            type="button"
   313→                            onClick={() => setInput((p) => ({ ...p, languages: opt }))}
   314→                            aria-pressed={active}
   315→                            className={cn(
   316→                              'rounded-xl border px-4 py-3 text-start transition-colors',
   317→                              active ? 'border-primary bg-primary/5' : 'border-border hover:bg-foreground/[0.02]'
   318→                            )}
   319→                          >
   320→                            <span className="flex items-center gap-2">
   321→                              <span className={cn('flex size-5 items-center justify-center rounded-full border', active ? 'border-primary bg-primary text-primary-foreground' : 'border-border')}>
   322→                                {active ? <Check className="size-3" aria-hidden="true" /> : null}
   323→                              </span>
   324→                              <span className="text-sm font-medium">{t(`languagesOptions.${opt}`)}</span>
   325→                            </span>
   326→                          </button>
   327→                        )
   328→                      })}
   329→                    </div>
   330→                  </div>
   331→
   332→                  <div>
   333→                    <h3 className="text-lg font-semibold">{t('threeD')}</h3>
   334→                    <div className="mt-3 grid gap-3 sm:grid-cols-2">
   335→                      {(['yes', 'no'] as ThreeDOption[]).map((opt) => {
   336→                        const active = input.threeD === opt
   337→                        const disabled = input.service === 'automation'
   338→                        return (
   339→                          <button
   340→                            key={opt}
   341→                            type="button"
   342→                            disabled={disabled}
   343→                            onClick={() => setInput((p) => ({ ...p, threeD: opt }))}
   344→                            aria-pressed={active}
   345→                            className={cn(
   346→                              'rounded-xl border px-4 py-3 text-start transition-colors',
   347→                              disabled && 'opacity-50',
   348→                              active ? 'border-primary bg-primary/5' : 'border-border hover:bg-foreground/[0.02]'
   349→                            )}
   350→                          >
   351→                            <span className="flex items-center gap-2">
   352→                              <span className={cn('flex size-5 items-center justify-center rounded-full border', active ? 'border-primary bg-primary text-primary-foreground' : 'border-border')}>
   353→                                {active ? <Check className="size-3" aria-hidden="true" /> : null}
   354→                              </span>
   355→                              <span className="text-sm font-medium">{t(`threeDOptions.${opt}`)}</span>
   356→                            </span>
   357→                          </button>
   358→                        )
   359→                      })}
   360→                    </div>
   361→                  </div>
   362→
   363→                  <div>
   364→                    <h3 className="text-lg font-semibold">{t('integrations')}</h3>
   365→                    <div className="mt-3 flex flex-wrap gap-2">
   366→                      {INTEGRATIONS.map((key) => {
   367→                        const active = input.integrations.includes(key)
   368→                        return (
   369→                          <button
   370→                            key={key}
   371→                            type="button"
   372→                            onClick={() => toggleIntegration(key)}
   373→                            aria-pressed={active}
   374→                            className={cn(
   375→                              'inline-flex items-center gap-1.5 rounded-full border px-3 py-1.5 text-sm transition-colors',
   376→                              active ? 'border-primary bg-primary/10 text-primary' : 'border-border hover:bg-foreground/[0.02]'
   377→                            )}
   378→                          >
   379→                            <span className={cn('flex size-4 items-center justify-center rounded-full border', active ? 'border-primary bg-primary text-primary-foreground' : 'border-border')}>
   380→                              {active ? <Check className="size-2.5" aria-hidden="true" /> : null}
   381→                            </span>
   382→                            {t(`integrationsOptions.${key}`)}
   383→                          </button>
   384→                        )
   385→                      })}
   386→                    </div>
   387→                  </div>
   388→
   389→                  <div>
   390→                    <h3 className="text-lg font-semibold">{t('automationLevel')}</h3>
   391→                    <div className="mt-3 grid gap-3 sm:grid-cols-2">
   392→                      {(['essential', 'advanced'] as AutomationLevel[]).map((opt) => {
   393→                        const active = input.automationLevel === opt
   394→                        const disabled = input.service === 'website'
   395→                        return (
   396→                          <button
   397→                            key={opt}
   398→                            type="button"
   399→                            disabled={disabled}
   400→                            onClick={() => setInput((p) => ({ ...p, automationLevel: opt }))}
   401→                            aria-pressed={active}
   402→                            className={cn(
   403→                              'rounded-xl border px-4 py-3 text-start transition-colors',
   404→                              disabled && 'opacity-50',
   405→                              active ? 'border-primary bg-primary/5' : 'border-border hover:bg-foreground/[0.02]'
   406→                            )}
   407→                          >
   408→                            <span className="flex items-center gap-2">
   409→                              <span className={cn('flex size-5 items-center justify-center rounded-full border', active ? 'border-primary bg-primary text-primary-foreground' : 'border-border')}>
   410→                                {active ? <Check className="size-3" aria-hidden="true" /> : null}
   411→                              </span>
   412→                              <span className="text-sm font-medium">{t(`automationLevelOptions.${opt}`)}</span>
   413→                            </span>
   414→                          </button>
   415→                        )
   416→                      })}
   417→                    </div>
   418→                  </div>
   419→                </div>
   420→              ) : null}
   421→
   422→              {/* STEP 2 — Result */}
   423→              {step === 2 ? (
   424→                <div>
   425→                  {done ? (
   426→                    <div className="flex flex-col items-center py-8 text-center">
   427→                      <div className="flex size-16 items-center justify-center rounded-full bg-g-green/15 text-g-green">
   428→                        <Check className="size-8" aria-hidden="true" />
   429→                      </div>
   430→                      <h3 className="mt-5 text-2xl font-semibold">{t('form.successTitle')}</h3>
   431→                      <p className="mt-2 max-w-md text-muted-foreground">{t('form.successDesc')}</p>
   432→                      {reference ? (
   433→                        <p className="mt-3 rounded-full border border-g-green/30 bg-g-green/5 px-4 py-1.5 font-mono text-sm font-semibold tracking-wide text-g-green">
   434→                          {t('form.successReference', { reference })}
   435→                        </p>
   436→                      ) : null}
   437→                      <button
   438→                        type="button"
   439→                        onClick={() => { setDone(false); setReference(null); setStep(0); setInput(INITIAL_INPUT); setForm({ name: '', email: '', whatsapp: '' }) }}
   440→                        className="mt-6 inline-flex h-11 items-center gap-2 rounded-full border border-border px-5 text-sm font-medium hover:bg-foreground/5"
   441→                      >
   442→                        <RotateCw className="size-4" aria-hidden="true" />
   443→                        {t('form.successRestart')}
   444→                      </button>
   445→                    </div>
   446→                  ) : (
   447→                    <div className="grid gap-8 lg:grid-cols-2">
   448→                      <div>
   449→                        <div className="rounded-2xl border border-border bg-gradient-to-br from-primary/5 to-transparent p-6">
   450→                          {/* WS-4: animated SVG ring gauges */}
   451→                          <div className="mb-6 flex items-center justify-center gap-12">
   452→                            <RingGauge
   453→                              fraction={Math.min(1, result.max / 20000)}
   454→                              value={result.max}
   455→                              formatValue={(n) => formatMoney(Math.round(n), locale)}
   456→                              label={t('result.budget')}
   457→                              color="#0071E3"
   458→                              isRtl={locale === 'ar'}
   459→                            />
   460→                            <RingGauge
   461→                              fraction={Math.min(1, result.weeksMax / 12)}
   462→                              value={result.weeksMax}
   463→                              formatValue={(n) => `${Math.round(n)}`}
   464→                              label={t('result.duration')}
   465→                              color="#34A853"
   466→                              isRtl={locale === 'ar'}
   467→                            />
   468→                          </div>
   469→                          <p className="text-xs font-medium uppercase tracking-wider text-muted-foreground">
   470→                            {t('result.budget')}
   471→                          </p>
   472→                          <p className="mt-2 text-3xl font-bold tracking-tight text-primary sm:text-4xl">
   473→                            {formatMoney(result.min, locale)} – {formatMoney(result.max, locale)}
   474→                          </p>
   475→                          <p className="mt-4 text-xs font-medium uppercase tracking-wider text-muted-foreground">
   476→                            {t('result.duration')}
   477→                          </p>
   478→                          <p className="mt-1 text-xl font-semibold">
   479→                            {t('result.weeks', { min: result.weeksMin, max: result.weeksMax })}
   480→                          </p>
   481→                        </div>
   482→
   483→                        <div className="mt-4">
   484→                          <p className="text-sm font-semibold">{t('result.breakdownTitle')}</p>
   485→                          <ul className="mt-2 space-y-1.5 text-sm">
   486→                            {result.breakdown.map((line) => (
   487→                              <li key={line.labelKey} className="flex items-center justify-between gap-2">
   488→                                <span className="text-muted-foreground">{breakdownLabels[line.labelKey as keyof typeof breakdownLabels]}</span>
   489→                                <span className="tabular-nums">
   490→                                  {formatMoney(line.min, locale)} – {formatMoney(line.max, locale)}
   491→                                </span>
   492→                              </li>
   493→                            ))}
   494→                          </ul>
   495→                        </div>
   496→                        <p className="mt-4 flex items-start gap-1.5 text-xs text-muted-foreground">
   497→                          <AlertCircle className="mt-0.5 size-3.5 shrink-0" aria-hidden="true" />
   498→                          {t('result.disclaimer')}
   499→                        </p>
   500→                      </div>
