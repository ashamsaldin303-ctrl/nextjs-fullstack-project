    17→/** Bento card with cursor-following radial glow (works in RTL & LTR). */
    18→function GlowCard({
    19→  children,
    20→  className,
    21→}: {
    22→  children: React.ReactNode
    23→  className?: string
    24→}) {
    25→  const ref = useRef<HTMLDivElement>(null)
    26→  // FIX(2-c/4): rAF-coalesced — the rect read + --mx/--my writes happen
    27→  // ONCE per frame using the latest event, not once per pointermove.
    28→  // With no new events the callback never reschedules (idle at zero cost).
    29→  const latestPointer = useRef<React.PointerEvent<HTMLDivElement> | null>(null)
    30→  const glowRaf = useRef(0)
    31→  const onMove = useCallback((e: React.PointerEvent<HTMLDivElement>) => {
    32→    latestPointer.current = e
    33→    if (glowRaf.current) return
    34→    glowRaf.current = requestAnimationFrame(() => {
    35→      glowRaf.current = 0
    36→      const el = ref.current
    37→      const ev = latestPointer.current
    38→      if (!el || !ev) return
    39→      const rect = el.getBoundingClientRect()
    40→      el.style.setProperty('--mx', `${ev.clientX - rect.left}px`)
    41→      el.style.setProperty('--my', `${ev.clientY - rect.top}px`)
    42→    })
    43→  }, [])
    44→  useEffect(
    45→    () => () => {
    46→      if (glowRaf.current) cancelAnimationFrame(glowRaf.current)
    47→    },
    48→    []
    49→  )
    50→
    51→  return (
    52→    <div
    53→      ref={ref}
    54→      onPointerMove={onMove}
    55→      className={cn(
    56→        'glow-cursor group relative overflow-hidden rounded-3xl border border-border bg-card p-6 sm:p-8',
    57→        className
    58→      )}
    59→    >
    60→      {children}
    61→    </div>
    62→  )
    63→}

    69→const SITE_PALETTE = ['#0071E3', '#34A853', '#EA4335', '#4285F4'] as const
    70→
    71→/* Contrast-safe accent derivations (module scope, pure & deterministic —
    72→   identical on server & client, so no hydration risk).
    73→   The mini-site CTA / badges render WHITE text on the accent and several
    74→   strings render as accent-colored TEXT on the dark browser panel — neither
    75→   pair passes WCAG AA for every swatch (the lighter swatches being the
    76→   worst offenders: white on #4285F4 is ~3.1:1). Instead of a blanket dark
    77→   overlay, each fixed palette entry is mixed toward black (white-text
    78→   surfaces) or toward white (accent text) in 2% steps until the pair clears
    79→   4.54:1 (AA + margin). */
    80→
    81→function hexToRgb(hex: string): [number, number, number] {
    82→  const n = parseInt(hex.slice(1), 16)
    83→  return [(n >> 16) & 255, (n >> 8) & 255, n & 255]
    84→}
    85→function rgbToHex(r: number, g: number, b: number): string {
    86→  return `#${[r, g, b].map((v) => v.toString(16).padStart(2, '0')).join('')}`
    87→}
    88→function relLuminance([r, g, b]: [number, number, number]): number {
    89→  const lin = (v: number) => {
    90→    const s = v / 255
    91→    return s <= 0.04045 ? s / 12.92 : Math.pow((s + 0.055) / 1.055, 2.4)
    92→  }
    93→  return 0.2126 * lin(r) + 0.7152 * lin(g) + 0.0722 * lin(b)
    94→}
    95→function contrastRatio(a: number, b: number): number {
    96→  const hi = Math.max(a, b)
    97→  const lo = Math.min(a, b)
    98→  return (hi + 0.05) / (lo + 0.05)
    99→}
   100→function mixToward(hex: string, towardWhite: boolean, f: number): string {
   101→  const [r, g, b] = hexToRgb(hex)
   102→  const ch = (c: number) => Math.round(towardWhite ? c + (255 - c) * f : c * (1 - f))
   103→  return rgbToHex(ch(r), ch(g), ch(b))
   104→}
   105→/** Darken the accent until WHITE text on it clears AA — used for the CTA,
   106→ *  cart badge and discount badges (keeps #0071E3 untouched: 4.5:1 already). */
   107→function ctaSafe(hex: string): string {
   108→  for (let f = 0; f <= 0.9; f += 0.02) {
   109→    const c = mixToward(hex, false, f)
   110→    if (contrastRatio(relLuminance(hexToRgb(c)), 1) >= 4.54) return c
   111→  }
   112→  return mixToward(hex, false, 0.9)
   113→}
   114→/** Lighten the accent until it reads AA as TEXT on the dark panel. The
   115→ *  worst-case backdrop is the panel tinted 20% with the accent itself
   116→ *  (chips / addToCart tint) — bg-elyra-dark/95 over a white card ≈ #1B2335. */
   117→function panelTextSafe(hex: string): string {
   118→  const [pr, pg, pb] = hexToRgb('#1B2335')
   119→  const [ar, ag, ab] = hexToRgb(hex)
   120→  const tinted = relLuminance([
   121→    Math.round(pr * 0.8 + ar * 0.2),
   122→    Math.round(pg * 0.8 + ag * 0.2),
   123→    Math.round(pb * 0.8 + ab * 0.2),
   124→  ])
   125→  for (let f = 0; f <= 0.9; f += 0.02) {
   126→    const c = mixToward(hex, true, f)
   127→    if (contrastRatio(relLuminance(hexToRgb(c)), tinted) >= 4.54) return c
   128→  }
   129→  return mixToward(hex, true, 0.9)
   130→}
   131→const SITE_CTA = SITE_PALETTE.map(ctaSafe)
   132→const SITE_PANEL_TEXT = SITE_PALETTE.map(panelTextSafe)

   134→/** "349$" / "$349" → 19 (%). Returns 0 when there is no discount. */
   135→function discountPct(price: string, old: string): number {
   136→  const digits = (s: string) => Number(s.replace(/[^\d]/g, ''))
   137→  const p = digits(price)
   138→  const o = digits(old)
   139→  return o > p && p > 0 ? Math.round((1 - p / o) * 100) : 0
   140→}

   336→const FLOW_ICONS = [Inbox, Database, Send] as const
   337→
   338→function MiniFlow() {
   339→  const t = useTranslations('bento.automation.mini')
   340→  const nodes = t.raw('nodes') as string[]
   341→  const reduced = usePrefersReducedMotion()
   342→  const [state, setState] = useState<'idle' | 'running' | 'done'>('idle')
   343→  // `step` = index of the node currently processing (3 = all processed).
   344→  // `runId` remounts the traveling dots so their one-shot CSS animation
   345→  // replays on every run.
   346→  const [step, setStep] = useState(0)
   347→  const [runId, setRunId] = useState(0)
   348→  // FIX(2-c/5): keep every scheduled timeout cancellable — cleared before
   349→  // re-arming and on unmount so a late `done` can never fire on a dead card.
   350→  const timers = useRef<number[]>([])
   351→  const clearTimers = () => {
   352→    timers.current.forEach((id) => window.clearTimeout(id))
   353→    timers.current = []
   354→  }
   355→  useEffect(() => () => clearTimers(), [])
   356→
   357→  const run = () => {
   358→    if (state === 'running') return
   359→    clearTimers()
   360→    setRunId((n) => n + 1)
   361→    setState('running')
   362→    setStep(0)
   363→    const stepMs = reduced ? 400 : 700
   364→    timers.current = [
   365→      window.setTimeout(() => setStep(1), stepMs),
   366→      window.setTimeout(() => setStep(2), stepMs * 2),
   367→      window.setTimeout(() => {
   368→        setStep(3)
   369→        setState('done')
   370→      }, stepMs * 3),
   371→    ]
   372→  }
   373→
   374→  const nodeStatus = (i: number): 'idle' | 'active' | 'done' => {
   375→    if (state === 'done' || step > i) return 'done'
   376→    if (state === 'running' && step === i) return 'active'
   377→    return 'idle'
   378→  }
   379→  const swept = (i: number) => state === 'done' || step > i
   380→
   381→  return (
   382→    <div className="mt-6">
   383→      <p className="text-xs text-muted-foreground">{t('title')}</p>
   384→      <p className="mt-1 text-[11px] text-muted-foreground">{t('hint')}</p>
   385→      <div className="mt-4 rounded-xl border border-border bg-elyra-dark/95 p-3.5 text-elyra-on-dark">
   386→        <div className="flex items-start">
   387→          {nodes.map((label, i) => {
   388→            const status = nodeStatus(i)
   389→            const Icon = FLOW_ICONS[i] ?? Inbox
   390→            return (
   391→              <Fragment key={label}>
   392→                <div className="flex w-16 shrink-0 flex-col items-center gap-1.5">
   393→                  <div
   394→                    className={cn(
   395→                      'relative flex size-9 items-center justify-center rounded-full border-2 transition-colors duration-300',
   396→                      status === 'active' && 'border-primary bg-primary/25',
   397→                      status === 'done' && 'border-g-green/70 bg-g-green/15',
   398→                      status === 'idle' && 'border-white/15 bg-white/5'
   399→                    )}
   400→                  >
   401→                    <Icon
   402→                      className={cn(
   403→                        'size-4 transition-colors duration-300',
   404→                        status === 'active' && 'text-white',
   405→                        status === 'done' && 'text-g-green',
   406→                        status === 'idle' && 'text-white/50'
   407→                      )}
   408→                    />
   409→                    {status === 'done' ? (
   410→                      <span className="absolute -end-1 -top-1 flex size-3.5 items-center justify-center rounded-full bg-g-green text-elyra-dark">
   411→                        <Check className="size-2.5" />
   412→                      </span>
   413→                    ) : status === 'active' ? (
   414→                      // Pulse via pure CSS (.elyra-pulse) — Phase 3 §4.3;
   415→                      // reduced-motion gets a static shadow instead.
   416→                      <span
   417→                        className={cn('absolute inset-0 rounded-full', !reduced && 'elyra-pulse')}
   418→                        style={reduced ? { boxShadow: '0 0 0 4px rgba(0,113,227,0.35)' } : undefined}
   419→                      />
   420→                    ) : null}
   421→                  </div>
   422→                  <span
   423→                    className={cn(
   424→                      'text-center text-[9px] leading-tight transition-colors duration-300',
   425→                      status === 'active' && 'text-white',
   426→                      status === 'done' && 'text-white/80',
   427→                      status === 'idle' && 'text-white/70'
   428→                    )}
   429→                  >
   430→                    {label}
   431→                  </span>
   432→                </div>
   433→                {i < 2 ? (
   434→                  <div className="relative mx-0.5 mt-[17px] h-0.5 flex-1 rounded-full bg-white/10">
   435→                    {/* progress sweep */}
   436→                    <span
   437→                      className={cn(
   438→                        'absolute inset-0 rounded-full bg-primary transition-transform duration-500 ease-out',
   439→                        swept(i) ? 'scale-x-100' : 'scale-x-0'
   440→                      )}
   441→                      style={{ transformOrigin: 'start', transitionDelay: swept(i) ? '80ms' : '0ms' }}
   442→                    />
   443→                    {/* traveling pulse dot along the swept connector */}

   504→   (≈5.7°) pitch per press ≈ a 17px drag, through the SAME (unclamped)

   506→const KEY_YAW_DEG = (0.15 * 180) / Math.PI
   507→const KEY_PITCH_DEG = (0.1 * 180) / Math.PI

   509→function MiniCube() {
   510→  const t = useTranslations('bento.threeD.mini')
   511→  // Reduced motion: skip the CSS easing between drag rotations AND disable
   512→  // the idle auto-rotation entirely (user-driven drags still work).
   513→  const reduced = usePrefersReducedMotion()
   514→  const [rot, setRot] = useState({ x: -18, y: 28 })
   515→  const [dragging, setDragging] = useState(false)
   516→  const [spinning, setSpinning] = useState(false)
   517→  const start = useRef({ x: 0, y: 0, rx: 0, ry: 0 })
   518→  const idleTimer = useRef(0)
   519→
   520→  const stopSpin = useCallback(() => {
   521→    setSpinning(false) // the spin-loop effect cleans its own rAF on flip
   522→  }, [])
   523→  // Idle auto-rotation: armed on mount and after every drag; fires ~2.5s
   524→  // after the last interaction. Disabled entirely for reduced-motion users.
   525→  const armIdle = useCallback(() => {
   526→    if (idleTimer.current) window.clearTimeout(idleTimer.current)
   527→    if (reduced) return
   528→    idleTimer.current = window.setTimeout(() => {
   529→      idleTimer.current = 0
   530→      setSpinning(true)
   531→    }, 2500)
   532→  }, [reduced])
   533→
   534→  // Reduced-motion flip mid-spin: the loop effect below early-returns + its
   535→  // cleanup cancels the rAF, and armIdle() refuses to re-arm, so nothing
   536→  // autonomous keeps running — no setState needed here (lint: no cascading
   537→  // renders from effect bodies).
   538→  useEffect(() => {
   539→    if (reduced) return
   540→    armIdle()
   541→    return () => {
   542→      if (idleTimer.current) {
   543→        window.clearTimeout(idleTimer.current)
   544→        idleTimer.current = 0
   545→      }
   546→    }
   547→  }, [reduced, armIdle])
   548→
   549→  // The spin loop — time-based (display-rate independent), clamped against
   550→  // tab-visibility spikes, fully self-cleaning.
   551→  useEffect(() => {
   552→    if (!spinning || reduced) return
   553→    let raf = 0
   554→    let last = performance.now()
   555→    const tick = (now: number) => {
   556→      const dt = Math.min(now - last, 100)
   557→      last = now
   558→      setRot((r) => ({ x: r.x, y: (r.y + dt * 0.012) % 360 })) // ~12°/s
   559→      raf = requestAnimationFrame(tick)
   560→    }
   561→    raf = requestAnimationFrame(tick)
   562→    return () => {
   563→      if (raf) cancelAnimationFrame(raf)
   564→    }
   565→  }, [spinning, reduced])
   566→
   567→  const onDown = (e: React.PointerEvent) => {
   568→    stopSpin()
   569→    if (idleTimer.current) {
   570→      window.clearTimeout(idleTimer.current)
   571→      idleTimer.current = 0
   572→    }
   573→    start.current = { x: e.clientX, y: e.clientY, rx: rot.x, ry: rot.y }
   574→    setDragging(true)
   575→    ;(e.target as HTMLElement).setPointerCapture?.(e.pointerId)
   576→  }
   577→  const onMove = (e: React.PointerEvent) => {
   578→    if (!dragging) return
   579→    const dx = e.clientX - start.current.x
   580→    const dy = e.clientY - start.current.y
   581→    setRot({ x: start.current.rx - dy * 0.5, y: start.current.ry + dx * 0.5 })
   582→  }
   583→  const onUp = () => {
   584→    setDragging(false)
   585→    armIdle()
   586→  }
   587→
   588→  // L3 FIX (R3): keyboard path for the pointer-only drag (three-d-section
   589→  // pattern). Arrow keys nudge the SAME rotation state the drag writes
   590→  // (setRot) with the same delta signs as a same-direction drag (right →
   591→  // +yaw, down → −pitch). preventDefault sits AFTER the switch, so only
   592→  // handled arrows are swallowed. A press counts as an interaction: it
   593→  // stops the idle spin and re-arms it, exactly like onDown/onUp do.
   594→  const onKeyDown = (e: React.KeyboardEvent) => {
   595→    switch (e.key) {
   596→      case 'ArrowLeft':
   597→        setRot((r) => ({ x: r.x, y: r.y - KEY_YAW_DEG }))
   598→        break
   599→      case 'ArrowRight':
   600→        setRot((r) => ({ x: r.x, y: r.y + KEY_YAW_DEG }))
   601→        break
   602→      case 'ArrowUp':
   603→        setRot((r) => ({ x: r.x + KEY_PITCH_DEG, y: r.y }))
   604→        break
   605→      case 'ArrowDown':
   606→        setRot((r) => ({ x: r.x - KEY_PITCH_DEG, y: r.y }))
   607→        break
   608→      default:
   609→        return
   610→    }
   611→    e.preventDefault()
   612→    stopSpin()
   613→    armIdle()
   614→  }
   615→
   616→  // Cheap floor-shadow approximation: an ellipse that squashes/shifts with
   617→  // the cube's yaw so the ground reads as reacting to the rotation.
   618→  const ryRad = (rot.y * Math.PI) / 180
   619→  const shadowScale = 0.7 + 0.3 * Math.abs(Math.cos(ryRad))
   620→  const shadowShift = Math.sin(ryRad) * 7
   621→
   622→  return (
   623→    <div className="mt-6">
   624→      <p className="text-[11px] text-muted-foreground">
   625→        {t('hint')} · {t('idle')}
   626→      </p>
   627→      {/* L3 FIX (R3): the STAGE is now the keyboard surface — role="img" +
   628→          aria-label + tabIndex + arrow keys mirror three-d-section's a11y
   629→          pattern (label moved here from the cube so it names the focusable
   630→          element; children of role="img" are presentational). NOT gated on
   631→          reduced motion, unlike three-d-section: MiniCube drags still work
   632→          for reduced users (only autonomous spin + the 0.2s easing are
   633→          disabled), so keyboard rotation stays available too — reduced
   634→          users get instant, un-eased nudges. Focus ring follows the
   635→          three-d-section convention (ring-ring offset elyra-dark). */}
   636→      <div
   637→        role="img"
   638→        aria-label={`${t('hint')} — ${t('idle')}`}
   639→        tabIndex={0}
   640→        onKeyDown={onKeyDown}
   641→        className="relative mt-3 flex h-36 items-center justify-center overflow-hidden rounded-xl border border-border bg-elyra-dark/95 touch-none select-none focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring focus-visible:ring-offset-2 focus-visible:ring-offset-elyra-dark"
   642→        style={{ perspective: '600px' }}
   643→      >
   644→        {/* faint accent point-glow behind the cube (depth) */}
   645→        <div
   646→          aria-hidden="true"
   647→          className="absolute inset-0"
   648→          style={{ background: 'radial-gradient(130px 90px at 50% 44%, rgba(0,113,227,0.28), transparent 70%)' }}
   649→        />
   650→        {/* soft elliptical floor shadow */}
   651→        <div
   652→          aria-hidden="true"
   653→          className="absolute bottom-5 left-1/2 h-2.5 w-16 rounded-[50%] bg-black/70 blur-[5px] transition-transform duration-300"
   654→          style={{ transform: `translateX(calc(-50% + ${shadowShift}px)) scaleX(${shadowScale})` }}
   655→        />
   656→        <div
   657→          onPointerDown={onDown}
   658→          onPointerMove={onMove}
   659→          onPointerUp={onUp}
   660→          onPointerCancel={onUp}
   661→          className="size-14 cursor-grab touch-none active:cursor-grabbing"
   662→          style={{
   663→            transformStyle: 'preserve-3d',
   664→            transform: `rotateX(${rot.x}deg) rotateY(${rot.y}deg)`,
   665→            transition: dragging || spinning || reduced ? 'none' : 'transform 0.2s ease-out',
   666→            willChange: 'transform',
   667→          }}
   668→        >
   669→          {CUBE_FACES.map((f, i) => {
   670→            const Icon = f.icon
   671→            return (
   672→              <div
   673→                key={i}
   674→                className="absolute inset-0 flex items-center justify-center rounded-[3px]"
   675→                style={{
   676→                  transform: f.tr,
   677→                  background: f.grad,
   678→                  backfaceVisibility: 'hidden',
   679→                  // 1px inner border highlight so edges catch the light
   680→                  boxShadow: 'inset 0 0 0 1px rgba(255,255,255,0.22), inset 0 1px 0 rgba(255,255,255,0.35)',
   681→                }}
   682→              >
   683→                {Icon ? (
   684→                  <Icon
   685→                    className={cn('size-4', f.dark ? 'text-elyra-dark/80' : 'text-white/85')}
   686→                    aria-hidden="true"
   687→                  />
   688→                ) : (
   689→                  <span
   690→                    className={cn(
   691→                      f.glyph === 'E' ? 'text-base font-black tracking-wide text-white' : 'text-[10px] text-elyra-dark/70'
   692→                    )}
   693→                    aria-hidden="true"
   694→                  >
   695→                    {f.glyph}
   696→                  </span>
   697→                )}
   698→              </div>
   699→            )
   700→          })}
   701→        </div>
   702→      </div>
   703→    </div>
   704→  )
   705→}

   720→  // Six distinct localized answers — every click advances the index (cycling,
   721→  // so the same answer is never shown twice in a row). `shown` counts clicks.
   722→  const responses = t.raw('responses') as string[]
   723→  const [shown, setShown] = useState(-1)
   724→  const [phase, setPhase] = useState<'idle' | 'thinking' | 'typing'>('idle')

   727→  const [idea, setIdea] = useState('')
   728→  const len = Math.max(responses.length, 1)
   729→  const current = shown >= 0 ? responses[shown % len] ?? '' : ''
   730→  const complete = phase === 'typing' && current.length > 0 && chars >= current.length
   731→  // FIX(2-c/5): hold the typewriter interval + thinking timeout in refs —
   732→  // cleared before re-arming (a second click mid-type must not race the
   733→  // first) and on unmount.
   734→  const typer = useRef(0)
   735→  const thinker = useRef(0)
   736→  useEffect(
   737→    () => () => {
   738→      if (typer.current) window.clearInterval(typer.current)
   739→      if (thinker.current) window.clearTimeout(thinker.current)
   740→    },
   741→    []
   742→  )

   750→      window.clearTimeout(thinker.current)
   751→      thinker.current = 0
   752→    }

   754→    const text = responses[next % len] ?? ''

   758→      setPhase('typing')
   759→      setChars(text.length)
   760→      return
   761→    }

   764→      thinker.current = 0
   765→      setPhase('typing')

   767→      typer.current = window.setInterval(() => {
   768→        i += 2
   769→        setChars(i)
   770→        if (i >= text.length) {
   771→          window.clearInterval(typer.current)
   772→          typer.current = 0
   773→        }
   774→      }, 25)
   775→    }, 450)
   776→  }
   777→

   779→   *  when present, otherwise the currently shown answer. URL contract:

   795→          data-cursor="magnet"
   796→          onClick={ask}
   797→          className="flex min-h-11 w-full items-start gap-2 rounded-lg px-2 py-1.5 text-start text-xs text-white/80 hover:bg-white/5 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
   798→        >

   802→        {/* stable min-height keeps the card from jumping between the
   803→            thinking / typing / completed states */}

   805→          {phase === 'thinking' ? (
