    17→/** Bento card with cursor-following radial glow (works in RTL & LTR). */
    18→function GlowCard({
    19→  children,
    20→  className,
    21→}: {
    22→  children: React.ReactNode
    23→  className?: string
    24→}) {
    25→  const ref = useRef<HTMLDivElement>(null)
    26→  // FIX(2-c/4): rAF-coalesced — the rect read + --mx/--my writes happen
    27→  // ONCE per frame using the latest event, not once per pointermove.
    28→  // With no new events the callback never reschedules (idle at zero cost).
    29→  const latestPointer = useRef<React.PointerEvent<HTMLDivElement> | null>(null)
    30→  const glowRaf = useRef(0)
    31→  const onMove = useCallback((e: React.PointerEvent<HTMLDivElement>) => {
    32→    latestPointer.current = e
    33→    if (glowRaf.current) return
    34→    glowRaf.current = requestAnimationFrame(() => {
    35→      glowRaf.current = 0
    36→      const el = ref.current
    37→      const ev = latestPointer.current
    38→      if (!el || !ev) return
    39→      const rect = el.getBoundingClientRect()
    40→      el.style.setProperty('--mx', `${ev.clientX - rect.left}px`)
    41→      el.style.setProperty('--my', `${ev.clientY - rect.top}px`)
    42→    })
    43→  }, [])
    44→  useEffect(
    45→    () => () => {
    46→      if (glowRaf.current) cancelAnimationFrame(glowRaf.current)
    47→    },
    48→    []
    49→  )
    50→
    51→  return (
    52→    <div
    53→      ref={ref}
    54→      onPointerMove={onMove}
    55→      className={cn(
    56→        'glow-cursor group relative overflow-hidden rounded-3xl border border-border bg-card p-6 sm:p-8',
    57→        className
    58→      )}
    59→    >
    60→      {children}
    61→    </div>
    62→  )
    63→}

    69→const SITE_PALETTE = ['#0071E3', '#34A853', '#EA4335', '#4285F4'] as const
    70→
    71→/* Contrast-safe accent derivations (module scope, pure & deterministic —
    72→   identical on server & client, so no hydration risk).
    73→   The mini-site CTA / badges render WHITE text on the accent and several
    74→   strings render as accent-colored TEXT on the dark browser panel — neither
    75→   pair passes WCAG AA for every swatch (the lighter swatches being the
    76→   worst offenders: white on #4285F4 is ~3.1:1). Instead of a blanket dark
    77→   overlay, each fixed palette entry is mixed toward black (white-text
    78→   surfaces) or toward white (accent text) in 2% steps until the pair clears
    79→   4.54:1 (AA + margin). */
    80→
    81→function hexToRgb(hex: string): [number, number, number] {
    82→  const n = parseInt(hex.slice(1), 16)
    83→  return [(n >> 16) & 255, (n >> 8) & 255, n & 255]
    84→}
    85→function rgbToHex(r: number, g: number, b: number): string {
    86→  return `#${[r, g, b].map((v) => v.toString(16).padStart(2, '0')).join('')}`
    87→}
    88→function relLuminance([r, g, b]: [number, number, number]): number {
    89→  const lin = (v: number) => {
    90→    const s = v / 255
    91→    return s <= 0.04045 ? s / 12.92 : Math.pow((s + 0.055) / 1.055, 2.4)
    92→  }
    93→  return 0.2126 * lin(r) + 0.7152 * lin(g) + 0.0722 * lin(b)
    94→}
    95→function contrastRatio(a: number, b: number): number {
    96→  const hi = Math.max(a, b)
    97→  const lo = Math.min(a, b)
    98→  return (hi + 0.05) / (lo + 0.05)
    99→}
   100→function mixToward(hex: string, towardWhite: boolean, f: number): string {
   101→  const [r, g, b] = hexToRgb(hex)
   102→  const ch = (c: number) => Math.round(towardWhite ? c + (255 - c) * f : c * (1 - f))
   103→  return rgbToHex(ch(r), ch(g), ch(b))
   104→}
   105→/** Darken the accent until WHITE text on it clears AA — used for the CTA,
   106→ *  cart badge and discount badges (keeps #0071E3 untouched: 4.5:1 already). */
   107→function ctaSafe(hex: string): string {
   108→  for (let f = 0; f <= 0.9; f += 0.02) {
   109→    const c = mixToward(hex, false, f)
   110→    if (contrastRatio(relLuminance(hexToRgb(c)), 1) >= 4.54) return c
   111→  }
   112→  return mixToward(hex, false, 0.9)
   113→}
   114→/** Lighten the accent until it reads AA as TEXT on the dark panel. The
   115→ *  worst-case backdrop is the panel tinted 20% with the accent itself
   116→ *  (chips / addToCart tint) — bg-elyra-dark/95 over a white card ≈ #1B2335. */
   117→function panelTextSafe(hex: string): string {
   118→  const [pr, pg, pb] = hexToRgb('#1B2335')
   119→  const [ar, ag, ab] = hexToRgb(hex)
   120→  const tinted = relLuminance([
   121→    Math.round(pr * 0.8 + ar * 0.2),
   122→    Math.round(pg * 0.8 + ag * 0.2),
   123→    Math.round(pb * 0.8 + ab * 0.2),
   124→  ])
   125→  for (let f = 0; f <= 0.9; f += 0.02) {
   126→    const c = mixToward(hex, true, f)
   127→    if (contrastRatio(relLuminance(hexToRgb(c)), tinted) >= 4.54) return c
   128→  }
   129→  return mixToward(hex, true, 0.9)
   130→}
   131→const SITE_CTA = SITE_PALETTE.map(ctaSafe)
   132→const SITE_PANEL_TEXT = SITE_PALETTE.map(panelTextSafe)

   134→/** "349$" / "$349" → 19 (%). Returns 0 when there is no discount. */
   135→function discountPct(price: string, old: string): number {
   136→  const digits = (s: string) => Number(s.replace(/[^\d]/g, ''))
   137→  const p = digits(price)
   138→  const o = digits(old)
   139→  return o > p && p > 0 ? Math.round((1 - p / o) * 100) : 0
   140→}

   336→const FLOW_ICONS = [Inbox, Database, Send] as const
   337→
   338→function MiniFlow() {
   339→  const t = useTranslations('bento.automation.mini')
   340→  const nodes = t.raw('nodes') as string[]
   341→  const reduced = usePrefersReducedMotion()
   342→  const [state, setState] = useState<'idle' | 'running' | 'done'>('idle')
   343→  // `step` = index of the node currently processing (3 = all processed).
   344→  // `runId` remounts the traveling dots so their one-shot CSS animation
   345→  // replays on every run.
   346→  const [step, setStep] = useState(0)
   347→  const [runId, setRunId] = useState(0)
   348→  // FIX(2-c/5): keep every scheduled timeout cancellable — cleared before
   349→  // re-arming and on unmount so a late `done` can never fire on a dead card.
   350→  const timers = useRef<number[]>([])
   351→  const clearTimers = () => {
   352→    timers.current.forEach((id) => window.clearTimeout(id))
   353→    timers.current = []
   354→  }
   355→  useEffect(() => () => clearTimers(), [])
   356→
   357→  const run = () => {
   358→    if (state === 'running') return
   359→    clearTimers()
   360→    setRunId((n) => n + 1)
   361→    setState('running')
   362→    setStep(0)
   363→    const stepMs = reduced ? 400 : 700
   364→    timers.current = [
   365→      window.setTimeout(() => setStep(1), stepMs),
   366→      window.setTimeout(() => setStep(2), stepMs * 2),
   367→      window.setTimeout(() => {
   368→        setStep(3)
   369→        setState('done')
   370→      }, stepMs * 3),
   371→    ]
   372→  }
   373→
   374→  const nodeStatus = (i: number): 'idle' | 'active' | 'done' => {
   375→    if (state === 'done' || step > i) return 'done'
   376→    if (state === 'running' && step === i) return 'active'
   377→    return 'idle'
   378→  }
   379→  const swept = (i: number) => state === 'done' || step > i
   380→
   381→  return (
   382→    <div className="mt-6">
   383→      <p className="text-xs text-muted-foreground">{t('title')}</p>
   384→      <p className="mt-1 text-[11px] text-muted-foreground">{t('hint')}</p>
   385→      <div className="mt-4 rounded-xl border border-border bg-elyra-dark/95 p-3.5 text-elyra-on-dark">
   386→        <div className="flex items-start">
   387→          {nodes.map((label, i) => {
   388→            const status = nodeStatus(i)
   389→            const Icon = FLOW_ICONS[i] ?? Inbox
   390→            return (
   391→              <Fragment key={label}>
   392→                <div className="flex w-16 shrink-0 flex-col items-center gap-1.5">
   393→                  <div
   394→                    className={cn(
   395→                      'relative flex size-9 items-center justify-center rounded-full border-2 transition-colors duration-300',
   396→                      status === 'active' && 'border-primary bg-primary/25',
   397→                      status === 'done' && 'border-g-green/70 bg-g-green/15',
   398→                      status === 'idle' && 'border-white/15 bg-white/5'
   399→                    )}
   400→                  >
   401→                    <Icon
   402→                      className={cn(
   403→                        'size-4 transition-colors duration-300',
   404→                        status === 'active' && 'text-white',
   405→                        status === 'done' && 'text-g-green',
   406→                        status === 'idle' && 'text-white/50'
   407→                      )}
   408→                    />
   409→                    {status === 'done' ? (
   410→                      <span className="absolute -end-1 -top-1 flex size-3.5 items-center justify-center rounded-full bg-g-green text-elyra-dark">
   411→                        <Check className="size-2.5" />
   412→                      </span>
   413→                    ) : status === 'active' ? (
   414→                      // Pulse via pure CSS (.elyra-pulse) — Phase 3 §4.3;
   415→                      // reduced-motion gets a static shadow instead.
   416→                      <span
   417→                        className={cn('absolute inset-0 rounded-full', !reduced && 'elyra-pulse')}
   418→                        style={reduced ? { boxShadow: '0 0 0 4px rgba(0,113,227,0.35)' } : undefined}
   419→                      />
   420→                    ) : null}
   421→                  </div>
   422→                  <span
   423→                    className={cn(
   424→                      'text-center text-[9px] leading-tight transition-colors duration-300',
   425→                      status === 'active' && 'text-white',
   426→                      status === 'done' && 'text-white/80',
   427→                      status === 'idle' && 'text-white/70'
   428→                    )}
   429→                  >
   430→                    {label}
   431→                  </span>
   432→                </div>
   433→                {i < 2 ? (
   434→                  <div className="relative mx-0.5 mt-[17px] h-0.5 flex-1 rounded-full bg-white/10">
   435→                    {/* progress sweep */}
   436→                    <span
   437→                      className={cn(
   438→                        'absolute inset-0 rounded-full bg-primary transition-transform duration-500 ease-out',
   439→                        swept(i) ? 'scale-x-100' : 'scale-x-0'
   440→                      )}
   441→                      style={{ transformOrigin: 'start', transitionDelay: swept(i) ? '80ms' : '0ms' }}
   442→                    />
   443→                    {/* traveling pulse dot along the swept connector */}

   489→  grad: string
   490→  icon: typeof Globe | null
   491→  glyph?: string
   492→  dark?: boolean
   493→}> = [
   494→  { tr: 'translateZ(28px)', grad: 'linear-gradient(135deg, #4285F4, #0071E3)', icon: null, glyph: 'E' },

   499→  { tr: 'rotateX(-90deg) translateZ(28px)', grad: 'linear-gradient(135deg, #FDF8ED, #D9CDAC)', icon: null, glyph: '◆', dark: true },
   500→]
   501→
   502→function MiniCube() {
   503→  const t = useTranslations('bento.threeD.mini')
   504→  // Reduced motion: skip the CSS easing between drag rotations AND disable
   505→  // the idle auto-rotation entirely (user-driven drags still work).
   506→  const reduced = usePrefersReducedMotion()
   507→  const [rot, setRot] = useState({ x: -18, y: 28 })
   508→  const [dragging, setDragging] = useState(false)
   509→  const [spinning, setSpinning] = useState(false)
   510→  const start = useRef({ x: 0, y: 0, rx: 0, ry: 0 })
   511→  const idleTimer = useRef(0)
   512→
   513→  const stopSpin = useCallback(() => {
   514→    setSpinning(false) // the spin-loop effect cleans its own rAF on flip
   515→  }, [])
   516→  // Idle auto-rotation: armed on mount and after every drag; fires ~2.5s
   517→  // after the last interaction. Disabled entirely for reduced-motion users.
   518→  const armIdle = useCallback(() => {
   519→    if (idleTimer.current) window.clearTimeout(idleTimer.current)
   520→    if (reduced) return
   521→    idleTimer.current = window.setTimeout(() => {
   522→      idleTimer.current = 0
   523→      setSpinning(true)
   524→    }, 2500)
   525→  }, [reduced])
   526→
   527→  // Reduced-motion flip mid-spin: the loop effect below early-returns + its
   528→  // cleanup cancels the rAF, and armIdle() refuses to re-arm, so nothing
   529→  // autonomous keeps running — no setState needed here (lint: no cascading
   530→  // renders from effect bodies).
   531→  useEffect(() => {
   532→    if (reduced) return
   533→    armIdle()
   534→    return () => {
   535→      if (idleTimer.current) {
   536→        window.clearTimeout(idleTimer.current)
   537→        idleTimer.current = 0
   538→      }
   539→    }
   540→  }, [reduced, armIdle])
   541→
   542→  // The spin loop — time-based (display-rate independent), clamped against
   543→  // tab-visibility spikes, fully self-cleaning.
   544→  useEffect(() => {
   545→    if (!spinning || reduced) return
   546→    let raf = 0
   547→    let last = performance.now()
   548→    const tick = (now: number) => {
   549→      const dt = Math.min(now - last, 100)
   550→      last = now
   551→      setRot((r) => ({ x: r.x, y: (r.y + dt * 0.012) % 360 })) // ~12°/s
   552→      raf = requestAnimationFrame(tick)
   553→    }
   554→    raf = requestAnimationFrame(tick)
   555→    return () => {
   556→      if (raf) cancelAnimationFrame(raf)
   557→    }
   558→  }, [spinning, reduced])
   559→
   560→  const onDown = (e: React.PointerEvent) => {
   561→    stopSpin()
   562→    if (idleTimer.current) {
   563→      window.clearTimeout(idleTimer.current)
   564→      idleTimer.current = 0
   565→    }
   566→    start.current = { x: e.clientX, y: e.clientY, rx: rot.x, ry: rot.y }
   567→    setDragging(true)
   568→    ;(e.target as HTMLElement).setPointerCapture?.(e.pointerId)
   569→  }
   570→  const onMove = (e: React.PointerEvent) => {
   571→    if (!dragging) return
   572→    const dx = e.clientX - start.current.x
   573→    const dy = e.clientY - start.current.y
   574→    setRot({ x: start.current.rx - dy * 0.5, y: start.current.ry + dx * 0.5 })
   575→  }
   576→  const onUp = () => {
   577→    setDragging(false)
   578→    armIdle()
   579→  }
   580→
   581→  // Cheap floor-shadow approximation: an ellipse that squashes/shifts with
   582→  // the cube's yaw so the ground reads as reacting to the rotation.
   583→  const ryRad = (rot.y * Math.PI) / 180
   584→  const shadowScale = 0.7 + 0.3 * Math.abs(Math.cos(ryRad))
   585→  const shadowShift = Math.sin(ryRad) * 7
   586→
   587→  return (
   588→    <div className="mt-6">
   589→      <p className="text-[11px] text-muted-foreground">
   590→        {t('hint')} · {t('idle')}
   591→      </p>
   592→      <div
   593→        className="relative mt-3 flex h-36 items-center justify-center overflow-hidden rounded-xl border border-border bg-elyra-dark/95 touch-none select-none"
   594→        style={{ perspective: '600px' }}
   595→      >
   596→        {/* faint accent point-glow behind the cube (depth) */}
   597→        <div
   598→          aria-hidden="true"
   599→          className="absolute inset-0"
   600→          style={{ background: 'radial-gradient(130px 90px at 50% 44%, rgba(0,113,227,0.28), transparent 70%)' }}
   601→        />
   602→        {/* soft elliptical floor shadow */}
   603→        <div
   604→          aria-hidden="true"
   605→          className="absolute bottom-5 left-1/2 h-2.5 w-16 rounded-[50%] bg-black/70 blur-[5px] transition-transform duration-300"
   606→          style={{ transform: `translateX(calc(-50% + ${shadowShift}px)) scaleX(${shadowScale})` }}
   607→        />
   608→        <div
   609→          onPointerDown={onDown}
   610→          onPointerMove={onMove}
   611→          onPointerUp={onUp}
   612→          onPointerCancel={onUp}
   613→          className="size-14 cursor-grab touch-none active:cursor-grabbing"
   614→          style={{
   615→            transformStyle: 'preserve-3d',
   616→            transform: `rotateX(${rot.x}deg) rotateY(${rot.y}deg)`,
   617→            transition: dragging || spinning || reduced ? 'none' : 'transform 0.2s ease-out',
   618→            willChange: 'transform',
   619→          }}
   620→          role="img"
   621→          aria-label={`${t('hint')} — ${t('idle')}`}
   622→        >
   623→          {CUBE_FACES.map((f, i) => {
   624→            const Icon = f.icon
   625→            return (
   626→              <div
   627→                key={i}
   628→                className="absolute inset-0 flex items-center justify-center rounded-[3px]"
   629→                style={{
   630→                  transform: f.tr,
   631→                  background: f.grad,
   632→                  backfaceVisibility: 'hidden',
   633→                  // 1px inner border highlight so edges catch the light
   634→                  boxShadow: 'inset 0 0 0 1px rgba(255,255,255,0.22), inset 0 1px 0 rgba(255,255,255,0.35)',
   635→                }}
   636→              >
   637→                {Icon ? (
   638→                  <Icon
   639→                    className={cn('size-4', f.dark ? 'text-elyra-dark/80' : 'text-white/85')}
   640→                    aria-hidden="true"
   641→                  />
   642→                ) : (
   643→                  <span
   644→                    className={cn(
   645→                      f.glyph === 'E' ? 'text-base font-black tracking-wide text-white' : 'text-[10px] text-elyra-dark/70'
   646→                    )}
   647→                    aria-hidden="true"
   648→                  >
   649→                    {f.glyph}
   650→                  </span>
   651→                )}
   652→              </div>
   653→            )
   654→          })}
   655→        </div>
   656→      </div>
   657→    </div>
   658→  )
   659→}

   665→function MiniAgent() {
   666→  const t = useTranslations('bento.ai.mini')
   667→  // Batch 2 item 8: locale-aware router (@/i18n/navigation) — the convert
   668→  // CTA hands the visitor's idea to the prefilled contact form.
   669→  const router = useRouter()
   670→  // Typewriter is a JS-driven (setInterval) animation — the global CSS
   671→  // reduced-motion override can't stop it, so we honor the hook directly:
   672→  // reduced-motion users get the full response instantly (audit P1-6).
   673→  const reduced = usePrefersReducedMotion()
   674→  // Six distinct localized answers — every click advances the index (cycling,
   675→  // so the same answer is never shown twice in a row). `shown` counts clicks.
   676→  const responses = t.raw('responses') as string[]
   677→  const [shown, setShown] = useState(-1)
   678→  const [phase, setPhase] = useState<'idle' | 'thinking' | 'typing'>('idle')
   679→  const [chars, setChars] = useState(0)
   680→  // Batch 2 item 8: free-text idea field below the cycling answer.
   681→  const [idea, setIdea] = useState('')
   682→  const len = Math.max(responses.length, 1)
   683→  const current = shown >= 0 ? responses[shown % len] ?? '' : ''
   684→  const complete = phase === 'typing' && current.length > 0 && chars >= current.length
   685→  // FIX(2-c/5): hold the typewriter interval + thinking timeout in refs —
   686→  // cleared before re-arming (a second click mid-type must not race the
   687→  // first) and on unmount.
   688→  const typer = useRef(0)
   689→  const thinker = useRef(0)
   690→  useEffect(
   691→    () => () => {
   692→      if (typer.current) window.clearInterval(typer.current)
   693→      if (thinker.current) window.clearTimeout(thinker.current)
   694→    },
   695→    []
   696→  )
   697→
   698→  const ask = () => {
   699→    if (typer.current) {
   700→      window.clearInterval(typer.current)
   701→      typer.current = 0
   702→    }
   703→    if (thinker.current) {
   704→      window.clearTimeout(thinker.current)
   705→      thinker.current = 0
   706→    }
   707→    const next = shown + 1
   708→    const text = responses[next % len] ?? ''
   709→    setShown(next)
   710→    setChars(0)
   711→    if (reduced) {
   712→      setPhase('typing')
   713→      setChars(text.length)
   714→      return
   715→    }
   716→    setPhase('thinking')
   717→    thinker.current = window.setTimeout(() => {
   718→      thinker.current = 0
   719→      setPhase('typing')
   720→      let i = 0
   721→      typer.current = window.setInterval(() => {
   722→        i += 2
   723→        setChars(i)
   724→        if (i >= text.length) {
   725→          window.clearInterval(typer.current)
   726→          typer.current = 0
   727→        }
   728→      }, 25)
   729→    }, 450)
   730→  }
   731→
   732→  /** Batch 2 item 8: convert the intent into a request — the typed idea
   733→   *  when present, otherwise the currently shown answer. URL contract:
   734→   *  /contact?service=agent&idea=<urlencoded> (same locale-correct
   735→   *  router.push pattern as hero-console's goToContact). */
   736→  const convert = () => {
   737→    const text = idea.trim() || current
   738→    const params = new URLSearchParams({ service: 'agent' })
   739→    if (text) params.set('idea', text)
   740→    router.push(`/contact?${params.toString()}`)
   741→  }
   742→
   743→  return (
   744→    <div className="mt-6">
   745→      <p className="text-[11px] text-muted-foreground">{t('hint')}</p>
   746→      <div className="mt-3 space-y-2 rounded-xl border border-border bg-elyra-dark/95 p-3 text-elyra-on-dark">
   747→        <button
   748→          type="button"
   749→          data-cursor="magnet"
   750→          onClick={ask}
   751→          className="flex min-h-11 w-full items-start gap-2 rounded-lg px-2 py-1.5 text-start text-xs text-white/80 hover:bg-white/5 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
   752→        >
   753→          <Bot className="mt-0.5 size-3.5 shrink-0 text-primary" aria-hidden="true" />
   754→          <span>{t('prompt')}</span>
   755→        </button>
   756→        {/* stable min-height keeps the card from jumping between the
   757→            thinking / typing / completed states */}
   758→        <div className={shown >= 0 ? 'min-h-[3rem]' : undefined}>
   759→          {phase === 'thinking' ? (
   760→            <div className="flex items-center gap-1.5 rounded-lg bg-white/5 px-2 py-2 text-xs">
   761→              <Bot className="size-3.5 shrink-0 text-primary" aria-hidden="true" />
   762→              <span className="flex items-center gap-1 py-0.5" aria-hidden="true">
   763→                <span className="size-1 animate-bounce rounded-full bg-white/60" />
   764→                <span className="size-1 animate-bounce rounded-full bg-white/60 [animation-delay:150ms]" />
   765→                <span className="size-1 animate-bounce rounded-full bg-white/60 [animation-delay:300ms]" />
   766→              </span>
   767→            </div>
   768→          ) : current ? (
   769→            <div className="relative rounded-lg bg-white/5 px-2 py-1.5">
   770→              <span
   771→                aria-hidden="true"
   772→                className="absolute -end-1 -top-2 rounded-md bg-g-blue/20 px-1 py-px text-[9px] font-bold text-g-blue ring-1 ring-g-blue/30"
   773→              >
   774→                AI
   775→              </span>
   776→              <div className="flex items-start gap-2 text-xs text-white/90">
   777→                <Sparkles className="mt-0.5 size-3.5 shrink-0 text-g-yellow" aria-hidden="true" />
   778→                <span>
   779→                  {current.slice(0, chars)}
   780→                  {phase === 'typing' && chars < current.length ? (
   781→                    <span className="animate-pulse">▌</span>
   782→                  ) : null}
   783→                </span>
   784→              </div>
   785→              {complete ? (
   786→                <div className="mt-1 flex items-center gap-1 ps-5 text-[10px] text-white/50">
   787→                  <RefreshCw className="size-2.5" aria-hidden="true" />
   788→                  {t('retryHint')}
   789→                </div>
   790→              ) : null}
   791→            </div>
   792→          ) : null}
   793→        </div>
   794→        {/* one clean SR announcement per COMPLETED answer — keeps the
