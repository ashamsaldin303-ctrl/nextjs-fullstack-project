     1→'use client'
     2→
     3→import { useEffect, useMemo, useRef, useState } from 'react'
     4→import { useLocale, useTranslations } from 'next-intl'
     5→import { AnimatePresence, motion, useReducedMotion } from 'framer-motion'
     6→import { z } from 'zod'
     7→import {
     8→  Globe, Workflow, Boxes, Check, ArrowLeft, ArrowRight,
     9→  Send, RotateCw, AlertCircle,
    10→} from 'lucide-react'
    11→import {
    12→  Slider,
    13→} from '@/components/ui/slider'
    14→import { Input } from '@/components/ui/input'
    15→import { Label } from '@/components/ui/label'
    16→import { Button } from '@/components/ui/button'
    17→import { cn } from '@/lib/utils'
    18→import { SectionHeading } from '@/components/shared/section-heading'
    19→import { leadEmailSchema, leadNameSchema, leadWhatsappSchema } from '@/lib/lead-fields'
    20→import { playSuccess } from '@/lib/sound'
    21→import { toast } from 'sonner'
    22→import { RingGauge } from './ring-gauge'
    23→import {
    24→  computeEstimate, formatMoney,
    25→  type CalculatorInput, type ServiceType, type IntegrationKey,
    26→  type AutomationLevel, type LanguageOption, type ThreeDOption, type Locale,
    27→} from '@/lib/calculator'
    28→
    29→type Step = 0 | 1 | 2
    30→
    31→const SERVICES: { id: ServiceType; icon: typeof Globe }[] = [
    32→  { id: 'website', icon: Globe },
    33→  { id: 'automation', icon: Workflow },
    34→  { id: 'full', icon: Boxes },
    35→]
    36→
    37→const INTEGRATIONS: IntegrationKey[] = ['crm', 'invoicing', 'email', 'telegram', 'sheets', 'ai']
    38→
    39→// Shared lead-field schemas (R2-MED-1 / R6-LOW-3): the SAME rules the
    40→// API enforces — name 2–100, email ≤254, whatsapp 5–30 + phone pattern —
    41→// so a client-side rejection can never diverge from a server-side one.
    42→const leadSchema = z.object({
    43→  name: leadNameSchema,
    44→  email: leadEmailSchema,
    45→  whatsapp: leadWhatsappSchema,
    46→})
    47→
    48→type LeadForm = z.infer<typeof leadSchema>
    49→
    50→const INITIAL_INPUT: CalculatorInput = {
    51→  service: 'website',
    52→  pages: 6,
    53→  languages: 'bilingual',
    54→  threeD: 'no',
    55→  integrations: [],
    56→  automationLevel: 'essential',
    57→}
    58→
    59→export function Calculator() {
    60→  const t = useTranslations('calculator')
    61→  // Whatsapp has no local calculator.errors key — reuse the API's own
    62→  // translated field copy (apiErrors.fields.whatsapp) so client and
    63→  // server rejections read identically (same rule, same message).
    64→  const tApiFields = useTranslations('apiErrors.fields')
    65→  // useLocale() returns a broad `string`; narrow to the routing union
    66→  // (unknown values fall back to `ar`, the site default) so formatMoney's
    67→  // tightened `Locale` param type-checks.
    68→  const localeRaw = useLocale()
    69→  const locale: Locale = localeRaw === 'en' ? 'en' : 'ar'
    70→  const reduced = useReducedMotion()
    71→
    72→  const [step, setStep] = useState<Step>(0)
    73→  const [dir, setDir] = useState<1 | -1>(1)
    74→  const [input, setInput] = useState<CalculatorInput>(INITIAL_INPUT)
    75→  const [form, setForm] = useState<LeadForm>({ name: '', email: '', whatsapp: '' })
    76→  const [errors, setErrors] = useState<{ name?: string; email?: string; whatsapp?: string }>({})
    77→  const [submitting, setSubmitting] = useState(false)
    78→  const [done, setDone] = useState(false)
    79→  const [reference, setReference] = useState<string | null>(null)
    80→  // FIX(2-c/18): honeypot trap — bots autofill hidden "companyWebsite"
    81→  // fields; humans never see it. The value rides along in the JSON body
    82→  // and the API silently discards bot submissions with a fake success.
    83→  const honeypotRef = useRef<HTMLInputElement>(null)
    84→
    85→  const result = useMemo(() => computeEstimate(input), [input])
    86→
    87→  // L3 FIX (R5): on 201 the `done` branch swaps the form (whose focused
    88→  // submit button just disappeared) for the success panel — focus fell to
    89→  // <body> and nothing was announced. Move focus to the success heading
    90→  // (tabIndex={-1} = programmatically focusable, out of tab order). The
    91→  // effect runs after the swap commits, so the ref is attached when it
    92→  // fires; done can only flip at step 2 (the form lives there).
    93→  const successHeadingRef = useRef<HTMLHeadingElement>(null)
    94→  useEffect(() => {
    95→    if (done) successHeadingRef.current?.focus()
    96→  }, [done])
    97→
    98→  // Breakdown line labels — resolved once for type-safety (guide §4.6).
    99→  const breakdownLabels = {
   100→    base: t('result.base'),
   101→    pages: t('result.pages'),
   102→    bilingual: t('result.bilingual'),
   103→    threeD: t('result.threeD'),
   104→    integrations: t('result.integrations'),
   105→    advanced: t('result.advanced'),
   106→  } as const
   107→
   108→  const goNext = () => { setDir(1); setStep((s) => Math.min(2, s + 1) as Step) }
   109→  const goBack = () => { setDir(-1); setStep((s) => Math.max(0, s - 1) as Step) }
   110→
   111→  const toggleIntegration = (key: IntegrationKey) => {
   112→    setInput((p) => ({
   113→      ...p,
   114→      integrations: p.integrations.includes(key)
   115→        ? p.integrations.filter((k) => k !== key)
   116→        : [...p.integrations, key],
   117→    }))
   118→  }
   119→
   120→  const onSubmit = async (e: React.FormEvent) => {
   121→    e.preventDefault()
   122→    const parsed = leadSchema.safeParse(form)
   123→    if (!parsed.success) {
   124→      const fe: { name?: string; email?: string; whatsapp?: string } = {}
   125→      for (const issue of parsed.error.issues) {
   126→        const path = issue.path[0]
   127→        if (path === 'name') fe.name = t('errors.name')
   128→        if (path === 'email') fe.email = t('errors.email')
   129→        if (path === 'whatsapp') fe.whatsapp = tApiFields('whatsapp')
   130→      }
   131→      setErrors(fe)
   132→      return
   133→    }
   134→    setErrors({})
   135→    setSubmitting(true)
   136→    // Phase 3: real storage — the server recomputes the estimate from the
   137→    // wizard options and returns a reference (guide §2.9). Client numbers
   138→    // are never sent: only the option set travels.
   139→    try {
   140→      const res = await fetch('/api/leads', {
   141→        method: 'POST',
   142→        headers: {
   143→          'Content-Type': 'application/json',
   144→          'x-elyra-locale': locale,
   145→        },
   146→        body: JSON.stringify({
   147→          source: 'calculator',
   148→          companyWebsite: honeypotRef.current?.value ?? '',
   149→          name: parsed.data.name,
   150→          email: parsed.data.email,
   151→          whatsapp: parsed.data.whatsapp || undefined,
   152→          service: input.service,
   153→          pages: input.pages,
   154→          languages: input.languages,
   155→          threeD: input.threeD,
   156→          integrations: input.integrations,
   157→          automationLevel: input.automationLevel,
   158→        }),
   159→      })
   160→
   161→      if (res.status === 201) {
   162→        const data = (await res.json()) as { reference?: string }
   163→        setReference(data.reference ?? null)
   164→        setDone(true)
   165→        playSuccess() // Phase 2 sensory feedback — fires on REAL success only
   166→        return
   167→      }
   168→
   169→      // Server rejected: surface translated server-side messages.
   170→      const data = (await res.json().catch(() => null)) as
   171→        | { message?: string; fields?: Record<string, string> }
   172→        | null
   173→      if (res.status === 400 && data?.fields) {
   174→        const fe: { name?: string; email?: string; whatsapp?: string } = {}
   175→        if (data.fields.name) fe.name = data.fields.name
   176→        if (data.fields.email) fe.email = data.fields.email
   177→        if (data.fields.whatsapp) fe.whatsapp = data.fields.whatsapp
   178→        setErrors(fe)
   179→      }
   180→      toast.error(t('form.errorTitle'), {
   181→        description: data?.message ?? t('form.errorNetwork'),
   182→      })
   183→    } catch {
   184→      // Network failure — data stays in the form for a retry.
   185→      toast.error(t('form.errorTitle'), { description: t('form.errorNetwork') })
   186→    } finally {
   187→      setSubmitting(false)
   188→    }
   189→  }
   190→
   191→  const slideVariants = reduced
   192→    ? undefined
   193→    : {
   194→        enter: (d: number) => ({ opacity: 0, x: d * 30 }),
   195→        center: { opacity: 1, x: 0 },
   196→        exit: (d: number) => ({ opacity: 0, x: -d * 30 }),
   197→      }
   198→
   199→  // L1-C P3 (fix 2-d): aria-label REMOVED — it overrode aria-labelledby
   200→  // and named the landmark by the short kicker instead of the full h2.
   201→  return (
   202→    <section id="calculator" className="bg-background py-20 sm:py-28" aria-labelledby="calc-title">
   203→      <div className="elyra-container max-w-5xl">
   204→        <SectionHeading
   205→          kicker={t('kicker')}
   206→          title={t('title')}
   207→          subtitle={t('subtitle')}
   208→          titleId="calc-title"
   209→        />
   210→
   211→        {/* Progress */}
   212→        <div className="mt-10 flex items-center justify-between gap-4">
   213→          <p className="text-sm text-muted-foreground">
   214→            {t('step', { current: step + 1, total: 3 })}
   215→          </p>
   216→          <div className="flex flex-1 gap-2">
   217→            {[0, 1, 2].map((s) => (
   218→              <div
   219→                key={s}
   220→                className="h-1.5 flex-1 overflow-hidden rounded-full bg-muted"
   221→                aria-hidden="true"
   222→              >
   223→                <motion.div
   224→                  className="h-full rounded-full bg-primary"
   225→                  initial={false}
   226→                  animate={{ width: step >= s ? '100%' : '0%' }}
   227→                  transition={{ duration: 0.4 }}
   228→                />
   229→              </div>
   230→            ))}
   231→          </div>
   232→        </div>
   233→
   234→        <div className="relative mt-8 overflow-hidden rounded-3xl border border-border bg-card p-6 sm:p-10">
   235→          {/* UI-5 (visual-only): gradient hairline accent along the wizard
   236→              card's top edge — decorative, no logic/state/pricing impact. */}
   237→          <span
   238→            aria-hidden="true"
   239→            className="pointer-events-none absolute inset-x-0 top-0 h-px bg-gradient-to-r from-transparent via-primary/40 to-transparent"
   240→          />
   241→          <AnimatePresence mode="wait" custom={dir}>
   242→            <motion.div
   243→              key={step}
   244→              custom={dir}
   245→              variants={slideVariants}
   246→              initial="enter"
   247→              animate="center"
   248→              exit="exit"
   249→              transition={{ duration: 0.4, ease: [0.16, 1, 0.3, 1] }}
   250→            >
   251→              {/* STEP 0 — Service type */}
   252→              {step === 0 ? (
   253→                <div>
   254→                  <h3 className="text-lg font-semibold">{t('steps.service')}</h3>
   255→                  <div className="mt-6 grid gap-4 sm:grid-cols-3">
   256→                    {SERVICES.map(({ id, icon: Icon }) => {
   257→                      const active = input.service === id
   258→                      const isFull = id === 'full'
   259→                      return (
   260→                        <button
   261→                          key={id}
   262→                          type="button"
   263→                          data-cursor="magnet"
   264→                          onClick={() => setInput((p) => ({ ...p, service: id }))}
   265→                          aria-pressed={active}
   266→                          className={cn(
   267→                            'group relative overflow-hidden rounded-2xl border p-5 text-start transition-all',
   268→                            active
   269→                              ? 'border-primary bg-primary/5 shadow-[0_0_0_1px_var(--color-primary)]'
   270→                              : 'border-border hover:border-primary/40 hover:bg-foreground/[0.02]'
   271→                          )}
   272→                        >
   273→                          {isFull ? (
   274→                            <span className="absolute end-3 top-3 rounded-full bg-primary px-2 py-0.5 text-[10px] font-semibold text-primary-foreground">
   275→                              {t('services.full.popular')}
   276→                            </span>
   277→                          ) : null}
   278→                          <div className={cn(
   279→                            'flex size-11 items-center justify-center rounded-xl transition-colors',
   280→                            active ? 'bg-primary text-primary-foreground' : 'bg-primary/10 text-primary'
   281→                          )}>
   282→                            <Icon className="size-5" aria-hidden="true" />
   283→                          </div>
   284→                          <p className="mt-3 font-semibold">{t(`services.${id}.title`)}</p>
   285→                          <p className="mt-1 text-sm text-muted-foreground">{t(`services.${id}.desc`)}</p>
   286→                          {active ? (
   287→                            <span className="absolute -end-2 -top-2 flex size-6 items-center justify-center rounded-full bg-primary text-primary-foreground">
   288→                              <Check className="size-3.5" aria-hidden="true" />
   289→                            </span>
   290→                          ) : null}
   291→                        </button>
   292→                      )
   293→                    })}
   294→                  </div>
   295→                </div>
   296→              ) : null}
   297→
   298→              {/* STEP 1 — Features */}
   299→              {step === 1 ? (
   300→                <div className="space-y-8">
   301→                  <div>
   302→                    <div className="flex items-center justify-between">
   303→                      <h3 className="text-lg font-semibold">{t('pages')}</h3>
   304→                      <span className="text-sm font-medium text-primary tabular-nums">
   305→                        {t('pagesValue', { count: input.pages })}
   306→                      </span>
   307→                    </div>
   308→                    <div className="mt-4 px-1">
   309→                      <Slider
   310→                        value={[input.pages]}
   311→                        onValueChange={(v) => setInput((p) => ({ ...p, pages: v[0] ?? p.pages }))}
   312→                        min={1}
   313→                        max={20}
   314→                        step={1}
   315→                        aria-label={t('pages')}
   316→                      />
   317→                      <div className="mt-2 flex justify-between text-xs text-muted-foreground">
   318→                        <span>1</span><span>20</span>
   319→                      </div>
   320→                    </div>
   321→                  </div>
   322→
   323→                  <div>
   324→                    <h3 className="text-lg font-semibold">{t('languages')}</h3>
   325→                    <div className="mt-3 grid gap-3 sm:grid-cols-2">
   326→                      {(['single', 'bilingual'] as LanguageOption[]).map((opt) => {
   327→                        const active = input.languages === opt
   328→                        return (
   329→                          <button
   330→                            key={opt}
   331→                            type="button"
   332→                            onClick={() => setInput((p) => ({ ...p, languages: opt }))}
   333→                            aria-pressed={active}
   334→                            className={cn(
   335→                              'rounded-xl border px-4 py-3 text-start transition-colors',
   336→                              active ? 'border-primary bg-primary/5' : 'border-border hover:bg-foreground/[0.02]'
   337→                            )}
   338→                          >
   339→                            <span className="flex items-center gap-2">
   340→                              <span className={cn('flex size-5 items-center justify-center rounded-full border', active ? 'border-primary bg-primary text-primary-foreground' : 'border-border')}>
   341→                                {active ? <Check className="size-3" aria-hidden="true" /> : null}
   342→                              </span>
   343→                              <span className="text-sm font-medium">{t(`languagesOptions.${opt}`)}</span>
   344→                            </span>
   345→                          </button>
   346→                        )
   347→                      })}
   348→                    </div>
   349→                  </div>
   350→
   351→                  <div>
   352→                    <h3 className="text-lg font-semibold">{t('threeD')}</h3>
   353→                    <div className="mt-3 grid gap-3 sm:grid-cols-2">
   354→                      {(['yes', 'no'] as ThreeDOption[]).map((opt) => {
   355→                        const active = input.threeD === opt
   356→                        const disabled = input.service === 'automation'
   357→                        return (
   358→                          <button
   359→                            key={opt}
   360→                            type="button"
   361→                            disabled={disabled}
   362→                            onClick={() => setInput((p) => ({ ...p, threeD: opt }))}
   363→                            aria-pressed={active}
   364→                            className={cn(
   365→                              'rounded-xl border px-4 py-3 text-start transition-colors',
   366→                              disabled && 'opacity-50',
   367→                              active ? 'border-primary bg-primary/5' : 'border-border hover:bg-foreground/[0.02]'
   368→                            )}
   369→                          >
   370→                            <span className="flex items-center gap-2">
   371→                              <span className={cn('flex size-5 items-center justify-center rounded-full border', active ? 'border-primary bg-primary text-primary-foreground' : 'border-border')}>
   372→                                {active ? <Check className="size-3" aria-hidden="true" /> : null}
   373→                              </span>
   374→                              <span className="text-sm font-medium">{t(`threeDOptions.${opt}`)}</span>
   375→                            </span>
   376→                          </button>
   377→                        )
   378→                      })}
   379→                    </div>
   380→                  </div>
   381→
   382→                  <div>
   383→                    <h3 className="text-lg font-semibold">{t('integrations')}</h3>
   384→                    <div className="mt-3 flex flex-wrap gap-2">
   385→                      {INTEGRATIONS.map((key) => {
   386→                        const active = input.integrations.includes(key)
   387→                        return (
   388→                          <button
   389→                            key={key}
   390→                            type="button"
   391→                            onClick={() => toggleIntegration(key)}
   392→                            aria-pressed={active}
   393→                            className={cn(
   394→                              'inline-flex min-h-11 items-center gap-1.5 rounded-full border px-3 py-1.5 text-sm transition-colors',
   395→                              active ? 'border-primary bg-primary/10 text-primary-strong' : 'border-border hover:bg-foreground/[0.02]'
   396→                            )}
   397→                          >
   398→                            <span className={cn('flex size-4 items-center justify-center rounded-full border', active ? 'border-primary bg-primary text-primary-foreground' : 'border-border')}>
   399→                              {active ? <Check className="size-2.5" aria-hidden="true" /> : null}
   400→                            </span>
   401→                            {t(`integrationsOptions.${key}`)}
   402→                          </button>
   403→                        )
   404→                      })}
   405→                    </div>
   406→                  </div>
   407→
   408→                  <div>
   409→                    <h3 className="text-lg font-semibold">{t('automationLevel')}</h3>
   410→                    <div className="mt-3 grid gap-3 sm:grid-cols-2">
   411→                      {(['essential', 'advanced'] as AutomationLevel[]).map((opt) => {
   412→                        const active = input.automationLevel === opt
   413→                        const disabled = input.service === 'website'
   414→                        return (
   415→                          <button
   416→                            key={opt}
   417→                            type="button"
   418→                            disabled={disabled}
   419→                            onClick={() => setInput((p) => ({ ...p, automationLevel: opt }))}
   420→                            aria-pressed={active}
   421→                            className={cn(
   422→                              'rounded-xl border px-4 py-3 text-start transition-colors',
   423→                              disabled && 'opacity-50',
   424→                              active ? 'border-primary bg-primary/5' : 'border-border hover:bg-foreground/[0.02]'
   425→                            )}
   426→                          >
   427→                            <span className="flex items-center gap-2">
   428→                              <span className={cn('flex size-5 items-center justify-center rounded-full border', active ? 'border-primary bg-primary text-primary-foreground' : 'border-border')}>
   429→                                {active ? <Check className="size-3" aria-hidden="true" /> : null}
   430→                              </span>
   431→                              <span className="text-sm font-medium">{t(`automationLevelOptions.${opt}`)}</span>
   432→                            </span>
   433→                          </button>
   434→                        )
   435→                      })}
   436→                    </div>
   437→                  </div>
   438→                </div>
   439→              ) : null}
   440→
   441→              {/* STEP 2 — Result */}
   442→              {step === 2 ? (
   443→                <div>
   444→                  {done ? (
   445→                    <div className="flex flex-col items-center py-8 text-center">
   446→                      <div className="flex size-16 items-center justify-center rounded-full bg-g-green/15 text-g-green">
   447→                        <Check className="size-8" aria-hidden="true" />
   448→                      </div>
   449→                      <h3
   450→                        ref={successHeadingRef}
   451→                        tabIndex={-1}
   452→                        className="mt-5 text-2xl font-semibold"
   453→                      >
   454→                        {t('form.successTitle')}
   455→                      </h3>
   456→                      <p className="mt-2 max-w-md text-muted-foreground">{t('form.successDesc')}</p>
   457→                      {reference ? (
   458→                        <p className="mt-3 rounded-full border border-g-green/30 bg-g-green/5 px-4 py-1.5 font-mono text-sm font-semibold tracking-wide text-g-green">
   459→                          {t('form.successReference', { reference })}
   460→                        </p>
   461→                      ) : null}
   462→                      <button
   463→                        type="button"
   464→                        onClick={() => { setDone(false); setReference(null); setStep(0); setInput(INITIAL_INPUT); setForm({ name: '', email: '', whatsapp: '' }) }}
   465→                        className="mt-6 inline-flex h-11 items-center gap-2 rounded-full border border-border px-5 text-sm font-medium hover:bg-foreground/5"
   466→                      >
   467→                        <RotateCw className="size-4" aria-hidden="true" />
   468→                        {t('form.successRestart')}
   469→                      </button>
   470→                    </div>
   471→                  ) : (
   472→                    <div className="grid gap-8 lg:grid-cols-2">
   473→                      <div>
   474→                        <div className="rounded-2xl border border-border bg-gradient-to-br from-primary/5 to-transparent p-6">
   475→                          {/* WS-4: animated SVG ring gauges */}
   476→                          <div className="mb-6 flex items-center justify-center gap-12">
   477→                            <RingGauge
   478→                              fraction={Math.min(1, result.max / 20000)}
   479→                              value={result.max}
   480→                              formatValue={(n) => formatMoney(Math.round(n), locale)}
   481→                              label={t('result.budget')}
   482→                              color="#0071E3"
   483→                              isRtl={locale === 'ar'}
   484→                            />
   485→                            <RingGauge
   486→                              fraction={Math.min(1, result.weeksMax / 12)}
   487→                              value={result.weeksMax}
   488→                              formatValue={(n) => `${Math.round(n)}`}
   489→                              label={t('result.duration')}
   490→                              color="#34A853"
   491→                              isRtl={locale === 'ar'}
   492→                            />
   493→                          </div>
   494→                          <p className="text-xs font-medium uppercase tracking-wider text-muted-foreground">
   495→                            {t('result.budget')}
   496→                          </p>
   497→                          <p className="mt-2 text-3xl font-bold tracking-tight text-primary sm:text-4xl">
   498→                            {formatMoney(result.min, locale)} – {formatMoney(result.max, locale)}
   499→                          </p>
   500→                          <p className="mt-4 text-xs font-medium uppercase tracking-wider text-muted-foreground">
   501→                            {t('result.duration')}
   502→                          </p>
   503→                          <p className="mt-1 text-xl font-semibold">
