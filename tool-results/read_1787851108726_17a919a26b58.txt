     8→/**
     9→ * R3F console scene (Phase 4 WS-1, prompt §4).
    10→ *
    11→ * Mounts ONLY after the user selects a preset or presses Enter (dynamic
    12→ * import with ssr:false in hero-console.tsx). Shows glowing 3D nodes +
    13→ * light connections forming the selected system's architecture.
    14→ *
    15→ * Reuses the existing three/R3F in the project — no new libraries.
    16→ * Drag to rotate (FIX(2-b), L1-C/L1-D P2 — now actually implemented):
    17→ * throttled ~60Hz pointer drag on the scene wrapper with pointer capture,
    18→ * mirroring capability-scene's proven pattern; yaw is free (the 360°
    19→ * affordance) while pitch is clamped to ±MAX_TILT so the console can never
    20→ * flip upside down, and the auto-drift pauses while dragging. No wheel
    21→ * zoom / no scroll hijack (§4). DPR ≤ 2, frameloop paused when offscreen
    22→ * (§9.4), plus a ContextLossGuard like the other two canvases.
    23→ *
    24→ * Phase 5 fix (P0-1): the original implementation used `<line>` JSX which
    25→ * collides with SVG's `<line>` element in R3F v9 + React 19, silently
    26→ * failing to render geometry. Connections are now THREE.Mesh tubes built
    27→ * explicitly via `<primitive>` (same bypass), so the renderer reliably
    28→ * draws them, and scene readiness is verified through the `sceneReady`
    29→ * gate pattern below (InitialRenderSafety).
    30→ *
    31→ * Task 4 visual upgrade: nodes are glowing spheres (0.32 radius, 32×32
    32→ * segments) with additive glow sprites behind them (one shared 64×64
    33→ * radial-gradient CanvasTexture), connections are curved QuadraticBezier
    34→ * arcs (midpoint lifted ~0.6 units perpendicular) rendered as additive
    35→ * TubeGeometry meshes, each main connection carries a bright traveling
    36→ * pulse (getPointAt((t·speed + phase) % 1)), and node emissive gently
    37→ * pulses out of phase. Perf budget: ≤ 12 tubes, pulses ≤ node count, no
    38→ * post-processing, no drei.
    39→ */

    94→/** Generate per-preset node positions */
    95→function buildPositions(config: PresetConfig): [number, number, number][] {
    96→  const positions: [number, number, number][] = []
    97→  for (let i = 0; i < config.count; i++) {
    98→    const angle = (i / config.count) * Math.PI * 2
    99→    const r = config.radius
   100→    if (config.layout === 'orbit') {
   101→      // Orbit: central "sun" at index 0, planets around
   102→      if (i === 0) {
   103→        positions.push([0, 0, 0])
   104→      } else {
   105→        const a = ((i - 1) / (config.count - 1)) * Math.PI * 2
   106→        positions.push([Math.cos(a) * r, Math.sin(a) * r * 0.7, Math.sin(a * 2) * 0.5])
   107→      }
   108→    } else if (config.layout === 'flow') {
   109→      // Flow: left-to-right pipeline with vertical jitter
   110→      const x = (i / Math.max(1, config.count - 1)) * (r * 2) - r
   111→      positions.push([x, Math.sin(angle * 1.5) * 0.8, Math.cos(angle * 1.5) * 0.5])
   112→    } else {
   113→      // Ring (default): a circular arrangement
   114→      positions.push([
   115→        Math.cos(angle) * r,
   116→        Math.sin(angle) * r * 0.6,
   117→        Math.sin(angle * 2) * 0.5,
   118→      ])
   119→    }
   120→  }
   121→  return positions
   122→}

   124→/** Shared radial-gradient glow texture — 64×64 white radial fade, generated
   125→ *  once per mount (component scope, this is a client-only dynamic import)
   126→ *  and reused by every node's SpriteMaterial. Material.dispose() never
   127→ *  frees textures, so it is disposed exactly once — at unmount — by its
   128→ *  own disposal effect below (LOOP-3 FIX 2: previously it rode the tubes
   129→ *  effect and was disposed mid-mount on every preset switch). */
   130→function makeGlowTexture(): THREE.CanvasTexture {
   131→  const size = 64
   132→  const canvas = document.createElement('canvas')
   133→  canvas.width = size
   134→  canvas.height = size
   135→  const ctx = canvas.getContext('2d')
   136→  if (ctx) {
   137→    const half = size / 2
   138→    const grad = ctx.createRadialGradient(half, half, 0, half, half, half)
   139→    grad.addColorStop(0, 'rgba(255, 255, 255, 1)')
   140→    grad.addColorStop(0.35, 'rgba(255, 255, 255, 0.4)')
   141→    grad.addColorStop(1, 'rgba(255, 255, 255, 0)')
   142→    ctx.fillStyle = grad
   143→    ctx.fillRect(0, 0, size, size)
   144→  }
   145→  const tex = new THREE.CanvasTexture(canvas)
   146→  tex.needsUpdate = true
   147→  return tex
   148→}

   150→const UP = new THREE.Vector3(0, 1, 0)
   151→const WHITE = new THREE.Color('#FFFFFF')
   152→
   153→/** Polar tilt clamp for drag + auto-drift — the console banks up to ~±29°
   154→ *  but can never flip upside down (the "clamped" in the drag affordance). */
   155→const MAX_TILT = 0.5
   156→
   157→/** Build the curved connection path between two nodes — a quadratic bezier
   158→ *  whose midpoint is lifted ~0.6 units perpendicular to the chord, so links
   159→ *  read as elegant arcs instead of straight wires. */
   160→function buildArc(a: THREE.Vector3, b: THREE.Vector3, lift: number): THREE.QuadraticBezierCurve3 {
   161→  const dir = new THREE.Vector3().subVectors(b, a)
   162→  const perp = new THREE.Vector3().crossVectors(dir, UP)
   163→  // chord parallel to UP → pick a stable fallback perpendicular
   164→  if (perp.lengthSq() < 1e-6) perp.set(1, 0, 0)
   165→  else perp.normalize()
   166→  const mid = new THREE.Vector3().addVectors(a, b).multiplyScalar(0.5).addScaledVector(perp, lift)
   167→  return new THREE.QuadraticBezierCurve3(a, mid, b)
   168→}
   169→
   170→/** Render an arc as a thin additive tube — Mesh form factor avoids the
   171→ *  `<line>` JSX/SVG intrinsic collision from P0-1 entirely. */
   172→function buildTube(curve: THREE.QuadraticBezierCurve3, color: number, opacity: number): THREE.Mesh {
   173→  const geometry = new THREE.TubeGeometry(curve, 24, 0.012, 8, false)
   174→  const material = new THREE.MeshBasicMaterial({
   175→    color,
   176→    transparent: true,
   177→    opacity,
   178→    blending: THREE.AdditiveBlending,
   179→    depthWrite: false,
   180→  })
   181→  return new THREE.Mesh(geometry, material)
   182→}

   184→/** Traveling pulse metadata — the mesh/material are JSX-declared (R3F
   185→ *  disposes them); only the curve + timing live here. */
   186→interface PulseSpec {
   187→  curve: THREE.QuadraticBezierCurve3
   188→  /** Loops per second along the path */
   189→  speed: number
   190→  /** Start offset along the path (also used for the static initial pose) */
   191→  phase: number
   192→  /** Bright whitened link color for the pulse core */
   193→  color: string
   194→  /** Initial position so the scene looks intentional even when the
   195→   *  frameloop is paused (reduced motion / offscreen). */
   196→  initial: [number, number, number]
   197→}

   199→function Nodes({
   200→  preset,
   201→  active,
   202→  dragging,
   203→  registerGroup,
   204→}: {
   205→  preset: PresetId
   206→  active: boolean
   207→  /** While true the auto-drift pauses — the visitor's drag owns rotation
   208→   *  (capability-scene's approach, kept for consistency). */
   209→  dragging: boolean
   210→  /** Hands the rotation group up to ConsoleScene so its DOM-level drag
   211→   *  handler can rotate the same group this component's useFrame drifts
   212→   *  (callback prop, not a ref prop — keeps every mutation on a
   213→   *  component-local ref so the react-compiler stays quiet in both
   214→   *  components). */
   215→  registerGroup: (g: THREE.Group | null) => void
   216→}) {
   217→  const groupRef = useRef<THREE.Group | null>(null)
   218→  const reduced = usePrefersReducedMotion()
   219→  const config = PRESET_CONFIG[preset] ?? PRESET_CONFIG.custom
   220→
   221→  // Per-node material refs (emissive pulsing) + per-pulse mesh/material refs
   222→  // — all mutations go through ref.current, the React-19-safe path.
   223→  const nodeMatRefs = useRef<(THREE.MeshStandardMaterial | null)[]>([])
   224→  const pulseMeshRefs = useRef<(THREE.Mesh | null)[]>([])
   225→  const pulseMatRefs = useRef<(THREE.MeshBasicMaterial | null)[]>([])
   226→
   227→  // One glow texture per mount, shared by every sprite.
   228→  const glowTexture = useMemo(() => makeGlowTexture(), [])
   229→
   230→  // Positions, colors, tube meshes and pulse specs — computed once per
   231→  // preset change (pure: no randomness).
   232→  const { positions, colors, tubes, pulses } = useMemo(() => {
   233→    const positions = buildPositions(config)
   234→    const colors = config.colors.map((hex) => new THREE.Color(hex))
   235→    const tubes: THREE.Mesh[] = []
   236→    const pulses: PulseSpec[] = []
   237→
   238→    const addLink = (
   239→      a: [number, number, number] | undefined,
   240→      b: [number, number, number] | undefined,
   241→      color: number,
   242→      opacity: number,
   243→      withPulse: boolean,
   244→      lift: number
   245→    ) => {
   246→      if (!a || !b) return
   247→      const curve = buildArc(new THREE.Vector3(...a), new THREE.Vector3(...b), lift)
   248→      tubes.push(buildTube(curve, color, opacity))
   249→      if (!withPulse) return
   250→      const phase = pulses.length * 0.37
   251→      const start = curve.getPointAt(phase % 1)
   252→      pulses.push({
   253→        curve,
   254→        speed: 0.22 + (pulses.length % 3) * 0.07,
   255→        phase,
   256→        // whitened link color — reads as a bright energy packet
   257→        color: '#' + new THREE.Color(color).lerp(WHITE, 0.55).getHexString(),
   258→        initial: [start.x, start.y, start.z],
   259→      })
   260→    }
   261→
   262→    if (config.layout === 'orbit') {
   263→      // Spokes from the central "sun" to each planet (all carry pulses)
   264→      const center = positions[0] ?? [0, 0, 0]
   265→      for (let i = 1; i < positions.length; i++) {
   266→        addLink(center, positions[i], 0x0071e3, 0.5, true, 0.6)
   267→      }
   268→    } else if (config.layout === 'flow') {
   269→      // Pipeline segments bow consistently toward the camera
   270→      for (let i = 0; i < positions.length - 1; i++) {
   271→        addLink(positions[i], positions[i + 1], 0x4285f4, 0.5, true, 0.6)
   272→      }
   273→    } else {
   274→      // Ring — neighbor links carry pulses and alternate bow direction;
   275→      // faint cross-chords (no pulses) bow the other way for depth layering
   276→      for (let i = 0; i < positions.length; i++) {
   277→        addLink(
   278→          positions[i],
   279→          positions[(i + 1) % positions.length],
   280→          0x4285f4,
   281→          0.5,
   282→          true,
   283→          i % 2 === 0 ? 0.6 : -0.6
   284→        )
   285→      }
   286→      for (let i = 0; i < positions.length; i++) {
   287→        addLink(
   288→          positions[i],
   289→          positions[(i + 2) % positions.length],
   290→          0x0071e3,
   291→          0.22,
   292→          false,
   293→          i % 2 === 0 ? -0.6 : 0.6
   294→        )
   295→      }
   296→    }
   297→    return { positions, colors, tubes, pulses }
   298→  }, [config])
   299→
   300→  // FIX(2-b): publish the rotation group to the parent's DOM drag handler.
   301→  // Refs attach before effects run, and the group persists across preset
   302→  // switches (the JSX group is unkeyed) — register once per mount cycle.
   303→  useEffect(() => {
   304→    registerGroup(groupRef.current)
   305→    return () => registerGroup(null)
   306→  }, [registerGroup])
   307→
   308→  // LOOP-3 FIX 7: R3F resets clock.elapsedTime to 0 on every frameloop
   309→  // 'never'↔'always' transition (offscreen/visibility gating flips it
   310→  // constantly here), which would rewind the emissive/pulse phases every
   311→  // time the hero pauses this canvas. Accumulate scene-local time from
   312→  // delta instead — the same immune pattern hero-canvas uses for uTime.
   313→  const tRef = useRef(0)
   314→
   315→  // Slow rotation of the whole group — pauses while the visitor drags
   316→  // (capability-scene's approach), with the polar tilt clamped so neither
   317→  // drag nor drift alone can flip the console upside down.
   318→  // Node emissive pulses out of phase; connection pulses travel their arcs.
   319→  useFrame((state, delta) => {
   320→    if (!groupRef.current) return
   321→    if (active && !reduced && !dragging) {
   322→      groupRef.current.rotation.y += delta * 0.08
   323→      groupRef.current.rotation.x = THREE.MathUtils.clamp(
   324→        groupRef.current.rotation.x + delta * 0.02,
   325→        -MAX_TILT,
   326→        MAX_TILT
   327→      )
   328→    }
   329→    if (reduced) return
   330→    tRef.current += delta
   331→    const t = tRef.current
   332→    nodeMatRefs.current.forEach((mat, i) => {
   333→      if (!mat) return
   334→      // gentle breathing glow, per-node phase offset so the constellation
   335→      // shimmers instead of flashing in lockstep
   336→      mat.emissiveIntensity = 0.9 + 0.5 * Math.sin(t * 1.7 + i * 1.9)
   337→    })
   338→    pulses.forEach((pulse, i) => {
   339→      const mesh = pulseMeshRefs.current[i]
   340→      const mat = pulseMatRefs.current[i]
   341→      if (!mesh || !mat) return
   342→      const u = (t * pulse.speed + pulse.phase) % 1
   343→      pulse.curve.getPointAt(u, mesh.position)
   344→      // bright mid-path, fading near the endpoints
   345→      mat.opacity = 0.2 + 0.8 * Math.sin(Math.PI * u)
   346→    })
   347→  })
   348→
   349→  // FIX(2-c/9, board-R3): R3F never disposes primitives (its removeChild →
   350→  // disposeOnIdle path explicitly skips them), so the tube geometry +
   351→  // material GPU buffers need explicit disposal. Dispose the `tubes`
   352→  // closure value in the cleanup: on a preset switch React has already
   353→  // unmounted the OLD set by the time the cleanup runs, and on final
   354→  // unmount the finally-mounted set IS disposed. Under StrictMode the
   355→  // dev double-invoke briefly disposes still-mounted geometries which
   356→  // three.js transparently re-uploads next frame — the same accepted
   357→  // semantics as capability-scene's Centerpiece disposal. Node spheres /
   358→  // sprites / pulse meshes are JSX-declared so R3F handles those.
   359→  // LOOP-3 FIX 2: scoped to [tubes] ONLY — this effect previously disposed
   360→  // glowTexture too, so a preset switch (new tubes identity) fired
   361→  // glowTexture.dispose() mid-mount while the texture was still bound to
   362→  // every sprite material. The glow texture now has its own effect below.
   363→  useEffect(() => {
   364→    return () => {
   365→      for (const tube of tubes) {
   366→        tube.geometry.dispose()
   367→        const material = tube.material
   368→        if (Array.isArray(material)) material.forEach((m) => m.dispose())
   369→        else material.dispose()
   370→      }
   371→    }
   372→  }, [tubes])
   373→
   374→  // Glow texture disposal — its own effect (LOOP-3 FIX 2). The memo above
   375→  // has stable [] deps, so glowTexture's identity never changes across
   376→  // preset switches: this cleanup fires exactly once, at unmount, never
   377→  // mid-mount. ([glowTexture] in the deps satisfies exhaustive-deps; the
   378→  // stable identity is what keeps it unmount-only.)
   379→  useEffect(() => {
   380→    return () => glowTexture.dispose()
   381→  }, [glowTexture])
   382→
   383→  return (
   384→    <group ref={groupRef}>
   385→      {positions.map((pos, i) => {
   386→        const color = colors[i % colors.length] ?? new THREE.Color('#4285F4')
   387→        return (
   388→          <group key={`node-${i}`} position={pos}>
   389→            <mesh>
   390→              <sphereGeometry args={[0.32, 32, 32]} />
   391→              <meshStandardMaterial
   392→                ref={(m) => {
   393→                  nodeMatRefs.current[i] = m
   394→                }}
   395→                color={color}
   396→                emissive={color}
   397→                emissiveIntensity={0.9}
   398→                roughness={0.3}
   399→                metalness={0.6}
   400→              />
   401→            </mesh>
   402→            {/* additive glow halo — depthTest off + late renderOrder so it
   403→                reads like bloom over the constellation instead of being
   404→                clipped by its own sphere */}
   405→            <sprite scale={[1.7, 1.7, 1]} renderOrder={2}>
   406→              <spriteMaterial
   407→                map={glowTexture}
   408→                color={color}
   409→                transparent
   410→                opacity={0.5}
   411→                blending={THREE.AdditiveBlending}
   412→                depthWrite={false}
   413→                depthTest={false}
   414→              />
   415→            </sprite>
   416→          </group>
   417→        )
   418→      })}
   419→      {/* Curved connections — mesh tubes via <primitive> (bypasses the
   420→          broken `<line>` JSX intrinsic) */}
   421→      {tubes.map((tube, i) => (
   422→        <primitive key={`tube-${i}`} object={tube} />
   423→      ))}
   424→      {/* Traveling pulses along the connection arcs */}
   425→      {pulses.map((pulse, i) => (
   426→        <mesh
   427→          key={`pulse-${i}`}
   428→          position={pulse.initial}
   429→          ref={(m) => {
   430→            pulseMeshRefs.current[i] = m
   431→          }}
   432→        >
   433→          <sphereGeometry args={[0.05, 12, 12]} />
   434→          <meshBasicMaterial
   435→            ref={(mat) => {
   436→              pulseMatRefs.current[i] = mat
   437→            }}
   438→            color={pulse.color}
   439→            transparent
   440→            blending={THREE.AdditiveBlending}
   441→            depthWrite={false}
   442→          />
   443→        </mesh>
   444→      ))}
   445→    </group>
   446→  )
   447→}

   449→/** Initial render safety: ensure the very first frame is scheduled even
   450→ *  when rAF timing is tight (headless/SwiftShader). */
   451→function InitialRenderSafety() {
   452→  const invalidate = useThree((s) => s.invalidate)
   453→  useEffect(() => {
   454→    // Force at least one render frame after mount + one after a tick (in
   455→    // case R3F's first auto-render raced the GL state setup).
   456→    invalidate()
   457→    const id = requestAnimationFrame(() => invalidate())
   458→    return () => cancelAnimationFrame(id)
   459→  }, [invalidate])
   460→  return null
   461→}

   463→/** Context-loss guard (FIX(2-b): parity with hero-canvas.tsx /
   464→ *  capability-scene.tsx, L1-C P2) — `preventDefault()` marks the event as
   465→ *  handled so the browser keeps the canvas alive for a possible restore
   466→ *  (and stops the default console error spam); we log once for
   467→ *  diagnostics. */
   468→function ContextLossGuard() {
   469→  const gl = useThree((s) => s.gl)
   470→  useEffect(() => {
   471→    const canvas = gl.domElement
   472→    const onLost = (e: Event) => {
   473→      e.preventDefault()
   474→      console.warn('[ConsoleScene] WebGL context lost')
   475→    }
   476→    canvas.addEventListener('webglcontextlost', onLost)
   477→    return () => canvas.removeEventListener('webglcontextlost', onLost)
   478→  }, [gl])
   479→  return null
   480→}

   482→export function ConsoleScene({
   483→  preset,
   484→  /** While false the Canvas frameloop pauses entirely (§9.4) — the hero
   485→   *  threads its IntersectionObserver + visibility state here so the
   486→   *  scene never burns GPU frames while offscreen or the tab is hidden. */
   487→  active = true,
   488→  /** LOOP-1 E2E fix: the hero's pending "payoff beat" navigation reads a
   489→   *  drag flag — while the visitor is dragging, navigation is POSTPONED
   490→   *  (never cancelled) so "اسحب للتدوير 360°" is a truthful, usable
   491→   *  affordance. Callback prop (NOT a ref prop) per the react-compiler
   492→   *  discipline used across this codebase — the parent owns the flag. */
   493→  onDragStateChange,
   494→  /** Accessible name for the drag surface (LOOP-3 FIX 4) — the wrapper is
   495→   *  role="img" + tabIndex, so this label is what screen readers announce
   496→   * (hero-console passes hero.console.sceneLabel, present in both
   497→   *  catalogs: “نموذج ثلاثي الأبعاد تفاعلي — اسحب بالماوس أو استخدم مفاتيح
   498→   *  الأسهم للتدوير”). Optional — omitting it simply leaves the img role
   499→   *  unnamed (legacy behavior). */
   500→  sceneLabel,
   501→}: {
   502→  preset: PresetId
   503→  active?: boolean
   504→  onDragStateChange?: (dragging: boolean) => void
   505→  sceneLabel?: string
   506→}) {
   507→  const reduced = usePrefersReducedMotion()
   508→  const config = PRESET_CONFIG[preset] ?? PRESET_CONFIG.custom
   509→
   510→  // FIX(2-b, L1-C/L1-D P2): the promised clamped drag — capability-scene's
   511→  // throttled pointer pattern applied at the DOM wrapper so the WHOLE scene
   512→  // box is a drag surface (R3F raycast handlers would only start a drag on
   513→  // a node/sprite hit, leaving presses on empty space dead — the ray targets
   514→  // are sparse in this scene).
   515→  const [dragging, setDragging] = useState(false)
   516→  // The rotation group itself, registered up by Nodes (callback prop, not
   517→  // a ref prop — keeps every mutation on a component-local ref so the
   518→  // react-compiler stays quiet in both components).
   519→  const groupRef = useRef<THREE.Group | null>(null)
   520→  const registerGroup = useCallback((g: THREE.Group | null) => {
   521→    groupRef.current = g
   522→  }, [])
   523→  const last = useRef({ x: 0, y: 0 })
   524→  const lastPointerTs = useRef(0)
   525→
   526→  // FIX(L2-V P3): if this subtree unmounts MID-DRAG (viewport crossing the
   527→  // 768px tier boundary or an OS reduced-motion toggle while dragging —
   528→  // webglSupported can only flip false→true, never back), pointerup never
   529→  // fires on a removed element, so endDrag's report is lost. Report the
   530→  // release on unmount so the hero's dragActiveRef can't strand true and
   531→  // postpone a pending payoff navigation (or block later intents) forever.
   532→  // Idempotent by design: the parent's reportDragState only writes a ref.
   533→  // It's a stable useCallback, so this cleanup fires exactly once at unmount.
   534→  useEffect(
   535→    () => () => onDragStateChange?.(false),
   536→    [onDragStateChange]
   537→  )
   538→
   539→  const onPointerDown = (e: React.PointerEvent<HTMLDivElement>) => {
   540→    setDragging(true)
   541→    onDragStateChange?.(true)
   542→    last.current.x = e.clientX
   543→    last.current.y = e.clientY
   544→    // capture so the drag keeps tracking after the pointer leaves the box
   545→    e.currentTarget.setPointerCapture?.(e.pointerId)
   546→  }
   547→
   548→  const onPointerMove = (e: React.PointerEvent<HTMLDivElement>) => {
   549→    // FIX(2-c/4 pattern): ~60Hz throttle; early return BEFORE updating
   550→    // `last` keeps drag deltas accumulation-correct.
   551→    const now = performance.now()
   552→    if (now - lastPointerTs.current < 16) return
   553→    lastPointerTs.current = now
   554→    const group = groupRef.current
   555→    if (dragging && group) {
   556→      const dx = e.clientX - last.current.x
   557→      const dy = e.clientY - last.current.y
   558→      // yaw free (the 360° affordance); pitch clamped — mutating the
   559→      // Three.js Object3D directly, same as capability-scene's drag.
   560→      group.rotation.y += dx * 0.008
   561→      group.rotation.x = THREE.MathUtils.clamp(
   562→        group.rotation.x + dy * 0.008,
   563→        -MAX_TILT,
   564→        MAX_TILT
   565→      )
   566→    }
   567→    last.current.x = e.clientX
   568→    last.current.y = e.clientY
   569→  }
   570→
   571→  const endDrag = (e: React.PointerEvent<HTMLDivElement>) => {
   572→    setDragging(false)
   573→    onDragStateChange?.(false)
   574→    // capture auto-releases on up/cancel per spec — explicit release for
   575→    // clarity, guarded so an unpinned pointer can never throw.
   576→    const el = e.currentTarget
   577→    if (el.hasPointerCapture?.(e.pointerId)) {
   578→      el.releasePointerCapture?.(e.pointerId)
   579→    }
   580→  }
   581→
   582→  // LOOP-3 FIX 4: keyboard path for the drag — mirrors three-d-section's
   583→  // pattern (role="img" + tabIndex + arrow keys + focus-visible ring).
   584→  // ArrowLeft/Right step the free yaw ±0.13 rad (~7.5°); ArrowUp/Down step
   585→  // the pitch ±0.07 rad clamped to ±MAX_TILT by the same clamp the pointer
   586→  // drag's pitch path uses. preventDefault sits AFTER the switch (default
   587→  // returns early) so only the four handled arrows swallow the page
   588→  // scroll. Null-guarded: before the lazy scene mounts, arrows scroll
   589→  // normally. Stable useCallback — mutations stay on the local groupRef.
   590→  // V-1 L3-2a P3: a keyboard step is an INTERACTION — set the same
   591→  // `dragging` flag the pointer drag sets so the +0.08 rad/s auto-drift
   592→  // pauses (otherwise it unwinds an ArrowLeft step in ~1.6s), then re-arm
   593→  // the drift after a short keyboard idle (each repeat re-arms; unmount
   594→  // clears the timer). Deliberately NOT routed through
   595→  // onDragStateChange — keyboard rotation must never postpone the hero's
   596→  // pending payoff navigation (that flag stays pointer-drag-only).
   597→  const keyIdleTimer = useRef<number | null>(null)
   598→  useEffect(
   599→    () => () => {
   600→      if (keyIdleTimer.current !== null) window.clearTimeout(keyIdleTimer.current)
   601→    },
   602→    []
   603→  )
   604→  const onKeyDown = useCallback((e: React.KeyboardEvent<HTMLDivElement>) => {
   605→    const group = groupRef.current
   606→    if (!group) return
   607→    switch (e.key) {
   608→      case 'ArrowLeft':
   609→        group.rotation.y -= 0.13
   610→        break
   611→      case 'ArrowRight':
   612→        group.rotation.y += 0.13
   613→        break
   614→      case 'ArrowUp':
   615→        group.rotation.x = THREE.MathUtils.clamp(
   616→          group.rotation.x - 0.07,
   617→          -MAX_TILT,
   618→          MAX_TILT
   619→        )
   620→        break
   621→      case 'ArrowDown':
   622→        group.rotation.x = THREE.MathUtils.clamp(
   623→          group.rotation.x + 0.07,
   624→          -MAX_TILT,
   625→          MAX_TILT
   626→        )
   627→        break
   628→      default:
   629→        return
   630→    }
   631→    e.preventDefault()
   632→    setDragging(true)
   633→    if (keyIdleTimer.current !== null) window.clearTimeout(keyIdleTimer.current)
   634→    keyIdleTimer.current = window.setTimeout(() => {
   635→      keyIdleTimer.current = null
   636→      setDragging(false)
   637→    }, 900)
   638→  }, [])
   639→
   640→  return (
   641→    <div
   642→      // LOOP-3 FIX 4: focus ring — three-d-section's pattern (ring-2 on the
   643→      // ring token), but the INSET variant: this wrapper is inset-0 inside
   644→      // hero-console's overflow-hidden scene frame, so an outward ring +
   645→      // offset (three-d-section's exact classes) would be clipped away by
   646→      // the parent's overflow-hidden and never seen. The inset ring draws
   647→      // the same 2px indicator just inside the scene edge — visible.
   648→      className="absolute inset-0 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring focus-visible:ring-inset"
   649→      onPointerDown={onPointerDown}
   650→      onPointerMove={onPointerMove}
   651→      onPointerUp={endDrag}
   652→      onPointerCancel={endDrag}
   653→      onKeyDown={onKeyDown}
   654→      role="img"
   655→      aria-label={sceneLabel}
   656→      tabIndex={0}
   657→      style={{
   658→        cursor: dragging ? 'grabbing' : 'grab',
   659→        // LOOP-3 FIX 1: claim the gesture for the drag, not page scroll —
   660→        // without this, touch devices ≥768px (iPads that mount this scene)
   661→        // hand the pointer to native panning → pointercancel → the
   662→        // "اسحب للتدوير 360°" affordance is dead. Repo precedent:
   663→        // before-after.tsx (touch-none / touchAction:'none').
   664→        touchAction: 'none',
   665→      }}
   666→    >
   667→      <Canvas
   668→        camera={{ position: [0, 0, config.cameraZ], fov: 50 }}
   669→        dpr={[1, 2]}
   670→        frameloop={reduced || !active ? 'never' : 'always'}
   671→        gl={{ antialias: true, alpha: true, powerPreference: 'high-performance' }}
   672→        style={{ position: 'absolute', inset: 0 }}
   673→      >
   674→        {/* Lighting: brighter to ensure glowing nodes pop visually */}
   675→        <ambientLight intensity={0.5} />
   676→        <pointLight position={[5, 5, 5]} intensity={1.2} color="#ffffff" />
   677→        <pointLight position={[-5, -3, 4]} intensity={0.7} color="#4285F4" />
   678→        <pointLight position={[0, 5, -5]} intensity={0.4} color="#34A853" />
   679→        <Nodes
   680→          preset={preset}
   681→          active={!reduced && active}
   682→          dragging={dragging}
   683→          registerGroup={registerGroup}
   684→        />
   685→        <InitialRenderSafety />
   686→        <ContextLossGuard />
   687→      </Canvas>
   688→    </div>
   689→  )
   690→}
