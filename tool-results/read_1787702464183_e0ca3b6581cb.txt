     1→'use client'
     2→
     3→import { useMemo, useState } from 'react'
     4→import { useLocale, useTranslations } from 'next-intl'
     5→import { AnimatePresence, motion, useReducedMotion } from 'framer-motion'
     6→import { z } from 'zod'
     7→import {
     8→  Globe, Workflow, Boxes, Check, ArrowLeft, ArrowRight,
     9→  Send, RotateCw, AlertCircle,
    10→} from 'lucide-react'
    11→import {
    12→  Slider,
    13→} from '@/components/ui/slider'
    14→import { Input } from '@/components/ui/input'
    15→import { Label } from '@/components/ui/label'
    16→import { Button } from '@/components/ui/button'
    17→import { cn } from '@/lib/utils'
    18→import { SectionHeading } from '@/components/shared/section-heading'
    19→import { playSuccess } from '@/lib/sound'
    20→import { toast } from 'sonner'
    21→import { RingGauge } from './ring-gauge'
    22→import {
    23→  computeEstimate, formatMoney,
    24→  type CalculatorInput, type ServiceType, type IntegrationKey,
    25→  type AutomationLevel, type LanguageOption, type ThreeDOption,
    26→} from '@/lib/calculator'
    27→
    28→type Step = 0 | 1 | 2
    29→
    30→const SERVICES: { id: ServiceType; icon: typeof Globe }[] = [
    31→  { id: 'website', icon: Globe },
    32→  { id: 'automation', icon: Workflow },
    33→  { id: 'full', icon: Boxes },
    34→]
    35→
    36→const INTEGRATIONS: IntegrationKey[] = ['crm', 'invoicing', 'email', 'telegram', 'sheets', 'ai']
    37→
    38→const leadSchema = z.object({
    39→  name: z.string().trim().min(2),
    40→  email: z.string().email(),
    41→  whatsapp: z.string().trim().optional(),
    42→})
    43→
    44→type LeadForm = z.infer<typeof leadSchema>
    45→
    46→const INITIAL_INPUT: CalculatorInput = {
    47→  service: 'website',
    48→  pages: 6,
    49→  languages: 'bilingual',
    50→  threeD: 'no',
    51→  integrations: [],
    52→  automationLevel: 'essential',
    53→}
    54→
    55→export function Calculator() {
    56→  const t = useTranslations('calculator')
    57→  const locale = useLocale()
    58→  const reduced = useReducedMotion()
    59→
    60→  const [step, setStep] = useState<Step>(0)
    61→  const [dir, setDir] = useState<1 | -1>(1)
    62→  const [input, setInput] = useState<CalculatorInput>(INITIAL_INPUT)
    63→  const [form, setForm] = useState<LeadForm>({ name: '', email: '', whatsapp: '' })
    64→  const [errors, setErrors] = useState<{ name?: string; email?: string }>({})
    65→  const [submitting, setSubmitting] = useState(false)
    66→  const [done, setDone] = useState(false)
    67→  const [reference, setReference] = useState<string | null>(null)
    68→
    69→  const result = useMemo(() => computeEstimate(input), [input])
    70→
    71→  // Breakdown line labels — resolved once for type-safety (guide §4.6).
    72→  const breakdownLabels = {
    73→    base: t('result.base'),
    74→    pages: t('result.pages'),
    75→    bilingual: t('result.bilingual'),
    76→    threeD: t('result.threeD'),
    77→    integrations: t('result.integrations'),
    78→    advanced: t('result.advanced'),
    79→  } as const
    80→
    81→  const goNext = () => { setDir(1); setStep((s) => Math.min(2, s + 1) as Step) }
    82→  const goBack = () => { setDir(-1); setStep((s) => Math.max(0, s - 1) as Step) }
    83→
    84→  const toggleIntegration = (key: IntegrationKey) => {
    85→    setInput((p) => ({
    86→      ...p,
    87→      integrations: p.integrations.includes(key)
    88→        ? p.integrations.filter((k) => k !== key)
    89→        : [...p.integrations, key],
    90→    }))
    91→  }
    92→
    93→  const onSubmit = async (e: React.FormEvent) => {
    94→    e.preventDefault()
    95→    const parsed = leadSchema.safeParse(form)
    96→    if (!parsed.success) {
    97→      const fe: { name?: string; email?: string } = {}
    98→      for (const issue of parsed.error.issues) {
    99→        const path = issue.path[0]
   100→        if (path === 'name') fe.name = t('errors.name')
   101→        if (path === 'email') fe.email = t('errors.email')
   102→      }
   103→      setErrors(fe)
   104→      return
   105→    }
   106→    setErrors({})
   107→    setSubmitting(true)
   108→    // Phase 3: real storage — the server recomputes the estimate from the
   109→    // wizard options and returns a reference (guide §2.9). Client numbers
   110→    // are never sent: only the option set travels.
   111→    try {
   112→      const res = await fetch('/api/leads', {
   113→        method: 'POST',
   114→        headers: {
   115→          'Content-Type': 'application/json',
   116→          'x-elyra-locale': locale,
   117→        },
   118→        body: JSON.stringify({
   119→          source: 'calculator',
   120→          name: parsed.data.name,
   121→          email: parsed.data.email,
   122→          whatsapp: parsed.data.whatsapp || undefined,
   123→          service: input.service,
   124→          pages: input.pages,
   125→          languages: input.languages,
   126→          threeD: input.threeD,
   127→          integrations: input.integrations,
   128→          automationLevel: input.automationLevel,
   129→        }),
   130→      })
   131→
   132→      if (res.status === 201) {
   133→        const data = (await res.json()) as { reference?: string }
   134→        setReference(data.reference ?? null)
   135→        setDone(true)
   136→        playSuccess() // Phase 2 sensory feedback — fires on REAL success only
   137→        return
   138→      }
   139→
   140→      // Server rejected: surface translated server-side messages.
   141→      const data = (await res.json().catch(() => null)) as
   142→        | { message?: string; fields?: Record<string, string> }
   143→        | null
   144→      if (res.status === 400 && data?.fields) {
   145→        const fe: { name?: string; email?: string } = {}
   146→        if (data.fields.name) fe.name = data.fields.name
   147→        if (data.fields.email) fe.email = data.fields.email
   148→        setErrors(fe)
   149→      }
   150→      toast.error(t('form.errorTitle'), {
   151→        description: data?.message ?? t('form.errorNetwork'),
   152→      })
   153→    } catch {
   154→      // Network failure — data stays in the form for a retry.
   155→      toast.error(t('form.errorTitle'), { description: t('form.errorNetwork') })
   156→    } finally {
   157→      setSubmitting(false)
   158→    }
   159→  }
   160→
   161→  const slideVariants = reduced
   162→    ? undefined
   163→    : {
   164→        enter: (d: number) => ({ opacity: 0, x: d * 30 }),
   165→        center: { opacity: 1, x: 0 },
   166→        exit: (d: number) => ({ opacity: 0, x: -d * 30 }),
   167→      }
   168→
   169→  return (
   170→    <section id="calculator" className="bg-background py-20 sm:py-28" aria-labelledby="calc-title" aria-label={t('kicker')}>
   171→      <div className="elyra-container max-w-5xl">
   172→        <SectionHeading
   173→          kicker={t('kicker')}
   174→          title={t('title')}
   175→          subtitle={t('subtitle')}
   176→        />
   177→
   178→        {/* Progress */}
   179→        <div className="mt-10 flex items-center justify-between gap-4">
   180→          <p className="text-sm text-muted-foreground">
   181→            {t('step', { current: step + 1, total: 3 })}
   182→          </p>
   183→          <div className="flex flex-1 gap-2">
   184→            {[0, 1, 2].map((s) => (
   185→              <div
   186→                key={s}
   187→                className="h-1.5 flex-1 overflow-hidden rounded-full bg-muted"
   188→                aria-hidden="true"
   189→              >
   190→                <motion.div
   191→                  className="h-full rounded-full bg-primary"
   192→                  initial={false}
   193→                  animate={{ width: step >= s ? '100%' : '0%' }}
   194→                  transition={{ duration: 0.4 }}
   195→                />
   196→              </div>
   197→            ))}
   198→          </div>
   199→        </div>
   200→
   201→        <div className="relative mt-8 overflow-hidden rounded-3xl border border-border bg-card p-6 sm:p-10">
   202→          <AnimatePresence mode="wait" custom={dir}>
   203→            <motion.div
   204→              key={step}
   205→              custom={dir}
   206→              variants={slideVariants}
   207→              initial="enter"
   208→              animate="center"
   209→              exit="exit"
   210→              transition={{ duration: 0.4, ease: [0.16, 1, 0.3, 1] }}
   211→            >
   212→              {/* STEP 0 — Service type */}
   213→              {step === 0 ? (
   214→                <div>
   215→                  <h3 className="text-lg font-semibold">{t('steps.service')}</h3>
   216→                  <div className="mt-6 grid gap-4 sm:grid-cols-3">
   217→                    {SERVICES.map(({ id, icon: Icon }) => {
   218→                      const active = input.service === id
   219→                      const isFull = id === 'full'
   220→                      return (
   221→                        <button
   222→                          key={id}
   223→                          type="button"
   224→                          data-cursor="magnet"
   225→                          onClick={() => setInput((p) => ({ ...p, service: id }))}
   226→                          aria-pressed={active}
   227→                          className={cn(
   228→                            'group relative overflow-hidden rounded-2xl border p-5 text-start transition-all',
   229→                            active
   230→                              ? 'border-primary bg-primary/5 shadow-[0_0_0_1px_var(--color-primary)]'
   231→                              : 'border-border hover:border-primary/40 hover:bg-foreground/[0.02]'
   232→                          )}
   233→                        >
   234→                          {isFull ? (
   235→                            <span className="absolute end-3 top-3 rounded-full bg-primary px-2 py-0.5 text-[10px] font-semibold text-primary-foreground">
   236→                              {t('services.full.popular')}
   237→                            </span>
   238→                          ) : null}
   239→                          <div className={cn(
   240→                            'flex size-11 items-center justify-center rounded-xl transition-colors',
   241→                            active ? 'bg-primary text-primary-foreground' : 'bg-primary/10 text-primary'
   242→                          )}>
   243→                            <Icon className="size-5" aria-hidden="true" />
   244→                          </div>
   245→                          <p className="mt-3 font-semibold">{t(`services.${id}.title`)}</p>
   246→                          <p className="mt-1 text-sm text-muted-foreground">{t(`services.${id}.desc`)}</p>
   247→                          {active ? (
   248→                            <span className="absolute -end-2 -top-2 flex size-6 items-center justify-center rounded-full bg-primary text-primary-foreground">
   249→                              <Check className="size-3.5" aria-hidden="true" />
   250→                            </span>
   251→                          ) : null}
   252→                        </button>
   253→                      )
   254→                    })}
   255→                  </div>
   256→                </div>
   257→              ) : null}
   258→
   259→              {/* STEP 1 — Features */}
   260→              {step === 1 ? (
   261→                <div className="space-y-8">
   262→                  <div>
   263→                    <div className="flex items-center justify-between">
   264→                      <h3 className="text-lg font-semibold">{t('pages')}</h3>
   265→                      <span className="text-sm font-medium text-primary tabular-nums">
   266→                        {t('pagesValue', { count: input.pages })}
   267→                      </span>
   268→                    </div>
   269→                    <div className="mt-4 px-1">
   270→                      <Slider
   271→                        value={[input.pages]}
   272→                        onValueChange={(v) => setInput((p) => ({ ...p, pages: v[0] ?? p.pages }))}
   273→                        min={1}
   274→                        max={20}
   275→                        step={1}
   276→                        aria-label={t('pages')}
   277→                      />
   278→                      <div className="mt-2 flex justify-between text-xs text-muted-foreground">
   279→                        <span>1</span><span>20</span>
   280→                      </div>
   281→                    </div>
   282→                  </div>
   283→
   284→                  <div>
   285→                    <h3 className="text-lg font-semibold">{t('languages')}</h3>
   286→                    <div className="mt-3 grid gap-3 sm:grid-cols-2">
   287→                      {(['single', 'bilingual'] as LanguageOption[]).map((opt) => {
   288→                        const active = input.languages === opt
   289→                        return (
   290→                          <button
   291→                            key={opt}
   292→                            type="button"
   293→                            onClick={() => setInput((p) => ({ ...p, languages: opt }))}
   294→                            aria-pressed={active}
   295→                            className={cn(
   296→                              'rounded-xl border px-4 py-3 text-start transition-colors',
   297→                              active ? 'border-primary bg-primary/5' : 'border-border hover:bg-foreground/[0.02]'
   298→                            )}
   299→                          >
   300→                            <span className="flex items-center gap-2">
   301→                              <span className={cn('flex size-5 items-center justify-center rounded-full border', active ? 'border-primary bg-primary text-primary-foreground' : 'border-border')}>
   302→                                {active ? <Check className="size-3" aria-hidden="true" /> : null}
   303→                              </span>
   304→                              <span className="text-sm font-medium">{t(`languagesOptions.${opt}`)}</span>
   305→                            </span>
   306→                          </button>
   307→                        )
   308→                      })}
   309→                    </div>
   310→                  </div>
   311→
   312→                  <div>
   313→                    <h3 className="text-lg font-semibold">{t('threeD')}</h3>
   314→                    <div className="mt-3 grid gap-3 sm:grid-cols-2">
   315→                      {(['yes', 'no'] as ThreeDOption[]).map((opt) => {
   316→                        const active = input.threeD === opt
   317→                        const disabled = input.service === 'automation'
   318→                        return (
   319→                          <button
   320→                            key={opt}
   321→                            type="button"
   322→                            disabled={disabled}
   323→                            onClick={() => setInput((p) => ({ ...p, threeD: opt }))}
   324→                            aria-pressed={active}
   325→                            className={cn(
   326→                              'rounded-xl border px-4 py-3 text-start transition-colors',
   327→                              disabled && 'opacity-50',
   328→                              active ? 'border-primary bg-primary/5' : 'border-border hover:bg-foreground/[0.02]'
   329→                            )}
   330→                          >
   331→                            <span className="flex items-center gap-2">
   332→                              <span className={cn('flex size-5 items-center justify-center rounded-full border', active ? 'border-primary bg-primary text-primary-foreground' : 'border-border')}>
   333→                                {active ? <Check className="size-3" aria-hidden="true" /> : null}
   334→                              </span>
   335→                              <span className="text-sm font-medium">{t(`threeDOptions.${opt}`)}</span>
   336→                            </span>
   337→                          </button>
   338→                        )
   339→                      })}
   340→                    </div>
   341→                  </div>
   342→
   343→                  <div>
   344→                    <h3 className="text-lg font-semibold">{t('integrations')}</h3>
   345→                    <div className="mt-3 flex flex-wrap gap-2">
   346→                      {INTEGRATIONS.map((key) => {
   347→                        const active = input.integrations.includes(key)
   348→                        return (
   349→                          <button
   350→                            key={key}
   351→                            type="button"
   352→                            onClick={() => toggleIntegration(key)}
   353→                            aria-pressed={active}
   354→                            className={cn(
   355→                              'inline-flex items-center gap-1.5 rounded-full border px-3 py-1.5 text-sm transition-colors',
   356→                              active ? 'border-primary bg-primary/10 text-primary' : 'border-border hover:bg-foreground/[0.02]'
   357→                            )}
   358→                          >
   359→                            <span className={cn('flex size-4 items-center justify-center rounded-full border', active ? 'border-primary bg-primary text-primary-foreground' : 'border-border')}>
   360→                              {active ? <Check className="size-2.5" aria-hidden="true" /> : null}
   361→                            </span>
   362→                            {t(`integrationsOptions.${key}`)}
   363→                          </button>
   364→                        )
   365→                      })}
   366→                    </div>
   367→                  </div>
   368→
   369→                  <div>
   370→                    <h3 className="text-lg font-semibold">{t('automationLevel')}</h3>
   371→                    <div className="mt-3 grid gap-3 sm:grid-cols-2">
   372→                      {(['essential', 'advanced'] as AutomationLevel[]).map((opt) => {
   373→                        const active = input.automationLevel === opt
   374→                        const disabled = input.service === 'website'
   375→                        return (
   376→                          <button
   377→                            key={opt}
   378→                            type="button"
   379→                            disabled={disabled}
   380→                            onClick={() => setInput((p) => ({ ...p, automationLevel: opt }))}
   381→                            aria-pressed={active}
   382→                            className={cn(
   383→                              'rounded-xl border px-4 py-3 text-start transition-colors',
   384→                              disabled && 'opacity-50',
   385→                              active ? 'border-primary bg-primary/5' : 'border-border hover:bg-foreground/[0.02]'
   386→                            )}
   387→                          >
   388→                            <span className="flex items-center gap-2">
   389→                              <span className={cn('flex size-5 items-center justify-center rounded-full border', active ? 'border-primary bg-primary text-primary-foreground' : 'border-border')}>
   390→                                {active ? <Check className="size-3" aria-hidden="true" /> : null}
   391→                              </span>
   392→                              <span className="text-sm font-medium">{t(`automationLevelOptions.${opt}`)}</span>
   393→                            </span>
   394→                          </button>
   395→                        )
   396→                      })}
   397→                    </div>
   398→                  </div>
   399→                </div>
   400→              ) : null}
   401→
   402→              {/* STEP 2 — Result */}
   403→              {step === 2 ? (
   404→                <div>
   405→                  {done ? (
   406→                    <div className="flex flex-col items-center py-8 text-center">
   407→                      <div className="flex size-16 items-center justify-center rounded-full bg-g-green/15 text-g-green">
   408→                        <Check className="size-8" aria-hidden="true" />
   409→                      </div>
   410→                      <h3 className="mt-5 text-2xl font-semibold">{t('form.successTitle')}</h3>
   411→                      <p className="mt-2 max-w-md text-muted-foreground">{t('form.successDesc')}</p>
   412→                      {reference ? (
   413→                        <p className="mt-3 rounded-full border border-g-green/30 bg-g-green/5 px-4 py-1.5 font-mono text-sm font-semibold tracking-wide text-g-green">
   414→                          {t('form.successReference', { reference })}
   415→                        </p>
   416→                      ) : null}
   417→                      <button
   418→                        type="button"
   419→                        onClick={() => { setDone(false); setReference(null); setStep(0); setInput(INITIAL_INPUT); setForm({ name: '', email: '', whatsapp: '' }) }}
   420→                        className="mt-6 inline-flex h-11 items-center gap-2 rounded-full border border-border px-5 text-sm font-medium hover:bg-foreground/5"
   421→                      >
   422→                        <RotateCw className="size-4" aria-hidden="true" />
   423→                        {t('form.successRestart')}
   424→                      </button>
   425→                    </div>
   426→                  ) : (
   427→                    <div className="grid gap-8 lg:grid-cols-2">
   428→                      <div>
   429→                        <div className="rounded-2xl border border-border bg-gradient-to-br from-primary/5 to-transparent p-6">
   430→                          {/* WS-4: animated SVG ring gauges */}
   431→                          <div className="mb-6 flex items-center justify-center gap-12">
   432→                            <RingGauge
   433→                              fraction={Math.min(1, result.max / 20000)}
   434→                              formatValue={(n) => formatMoney(Math.round(n), locale)}
   435→                              label={t('result.budget')}
   436→                              color="#0071E3"
   437→                              isRtl={locale === 'ar'}
   438→                            />
   439→                            <RingGauge
   440→                              fraction={Math.min(1, result.weeksMax / 12)}
   441→                              formatValue={(n) => `${Math.round(n)}`}
   442→                              label={t('result.duration')}
   443→                              color="#34A853"
   444→                              isRtl={locale === 'ar'}
   445→                            />
   446→                          </div>
   447→                          <p className="text-xs font-medium uppercase tracking-wider text-muted-foreground">
   448→                            {t('result.budget')}
   449→                          </p>
   450→                          <p className="mt-2 text-3xl font-bold tracking-tight text-primary sm:text-4xl">
   451→                            {formatMoney(result.min, locale)} – {formatMoney(result.max, locale)}
   452→                          </p>
   453→                          <p className="mt-4 text-xs font-medium uppercase tracking-wider text-muted-foreground">
   454→                            {t('result.duration')}
   455→                          </p>
   456→                          <p className="mt-1 text-xl font-semibold">
   457→                            {t('result.weeks', { min: result.weeksMin, max: result.weeksMax })}
   458→                          </p>
   459→                        </div>
   460→
   461→                        <div className="mt-4">
   462→                          <p className="text-sm font-semibold">{t('result.breakdownTitle')}</p>
   463→                          <ul className="mt-2 space-y-1.5 text-sm">
   464→                            {result.breakdown.map((line) => (
   465→                              <li key={line.labelKey} className="flex items-center justify-between gap-2">
   466→                                <span className="text-muted-foreground">{breakdownLabels[line.labelKey as keyof typeof breakdownLabels]}</span>
   467→                                <span className="tabular-nums">
   468→                                  {formatMoney(line.min, locale)} – {formatMoney(line.max, locale)}
   469→                                </span>
   470→                              </li>
   471→                            ))}
   472→                          </ul>
   473→                        </div>
   474→                        <p className="mt-4 flex items-start gap-1.5 text-xs text-muted-foreground">
   475→                          <AlertCircle className="mt-0.5 size-3.5 shrink-0" aria-hidden="true" />
   476→                          {t('result.disclaimer')}
   477→                        </p>
   478→                      </div>
   479→
   480→                      <form onSubmit={onSubmit} className="rounded-2xl border border-border p-6" noValidate>
   481→                        <h3 className="text-lg font-semibold">{t('form.title')}</h3>
   482→                        <div className="mt-4 space-y-4">
   483→                          <div>
   484→                            <Label htmlFor="calc-name" className="text-sm">{t('form.name')}</Label>
   485→                            <Input
   486→                              id="calc-name"
   487→                              value={form.name}
   488→                              onChange={(e) => setForm((f) => ({ ...f, name: e.target.value }))}
   489→                              autoComplete="name"
   490→                              aria-invalid={!!errors.name}
   491→                              aria-describedby={errors.name ? 'calc-name-err' : undefined}
   492→                              className="mt-1.5"
   493→                            />
   494→                            {errors.name ? (
   495→                              <p id="calc-name-err" role="alert" className="mt-1 text-xs text-destructive">{errors.name}</p>
   496→                            ) : null}
   497→                          </div>
   498→                          <div>
   499→                            <Label htmlFor="calc-email" className="text-sm">{t('form.email')}</Label>
   500→                            <Input
   501→                              id="calc-email"
   502→                              type="email"
   503→                              value={form.email}
   504→                              onChange={(e) => setForm((f) => ({ ...f, email: e.target.value }))}
   505→                              autoComplete="email"
   506→                              aria-invalid={!!errors.email}
   507→                              aria-describedby={errors.email ? 'calc-email-err' : undefined}
