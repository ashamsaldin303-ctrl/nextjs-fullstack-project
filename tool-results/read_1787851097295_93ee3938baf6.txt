    67→/* ------------------------------------------------------------------ *
    68→ * GLSL — Ashima 3D simplex noise (Stefan Gustavson / Ian McEwan, MIT)
    69→ * + finite-difference curl. Written in GLSL1 style (attribute/varying)
    70→ * to match the existing shaderMaterial convention in this repo.

    72→const NOISE_GLSL = /* glsl */ `
    73→  vec3 mod289(vec3 x) {
    74→    return x - floor(x * (1.0 / 289.0)) * 289.0;
    75→  }
    76→  vec4 mod289(vec4 x) {
    77→    return x - floor(x * (1.0 / 289.0)) * 289.0;
    78→  }
    79→  vec4 permute(vec4 x) {
    80→    return mod289(((x * 34.0) + 1.0) * x);
    81→  }
    82→  vec4 taylorInvSqrt(vec4 r) {
    83→    return 1.79284291400159 - 0.85373472095314 * r;
    84→  }
    85→  float snoise(vec3 v) {
    86→    const vec2 C = vec2(1.0 / 6.0, 1.0 / 3.0);
    87→    const vec4 D = vec4(0.0, 0.5, 1.0, 2.0);
    88→
    89→    // First corner
    90→    vec3 i  = floor(v + dot(v, C.yyy));
    91→    vec3 x0 = v - i + dot(i, C.xxx);
    92→
    93→    // Other corners
    94→    vec3 g = step(x0.yzx, x0.xyz);
    95→    vec3 l = 1.0 - g;
    96→    vec3 i1 = min(g.xyz, l.zxy);
    97→    vec3 i2 = max(g.xyz, l.zxy);
    98→
    99→    vec3 x1 = x0 - i1 + C.xxx;
   100→    vec3 x2 = x0 - i2 + C.yyy;
   101→    vec3 x3 = x0 - D.yyy;
   102→
   103→    // Permutations
   104→    i = mod289(i);
   105→    vec4 p = permute(permute(permute(
   106→              i.z + vec4(0.0, i1.z, i2.z, 1.0))
   107→            + i.y + vec4(0.0, i1.y, i2.y, 1.0))
   108→            + i.x + vec4(0.0, i1.x, i2.x, 1.0));
   109→
   110→    // Gradients: 7x7 points over a square, mapped onto an octahedron.
   111→    float n_ = 0.142857142857;
   112→    vec3 ns = n_ * D.wyz - D.xzx;
   113→
   114→    vec4 j = p - 49.0 * floor(p * ns.z * ns.z);
   115→
   116→    vec4 x_ = floor(j * ns.z);
   117→    vec4 y_ = floor(j - 7.0 * x_);
   118→
   119→    vec4 x = x_ * ns.x + ns.yyyy;
   120→    vec4 y = y_ * ns.x + ns.yyyy;
   121→    vec4 h = 1.0 - abs(x) - abs(y);
   122→
   123→    vec4 b0 = vec4(x.xy, y.xy);
   124→    vec4 b1 = vec4(x.zw, y.zw);
   125→
   126→    vec4 s0 = floor(b0) * 2.0 + 1.0;
   127→    vec4 s1 = floor(b1) * 2.0 + 1.0;
   128→    vec4 sh = -step(h, vec4(0.0));
   129→
   130→    vec4 a0 = b0.xzyw + s0.xzyw * sh.xxyy;
   131→    vec4 a1 = b1.xzyw + s1.xzyw * sh.zzww;
   132→
   133→    vec3 p0 = vec3(a0.xy, h.x);
   134→    vec3 p1 = vec3(a0.zw, h.y);
   135→    vec3 p2 = vec3(a1.xy, h.z);
   136→    vec3 p3 = vec3(a1.zw, h.w);
   137→
   138→    // Normalise gradients
   139→    vec4 norm = taylorInvSqrt(vec4(dot(p0, p0), dot(p1, p1), dot(p2, p2), dot(p3, p3)));
   140→    p0 *= norm.x;
   141→    p1 *= norm.y;
   142→    p2 *= norm.z;
   143→    p3 *= norm.w;
   144→
   145→    // Mix final noise value
   146→    vec4 m = max(0.6 - vec4(dot(x0, x0), dot(x1, x1), dot(x2, x2), dot(x3, x3)), 0.0);
   147→    m = m * m;
   148→    return 42.0 * dot(m * m, vec4(dot(p0, x0), dot(p1, x1), dot(p2, x2), dot(p3, x3)));
   149→  }
   150→
   151→  // Vector noise — three offset simplex samples packed into a vec3.
   152→  vec3 snoiseVec3(vec3 x) {
   153→    float s  = snoise(x);
   154→    float s1 = snoise(vec3(x.y - 19.1, x.z + 33.4, x.x + 47.2));
   155→    float s2 = snoise(vec3(x.z + 74.2, x.x - 124.5, x.y + 99.4));
   156→    return vec3(s, s1, s2);
   157→  }
   158→
   159→  // Curl (rotational derivative) of the vector noise via central finite
   160→  // differences. Divergence-free by construction — flow never bunches
   161→  // into sinks, and neighbours share velocity => coherent ribbons.
   162→  vec3 curlNoise(vec3 p) {
   163→    const float e = 0.12;
   164→    vec3 dx = vec3(e, 0.0, 0.0);
   165→    vec3 dy = vec3(0.0, e, 0.0);
   166→    vec3 dz = vec3(0.0, 0.0, e);
   167→
   168→    vec3 px0 = snoiseVec3(p - dx);
   169→    vec3 px1 = snoiseVec3(p + dx);
   170→    vec3 py0 = snoiseVec3(p - dy);
   171→    vec3 py1 = snoiseVec3(p + dy);
   172→    vec3 pz0 = snoiseVec3(p - dz);
   173→    vec3 pz1 = snoiseVec3(p + dz);
   174→
   175→    float x = py1.z - py0.z - pz1.y + pz0.y;
   176→    float y = pz1.x - pz0.x - px1.z + px0.z;
   177→    float z = px1.y - px0.y - py1.x + py0.x;
   178→
   179→    vec3 curl = vec3(x, y, z) / (2.0 * e);
   180→    // safe normalise — never NaN, even if the finite differences cancel
   181→    return curl / max(length(curl), 1e-4);
   182→  }
   183→`

   185→const PARTICLE_VERTEX = /* glsl */ `
   186→  ${NOISE_GLSL}
   187→
   188→  attribute float aScale;
   189→  attribute float aSpeed;
   190→  attribute vec3 aColor;
   191→  uniform float uTime;
   192→  uniform vec3 uMouse;
   193→  uniform float uPixelRatio;
   194→  uniform float uScroll;
   195→  varying vec3 vColor;
   196→  varying float vAlpha;
   197→
   198→  void main() {
   199→    vec3 p = position;
   200→
   201→    // --- curl flow field -------------------------------------------------
   202→    // Sample the field in a slowly-translating noise space: the coherent
   203→    // structures sweep past the volume so each particle's velocity keeps
   204→    // re-forming — a streaming current, not in-place jitter.
   205→    vec3 q = p * 0.16 + vec3(0.0, -uTime * 0.18, 0.0);
   206→    vec3 vel = curlNoise(q);
   207→
   208→    // Per-particle amplitude (aSpeed 0.2…0.8) + a gentle swell as the
   209→    // methodology section scrolls through (uScroll 0…1).
   210→    float amp = (1.1 + aSpeed * 1.6) * (1.0 + uScroll * 0.4);
   211→    p += vel * amp;
   212→
   213→    // --- slow global drift ------------------------------------------------
   214→    // Embers rising through a tall wrapped domain; the edge fade below
   215→    // dissolves particles at the recycle seam so the wrap never pops.
   216→    const float DOMAIN = 20.0;
   217→    float y = mod(p.y + uTime * (0.22 + aSpeed * 0.18), DOMAIN) - DOMAIN * 0.5;
   218→    p.y = y;
   219→
   220→    // subtle attraction toward the mouse plane (kept from v1)
   221→    p += uMouse * 0.18;
   222→
   223→    vec4 mv = modelViewMatrix * vec4(p, 1.0);
   224→    gl_Position = projectionMatrix * mv;
   225→    float size = aScale * uPixelRatio * (28.0 / -mv.z);
   226→    gl_PointSize = clamp(size, 1.0, 64.0);
   227→    vColor = aColor;
   228→    // fade by distance (kept) + fade near the wrap seam
   229→    float depthFade = clamp(1.0 - (-mv.z - 3.0) / 22.0, 0.0, 1.0);
   230→    float edgeFade = 1.0 - smoothstep(8.5, 10.0, abs(y));
   231→    vAlpha = depthFade * edgeFade;
   232→  }
   233→`

   263→
   264→const SILK_FRAGMENT = /* glsl */ `
   265→  // Domain-warped fbm "silk" (Inigo Quilez warp structure): the sampling
   266→  // domain is bent through two offset fields — q = fbm(p + drift), then
   267→  // r = fbm(p*1.3 + q + counter-drift) — and the final sample
   268→  // f = fbm(p + r*2.2) flows as broad luminous bands. A gaussian vortex
   269→  // around the cursor stirs the whole domain (liquid-silk feel); graded
   270→  // brand blues, with the green confined to thin warp-crossing filaments.
   271→  uniform float uTime;
   272→  uniform vec2 uMouse;   // world-space (plane-local) cursor position, CPU-lerped
   273→  uniform vec3 uColorA;  // primary blue #4285F4 — silk body
   274→  uniform vec3 uColorB;  // brand green #34A853 — warp-crossing filaments only
   275→  uniform vec3 uColorC;  // deep blue #0A2A5E — low-energy undertone
   276→  uniform float uScroll; // methodology bridge 0…1 → gentle intensity swell
   277→  varying vec2 vUv;
   278→  varying vec2 vPos;
   279→
   280→  float hash(vec2 p) {
   281→    return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453123);
   282→  }
   283→
   284→  float vnoise(vec2 p) {
   285→    vec2 i = floor(p);
   286→    vec2 f = fract(p);
   287→    vec2 u = f * f * (3.0 - 2.0 * f);
   288→    float a = hash(i);
   289→    float b = hash(i + vec2(1.0, 0.0));
   290→    float c = hash(i + vec2(0.0, 1.0));
   291→    float d = hash(i + vec2(1.0, 1.0));
   292→    return mix(mix(a, b, u.x), mix(c, d, u.x), u.y);
   293→  }
   294→
   295→  // 5 octaves; the fixed rotation between octaves kills axis alignment.
   296→  float fbm(vec2 p) {
   297→    float v = 0.0;
   298→    float a = 0.5;
   299→    for (int i = 0; i < 5; i++) {
   300→      v += a * vnoise(p);
   301→      p = mat2(0.8, 0.6, -0.6, 0.8) * p * 2.03;
   302→      a *= 0.5;
   303→    }
   304→    return v;
   305→  }
   306→
   307→  void main() {
   308→    float t = uTime * 0.13;
   309→
   310→    // --- cursor vortex --------------------------------------------------
   311→    // Stir the sampling domain around the pointer: rotate the plane-local
   312→    // position around uMouse under a gaussian falloff (~6 world units).
   313→    // Angular displacement tops out at 0.35 rad, so the total shift stays
   314→    // under ~0.7 world units — the composition never turns chaotic.
   315→    vec2 toM = vPos - uMouse;
   316→    float md = length(toM);
   317→    float stir = exp(-(md * md) / 24.0);
   318→    float ang = 0.35 * stir;
   319→    float ca = cos(ang);
   320→    float sa = sin(ang);
   321→    vec2 pos = uMouse + mat2(ca, -sa, sa, ca) * toM;
   322→
   323→    // --- two-level domain warp (Inigo Quilez) ----------------------------
   324→    // q and r are offset fields sampled from the same fbm; f is the fbm
   325→    // sampled through BOTH warps. The two time offsets counter-drift, so
   326→    // the fields shear past each other and the fold pattern keeps
   327→    // re-forming instead of looping.
   328→    vec2 p = pos * 0.08;
   329→    float q = fbm(p + vec2(t * 0.6, -t * 0.4));
   330→    float r = fbm(p * 1.3 + vec2(q * 1.7, q * 1.1) + vec2(-t * 0.5, t * 0.3) + 2.7);
   331→    float f = fbm(p + r * 2.2);
   332→
   333→    // --- window / vignette / energy --------------------------------------
   334→    // Vertical melt at the viewport edges (aspect-safe: the fov is
   335→    // vertical, so ±10.5 world units at the plane distance is the screen
   336→    // height) + radial vignette on the light itself, nudged a touch high
   337→    // so the silk breathes brightest in the upper-middle band behind the
   338→    // headline.
   339→    float vert = smoothstep(-10.5, -5.0, vPos.y) * (1.0 - smoothstep(5.0, 10.5, vPos.y));
   340→    float vig = 1.0 - smoothstep(0.15, 0.42, length(vUv - vec2(0.5, 0.44)));
   341→    float lift = 0.85 + 0.30 * smoothstep(-4.0, 5.0, vPos.y);
   342→    float swell = 1.0 + uScroll * 0.25;
   343→
   344→    // --- masks ------------------------------------------------------------
   345→    float body  = smoothstep(0.36, 0.78, f);               // broad fabric body
   346→    float ridge = smoothstep(0.60, 0.92, f);               // luminous band ridges
   347→    float deep  = 1.0 - smoothstep(0.18, 0.55, f);         // low-energy zones
   348→    float sheen = 0.62 + 0.38 * smoothstep(0.25, 0.85, q); // q shimmers the ridges
   349→
   350→    // Green lives ONLY in thin filaments: where the second warp field r
   351→    // crosses a narrow level band while the body is lit — curved
   352→    // interference hairlines that drift as the warp fields slide.
   353→    float lvl = smoothstep(0.46, 0.55, r) * (1.0 - smoothstep(0.60, 0.70, r));
   354→    float filament = lvl * body * sheen;
   355→
   356→    // --- brand grades (derived from the uniform palette) ------------------
   357→    vec3 midBlue   = mix(uColorC, uColorA, 0.55);   // ≈ #0071E3 band depth
   358→    vec3 lightBlue = mix(uColorA, vec3(1.0), 0.22); // ≈ #60A5FA ridge sheen
   359→
   360→    vec3 col = vec3(0.0); // additive: the CSS stage shows through
   361→
   362→    // deep undertone between the bands
   363→    col += uColorC * deep * deep * 0.42 * vert * vig;
   364→
   365→    // silk body — hue graded by q, lifting into the light-blue sheen on
   366→    // the ridges
   367→    vec3 bandCol = mix(midBlue, uColorA, smoothstep(0.30, 0.70, q));
   368→    col += bandCol * body * (0.26 + 0.20 * ridge) * vert * vig * lift * swell;
   369→    col += lightBlue * body * ridge * sheen * 0.30 * vert * vig * lift * swell;
   370→
   371→    // warp-crossing filaments
   372→    col += uColorB * filament * 0.48 * vert * vig * lift * swell;
   373→
   374→    // broad cursor wash (blue family — green is reserved for filaments)
   375→    col += mix(uColorC, uColorA, 0.75) * exp(-md * md / 220.0) * 0.14 * swell;
   376→
   377→    // intensity governor: hard per-channel cap — the additive layer stays
   378→    // premium-dark even where masks coincidentally stack
   379→    col = min(col, vec3(0.50));
   380→
   381→    gl_FragColor = vec4(col, 1.0);
   382→  }

   385→function Particles({
   386→  mouse,
   387→  count,
   388→  dprMax = 2,
   389→}: {
   390→  mouse: React.MutableRefObject<{ x: number; y: number; tx: number; ty: number }>
   391→  /** Tier-dependent particle budget (desktop 4500 / mobile 1800) — the
   392→   *  parent keys the component on it, so a tier flip rebuilds the buffers. */
   393→  count: number
   394→  /** Tier-dependent upper bound for the uPixelRatio uniform — MUST track the
   395→   *  Canvas dpr prop (desktop 2 / mobile 1.5) or gl_PointSize drifts ~33%
   396→   *  oversize on retina phones (FIX L2-V P3). The parent's key remount on
   397→   *  tier flip re-runs the uniforms memo below. */
   398→  dprMax?: number
   399→}) {
   400→  const pointsRef = useRef<THREE.Points>(null)
   401→  const mouseVec = useRef(new THREE.Vector3(0, 0, 0))
   402→
   403→  // Particle geometry is generated ONCE per mount with Math.random — an impure
   404→  // call that React 19 only permits inside a lazy state initializer (never in
   405→  // render body or useMemo factories). See react-hooks/purity.
   406→  const [buffers] = useState(() => {
   407→    const positions = new Float32Array(count * 3)
   408→    const scales = new Float32Array(count)
   409→    const speeds = new Float32Array(count)
   410→    const colors = new Float32Array(count * 3)
   411→    for (let i = 0; i < count; i++) {
   412→      // Spread in a wide box around the camera
   413→      const radius = 4 + Math.random() * 9
   414→      const theta = Math.random() * Math.PI * 2
   415→      const phi = Math.acos(2 * Math.random() - 1)
   416→      positions[i * 3 + 0] = radius * Math.sin(phi) * Math.cos(theta)
   417→      positions[i * 3 + 1] = radius * Math.sin(phi) * Math.sin(theta) * 0.6
   418→      positions[i * 3 + 2] = radius * Math.cos(phi) * 0.5 - 2
   419→      scales[i] = Math.random() * 1.4 + 0.4
   420→      speeds[i] = Math.random() * 0.6 + 0.2
   421→      const c = COLORS[Math.floor(Math.random() * COLORS.length)]
   422→      if (!c) continue
   423→      colors[i * 3 + 0] = c.r
   424→      colors[i * 3 + 1] = c.g
   425→      colors[i * 3 + 2] = c.b
   426→    }
   427→    return { positions, scales, speeds, colors }
   428→  })
   429→  const { positions, scales, speeds, colors } = buffers
   430→
   431→  const uniforms = useMemo(
   432→    () => ({
   433→      uTime: { value: 0 },
   434→      uMouse: { value: new THREE.Vector3(0, 0, 0) },
   435→      // FIX(L2-V P3): tier-aware — clamped to the SAME ceiling the renderer
   436→      // uses (dprMax), so gl_PointSize matches the actual render scale on
   437→      // every tier. Desktop path is byte-equivalent (min(dpr, 2)).
   438→      uPixelRatio: { value: Math.min(typeof window !== 'undefined' ? window.devicePixelRatio : 1, dprMax) },
   439→      uScroll: { value: 0 },
   440→    }),
   441→    [dprMax]
   442→  )
   443→
   444→  useFrame((state, delta) => {
   445→    if (!pointsRef.current) return
   446→    // Three.js shader uniforms are mutated per-frame — the canonical R3F pattern.
   447→    // eslint-disable-next-line react-hooks/immutability
   448→    uniforms.uTime.value += delta
   449→    // uPixelRatio live sync (LOOP-3 FIX 6): state.viewport.dpr is R3F's
   450→    // clamped actual dpr, so gl_PointSize always matches the render scale —
   451→    // even when the mobile tier flips the Canvas dpr prop mid-session (the
   452→    // memoized initial value above is only the first-frame best guess).
   453→    // Covered by the uTime disable above (the immutability rule reports one
   454→    // finding per memoized value).
   455→    uniforms.uPixelRatio.value = state.viewport.dpr
   456→
   457→    // Lerp mouse target into the actual mouse vector (smoothing) —
   458→    // dt-compensated (was ×0.04/frame; k=-60·ln(1-0.04)≈2.449) so 120Hz
   459→    // displays don't double the easing speed (LOOP-3 FIX 10).
   460→    const mouseS = 1 - Math.exp(-2.449 * delta)
   461→    mouseVec.current.x += (mouse.current.tx - mouseVec.current.x) * mouseS
   462→    mouseVec.current.y += (mouse.current.ty - mouseVec.current.y) * mouseS
   463→    uniforms.uMouse.value.copy(mouseVec.current)
   464→
   465→    // Scroll bridge (methodology section) → gentle flow swell, lerped —
   466→    // dt-compensated (was ×0.08/frame; k≈5.003).
   467→    const scrollTarget = getHeroScroll()
   468→    const scrollS = 1 - Math.exp(-5.003 * delta)
   469→    uniforms.uScroll.value += (scrollTarget - uniforms.uScroll.value) * scrollS
   470→
   471→    // Gentle group tilt toward mouse — dt-compensated (was ×0.05/frame;
   472→    // k≈3.078).
   473→    const tiltS = 1 - Math.exp(-3.078 * delta)
   474→    pointsRef.current.rotation.y += ((mouseVec.current.x * 0.25) - pointsRef.current.rotation.y) * tiltS
   475→    pointsRef.current.rotation.x += ((mouseVec.current.y * 0.15) - pointsRef.current.rotation.x) * tiltS
   476→  })
   477→
   478→  return (
   479→    <points ref={pointsRef} frustumCulled={false}>
   480→      <bufferGeometry>
   481→        <bufferAttribute attach="attributes-position" args={[positions, 3]} />
   482→        <bufferAttribute attach="attributes-aScale" args={[scales, 1]} />
   483→        <bufferAttribute attach="attributes-aSpeed" args={[speeds, 1]} />
   484→        <bufferAttribute attach="attributes-aColor" args={[colors, 3]} />
   485→      </bufferGeometry>
   486→      <shaderMaterial
   487→        transparent
   488→        depthWrite={false}
   489→        blending={THREE.AdditiveBlending}
   490→        uniforms={uniforms}
   491→        vertexShader={PARTICLE_VERTEX}
   492→        fragmentShader={PARTICLE_FRAGMENT}
   493→      />
   494→    </points>
   495→  )
   496→}

   498→function SilkBackdrop({ mouse }: { mouse: React.MutableRefObject<{ x: number; y: number; tx: number; ty: number }> }) {
   499→  const mouseVec = useRef(new THREE.Vector2(0, 0))
   500→
   501→  const uniforms = useMemo(
   502→    () => ({
   503→      uTime: { value: 0 },
   504→      uMouse: { value: new THREE.Vector2(0, 0) },
   505→      uColorA: { value: new THREE.Color('#4285F4') }, // silk body blue
   506→      uColorB: { value: new THREE.Color('#34A853') }, // warp-crossing filaments only
   507→      uColorC: { value: new THREE.Color('#0A2A5E') }, // deep undertone
   508→      uScroll: { value: 0 },
   509→    }),
   510→    []
   511→  )
   512→
   513→  useFrame((state, delta) => {
   514→    // Three.js shader uniforms are mutated per-frame — the canonical R3F pattern.
   515→    // eslint-disable-next-line react-hooks/immutability
   516→    uniforms.uTime.value += delta
   517→
   518→    // Project the pointer onto the silk plane in world space (aspect-
   519→    // correct; numbers only — no per-frame allocations). The CPU lerp
   520→    // smooths the vortex centre, so the stir glides instead of snapping —
   521→    // dt-compensated (was ×0.05/frame; k≈3.078; LOOP-3 FIX 10).
   522→    const aspect = state.size.width / state.size.height
   523→    const halfH = Math.tan(CAMERA_FOV_RAD * 0.5) * SILK_PLANE_DIST
   524→    const silkS = 1 - Math.exp(-3.078 * delta)
   525→    mouseVec.current.x += (mouse.current.tx * halfH * aspect - mouseVec.current.x) * silkS
   526→    mouseVec.current.y += (mouse.current.ty * halfH - mouseVec.current.y) * silkS
   527→    uniforms.uMouse.value.copy(mouseVec.current)
   528→
   529→    // Scroll bridge (methodology section) → gentle silk swell, lerped
   530→    // with the same pattern Particles uses — dt-compensated (was
   531→    // ×0.08/frame; k≈5.003).
   532→    const scrollTarget = getHeroScroll()
   533→    const scrollS = 1 - Math.exp(-5.003 * delta)
   534→    uniforms.uScroll.value += (scrollTarget - uniforms.uScroll.value) * scrollS
   535→  })
   536→
   537→  return (
   538→    <mesh position={[0, 0, SILK_PLANE_Z]} renderOrder={-1} frustumCulled={false}>
   539→      {/* 80 wide covers 32:9 ultrawides (FIX(2-b), L1-D P3): the visible
   540→          half-width at the dolly's farthest z (dist 15.8, aspect 3.56) is
   541→          ≈39.4 world units < 40 — one fullscreen 1-segment quad, so the
   542→          extra width is cost-free. Height stays 32 (vertical fov). */}
   543→      <planeGeometry args={[80, 32]} />
   544→      {/* True additive over the PAGE, not just over the framebuffer: the
   545→          canvas is transparent, so plain AdditiveBlending (SRC_ALPHA, ONE)
   546→          would accumulate alpha≈1 across the fullscreen plane and hide the
   547→          CSS stage underneath. CustomBlending with (ONE,ONE / ZERO,ONE)
   548→          adds light to the color buffer while leaving canvas alpha at 0 —
   549→          the deep background, hero-fallback gradient and blueprint
   550→          spotlight grid all stay visible beneath the silk. */}
   551→      <shaderMaterial
   552→        transparent
   553→        depthWrite={false}
   554→        depthTest={false}
   555→        blending={THREE.CustomBlending}
   556→        blendSrc={THREE.OneFactor}
   557→        blendDst={THREE.OneFactor}
   558→        blendSrcAlpha={THREE.ZeroFactor}
   559→        blendDstAlpha={THREE.OneFactor}
   560→        uniforms={uniforms}
   561→        vertexShader={SILK_VERTEX}
   562→        fragmentShader={SILK_FRAGMENT}
   563→      />
   564→    </mesh>
   565→  )
   566→}

   568→/** Batch 3 item 15 — scroll-driven camera dolly. Eases the camera +0.8
   569→ *  units back as the methodology section's scroll progress (bridge)
   570→ *  goes 0→1 — a subtle pull-away, RTL-safe (pure view-axis motion). */
   571→function ScrollDolly() {
   572→  useFrame((state, delta) => {
   573→    const targetZ = BASE_CAMERA_Z + getHeroScroll() * 0.8
   574→    // dt-compensated (was ×0.05/frame; k≈3.078; LOOP-3 FIX 10)
   575→    const dollyS = 1 - Math.exp(-3.078 * delta)
   576→    state.camera.position.z += (targetZ - state.camera.position.z) * dollyS
   577→  })
   578→  return null
   579→}

   581→/** Context-loss guard — `preventDefault()` marks the event as handled so
   582→ *  the browser keeps the canvas alive for a possible restore (and stops
   583→ *  the default console error spam); we log once for diagnostics. */
   584→function ContextLossGuard() {
   585→  const gl = useThree((s) => s.gl)
   586→  useEffect(() => {
   587→    const canvas = gl.domElement
   588→    const onLost = (e: Event) => {
   589→      e.preventDefault()
   590→      console.warn('[HeroCanvas] WebGL context lost')
   591→    }
   592→    canvas.addEventListener('webglcontextlost', onLost)
   593→    return () => canvas.removeEventListener('webglcontextlost', onLost)
   594→  }, [gl])
   595→  return null
   596→}

   606→export function HeroCanvas({ active }: HeroCanvasProps) {
   607→  const mouse = useRef({ x: 0, y: 0, tx: 0, ty: 0 })
   608→  const [glAvailable, setGlAvailable] = useState(true)
   609→  // FIX(2-b), L1-D P2: mobile tier — coarse-pointer/<768px viewports get
   610→  // 1800 particles + dpr ≤ 1.5 (was: full-fat 4500 + dpr ≤ 2 on phones).
   611→  const mobileTier = useMobileTier()
   612→  const particleCount = mobileTier ? PARTICLE_COUNT_MOBILE : PARTICLE_COUNT_DESKTOP
   613→
   614→  useEffect(() => {
   615→    // WebGL feature detection — deferred to rAF so the setState isn't synchronous
   616→    // inside the effect body (hydration-safe + lint-compliant). LOOP-3 FIX 8:
   617→    // the inline createElement('canvas') probe now delegates to the shared
   618→    // module-memoized probeWebGL() (src/lib/use-webgl.ts) — identical timing
   619→    // semantics (true-first, flips false at the first rAF only on failure),
   620→    // one probe implementation for the whole app.
   621→    let cancelled = false
   622→    const id = requestAnimationFrame(() => {
   623→      if (cancelled) return
   624→      if (!probeWebGL()) setGlAvailable(false)
   625→    })
   626→    return () => {
   627→      cancelled = true
   628→      cancelAnimationFrame(id)
   629→    }
   630→  }, [])
   631→
   632→  useEffect(() => {
   633→    const onMove = (e: PointerEvent) => {
   634→      mouse.current.tx = (e.clientX / window.innerWidth) * 2 - 1
   635→      mouse.current.ty = -((e.clientY / window.innerHeight) * 2 - 1)
   636→    }
   637→    window.addEventListener('pointermove', onMove, { passive: true })
   638→    return () => window.removeEventListener('pointermove', onMove)
   639→  }, [])
   640→
   641→  if (!glAvailable) return null
   642→
   643→  return (
   644→    <div className="absolute inset-0 -z-0">
   645→      <Canvas
   646→        dpr={mobileTier ? [1, 1.5] : [1, 2]}
   647→        frameloop={active ? 'always' : 'never'}
   648→        gl={{ antialias: false, alpha: true, powerPreference: 'high-performance' }}
   649→        camera={{ position: [0, 0, BASE_CAMERA_Z], fov: 70 }}
   650→        style={{ background: 'transparent' }}
   651→      >
   652→        <SilkBackdrop mouse={mouse} />
   653→        {/* key: the buffers are sized once per mount in the lazy useState
   654→            initializer, so a tier flip remounts the Points with the new
   655→            count — the Canvas/GL context itself survives the change */}
   656→        <Particles key={particleCount} mouse={mouse} count={particleCount} dprMax={mobileTier ? 1.5 : 2} />
   657→        <ScrollDolly />
   658→        <ContextLossGuard />
   659→      </Canvas>
   660→    </div>
   661→  )
   662→}
