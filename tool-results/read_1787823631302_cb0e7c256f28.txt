   102→   102→const JSON_CLS = {
   103→   103→  key: 'text-g-blue',

   109→   109→interface JsonToken {
   110→   110→  text: string
   111→   111→  cls: string
   112→   112→}
   113→   113→
   114→   114→function isJsonObject(value: JsonValue): value is { [key: string]: JsonValue } {
   115→   115→  return typeof value === 'object'
   116→   116→}
   117→   117→
   118→   118→function scalarToken(value: Exclude<JsonValue, { [key: string]: JsonValue }>): JsonToken {
   119→   119→  if (typeof value === 'string') return { text: JSON.stringify(value), cls: JSON_CLS.string }
   120→   120→  if (typeof value === 'number') return { text: String(value), cls: JSON_CLS.number }
   121→   121→  return { text: String(value), cls: JSON_CLS.number } // boolean
   122→   122→}
   123→   123→
   124→   124→/** Pretty-prints a JsonValue into syntax-colored display lines (2-space indent). */
   125→   125→function jsonLines(value: JsonValue, indent = 0): JsonToken[][] {
   126→   126→  if (!isJsonObject(value)) return [[scalarToken(value)]]
   127→   127→  const pad = '  '.repeat(indent)
   128→   128→  const lines: JsonToken[][] = [[{ text: '{', cls: JSON_CLS.punct }]]
   129→   129→  const entries = Object.entries(value)
   130→   130→  entries.forEach(([key, val], i) => {
   131→   131→    const comma = i < entries.length - 1
   132→   132→    const keyTok: JsonToken = { text: `${pad}  ${JSON.stringify(key)}`, cls: JSON_CLS.key }
   133→   133→    if (isJsonObject(val)) {
   134→   134→      lines.push([keyTok, { text: ': ', cls: JSON_CLS.punct }])
   135→   135→      const nested = jsonLines(val, indent + 1)
   136→   136→      if (comma) {
   137→   137→        const lastLine = nested[nested.length - 1]
   138→   138→        const lastTok = lastLine?.[lastLine.length - 1]
   139→   139→        if (lastTok) lastTok.text += ','
   140→   140→      }
   141→   141→      nested.forEach((line) => lines.push(line))
   142→   142→    } else {
   143→   143→      const tok = scalarToken(val)
   144→   144→      lines.push([
   145→   145→        keyTok,
   146→   146→        { text: ': ', cls: JSON_CLS.punct },
   147→   147→        { text: tok.text + (comma ? ',' : ''), cls: tok.cls },
   148→   148→      ])
   149→   149→    }
   150→   150→  })
   151→   151→  lines.push([{ text: `${pad}}`, cls: JSON_CLS.punct }])
   152→   152→  return lines
   153→   153→}

   155→   155→/** UI-3: wall-clock timestamp for log lines — HH:MM:SS.mmm, LTR-safe. */
   156→   156→function logTimestamp(date: Date): string {
   157→   157→  const p2 = (n: number) => String(n).padStart(2, '0')
   158→   158→  const p3 = (n: number) => String(n).padStart(3, '0')
   159→   159→  return `${p2(date.getHours())}:${p2(date.getMinutes())}:${p2(date.getSeconds())}.${p3(date.getMilliseconds())}`
   160→   160→}
