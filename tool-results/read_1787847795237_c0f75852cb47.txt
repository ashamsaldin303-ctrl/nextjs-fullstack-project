    11→ * Motion model (item 12): particles are advected by a proper 3D simplex

    13→ * velocity, so clumps stream as coherent ribbons instead of the old

    65→/* ------------------------------------------------------------------ *
    66→ * GLSL — Ashima 3D simplex noise (Stefan Gustavson / Ian McEwan, MIT)
    67→ * + finite-difference curl. Written in GLSL1 style (attribute/varying)
    68→ * to match the existing shaderMaterial convention in this repo.
    69→ * ------------------------------------------------------------------ */
    70→const NOISE_GLSL = /* glsl */ `
    71→  vec3 mod289(vec3 x) {
    72→    return x - floor(x * (1.0 / 289.0)) * 289.0;
    73→  }
    74→  vec4 mod289(vec4 x) {
    75→    return x - floor(x * (1.0 / 289.0)) * 289.0;
    76→  }
    77→  vec4 permute(vec4 x) {
    78→    return mod289(((x * 34.0) + 1.0) * x);
    79→  }
    80→  vec4 taylorInvSqrt(vec4 r) {
    81→    return 1.79284291400159 - 0.85373472095314 * r;
    82→  }
    83→  float snoise(vec3 v) {
    84→    const vec2 C = vec2(1.0 / 6.0, 1.0 / 3.0);
    85→    const vec4 D = vec4(0.0, 0.5, 1.0, 2.0);
    86→
    87→    // First corner
    88→    vec3 i  = floor(v + dot(v, C.yyy));
    89→    vec3 x0 = v - i + dot(i, C.xxx);
    90→
    91→    // Other corners
    92→    vec3 g = step(x0.yzx, x0.xyz);
    93→    vec3 l = 1.0 - g;
    94→    vec3 i1 = min(g.xyz, l.zxy);
    95→    vec3 i2 = max(g.xyz, l.zxy);
    96→
    97→    vec3 x1 = x0 - i1 + C.xxx;
    98→    vec3 x2 = x0 - i2 + C.yyy;
    99→    vec3 x3 = x0 - D.yyy;
   100→
   101→    // Permutations
   102→    i = mod289(i);
   103→    vec4 p = permute(permute(permute(
   104→              i.z + vec4(0.0, i1.z, i2.z, 1.0))
   105→            + i.y + vec4(0.0, i1.y, i2.y, 1.0))
   106→            + i.x + vec4(0.0, i1.x, i2.x, 1.0));
   107→
   108→    // Gradients: 7x7 points over a square, mapped onto an octahedron.
   109→    float n_ = 0.142857142857;
   110→    vec3 ns = n_ * D.wyz - D.xzx;
   111→
   112→    vec4 j = p - 49.0 * floor(p * ns.z * ns.z);
   113→
   114→    vec4 x_ = floor(j * ns.z);
   115→    vec4 y_ = floor(j - 7.0 * x_);
   116→
   117→    vec4 x = x_ * ns.x + ns.yyyy;
   118→    vec4 y = y_ * ns.x + ns.yyyy;
   119→    vec4 h = 1.0 - abs(x) - abs(y);
   120→
   121→    vec4 b0 = vec4(x.xy, y.xy);
   122→    vec4 b1 = vec4(x.zw, y.zw);
   123→
   124→    vec4 s0 = floor(b0) * 2.0 + 1.0;
   125→    vec4 s1 = floor(b1) * 2.0 + 1.0;
   126→    vec4 sh = -step(h, vec4(0.0));
   127→
   128→    vec4 a0 = b0.xzyw + s0.xzyw * sh.xxyy;
   129→    vec4 a1 = b1.xzyw + s1.xzyw * sh.zzww;
   130→
   131→    vec3 p0 = vec3(a0.xy, h.x);
   132→    vec3 p1 = vec3(a0.zw, h.y);
   133→    vec3 p2 = vec3(a1.xy, h.z);
   134→    vec3 p3 = vec3(a1.zw, h.w);
   135→
   136→    // Normalise gradients
   137→    vec4 norm = taylorInvSqrt(vec4(dot(p0, p0), dot(p1, p1), dot(p2, p2), dot(p3, p3)));
   138→    p0 *= norm.x;
   139→    p1 *= norm.y;
   140→    p2 *= norm.z;
   141→    p3 *= norm.w;
   142→
   143→    // Mix final noise value
   144→    vec4 m = max(0.6 - vec4(dot(x0, x0), dot(x1, x1), dot(x2, x2), dot(x3, x3)), 0.0);
   145→    m = m * m;
   146→    return 42.0 * dot(m * m, vec4(dot(p0, x0), dot(p1, x1), dot(p2, x2), dot(p3, x3)));
   147→  }
   148→
   149→  // Vector noise — three offset simplex samples packed into a vec3.
   150→  vec3 snoiseVec3(vec3 x) {
   151→    float s  = snoise(x);
   152→    float s1 = snoise(vec3(x.y - 19.1, x.z + 33.4, x.x + 47.2));
   153→    float s2 = snoise(vec3(x.z + 74.2, x.x - 124.5, x.y + 99.4));
   154→    return vec3(s, s1, s2);
   155→  }
   156→
   157→  // Curl (rotational derivative) of the vector noise via central finite
   158→  // differences. Divergence-free by construction — flow never bunches
   159→  // into sinks, and neighbours share velocity => coherent ribbons.
   160→  vec3 curlNoise(vec3 p) {
   161→    const float e = 0.12;
   162→    vec3 dx = vec3(e, 0.0, 0.0);
   163→    vec3 dy = vec3(0.0, e, 0.0);
   164→    vec3 dz = vec3(0.0, 0.0, e);
   165→
   166→    vec3 px0 = snoiseVec3(p - dx);
   167→    vec3 px1 = snoiseVec3(p + dx);
   168→    vec3 py0 = snoiseVec3(p - dy);
   169→    vec3 py1 = snoiseVec3(p + dy);
   170→    vec3 pz0 = snoiseVec3(p - dz);
   171→    vec3 pz1 = snoiseVec3(p + dz);
   172→
   173→    float x = py1.z - py0.z - pz1.y + pz0.y;
   174→    float y = pz1.x - pz0.x - px1.z + px0.z;
   175→    float z = px1.y - px0.y - py1.x + py0.x;
   176→
   177→    vec3 curl = vec3(x, y, z) / (2.0 * e);
   178→    // safe normalise — never NaN, even if the finite differences cancel
   179→    return curl / max(length(curl), 1e-4);
   180→  }
   181→`

   183→const PARTICLE_VERTEX = /* glsl */ `
   184→  ${NOISE_GLSL}
   185→
   186→  attribute float aScale;
   187→  attribute float aSpeed;
   188→  attribute vec3 aColor;
   189→  uniform float uTime;
   190→  uniform vec3 uMouse;
   191→  uniform float uPixelRatio;
   192→  uniform float uScroll;
   193→  varying vec3 vColor;
   194→  varying float vAlpha;
   195→
   196→  void main() {
   197→    vec3 p = position;
   198→
   199→    // --- curl flow field -------------------------------------------------
   200→    // Sample the field in a slowly-translating noise space: the coherent
   201→    // structures sweep past the volume so each particle's velocity keeps
   202→    // re-forming — a streaming current, not in-place jitter.
   203→    vec3 q = p * 0.16 + vec3(0.0, -uTime * 0.18, 0.0);
   204→    vec3 vel = curlNoise(q);
   205→
   206→    // Per-particle amplitude (aSpeed 0.2…0.8) + a gentle swell as the
   207→    // methodology section scrolls through (uScroll 0…1).
   208→    float amp = (1.1 + aSpeed * 1.6) * (1.0 + uScroll * 0.4);
   209→    p += vel * amp;
   210→
   211→    // --- slow global drift ------------------------------------------------
   212→    // Embers rising through a tall wrapped domain; the edge fade below
   213→    // dissolves particles at the recycle seam so the wrap never pops.
   214→    const float DOMAIN = 20.0;
   215→    float y = mod(p.y + uTime * (0.22 + aSpeed * 0.18), DOMAIN) - DOMAIN * 0.5;
   216→    p.y = y;
   217→
   218→    // subtle attraction toward the mouse plane (kept from v1)
   219→    p += uMouse * 0.18;
   220→
   221→    vec4 mv = modelViewMatrix * vec4(p, 1.0);
   222→    gl_Position = projectionMatrix * mv;
   223→    float size = aScale * uPixelRatio * (28.0 / -mv.z);
   224→    gl_PointSize = clamp(size, 1.0, 64.0);
   225→    vColor = aColor;
   226→    // fade by distance (kept) + fade near the wrap seam
   227→    float depthFade = clamp(1.0 - (-mv.z - 3.0) / 22.0, 0.0, 1.0);
   228→    float edgeFade = 1.0 - smoothstep(8.5, 10.0, abs(y));
   229→    vAlpha = depthFade * edgeFade;
   230→  }
   231→`

   258→
   259→const SILK_FRAGMENT = /* glsl */ `
   260→  // Domain-warped fbm "silk" (Inigo Quilez warp structure): the sampling
   261→  // domain is bent through two offset fields — q = fbm(p + drift), then
   262→  // r = fbm(p*1.3 + q + counter-drift) — and the final sample
   263→  // f = fbm(p + r*2.2) flows as broad luminous bands. A gaussian vortex
   264→  // around the cursor stirs the whole domain (liquid-silk feel); graded
   265→  // brand blues, with the green confined to thin warp-crossing filaments.
   266→  uniform float uTime;
   267→  uniform vec2 uMouse;   // world-space (plane-local) cursor position, CPU-lerped
   268→  uniform vec3 uColorA;  // primary blue #4285F4 — silk body
   269→  uniform vec3 uColorB;  // brand green #34A853 — warp-crossing filaments only
   270→  uniform vec3 uColorC;  // deep blue #0A2A5E — low-energy undertone
   271→  uniform float uScroll; // methodology bridge 0…1 → gentle intensity swell
   272→  varying vec2 vUv;
   273→  varying vec2 vPos;
   274→
   275→  float hash(vec2 p) {
   276→    return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453123);
   277→  }
   278→
   279→  float vnoise(vec2 p) {
   280→    vec2 i = floor(p);
   281→    vec2 f = fract(p);
   282→    vec2 u = f * f * (3.0 - 2.0 * f);
   283→    float a = hash(i);
   284→    float b = hash(i + vec2(1.0, 0.0));
   285→    float c = hash(i + vec2(0.0, 1.0));
   286→    float d = hash(i + vec2(1.0, 1.0));
   287→    return mix(mix(a, b, u.x), mix(c, d, u.x), u.y);
   288→  }
   289→
   290→  // 5 octaves; the fixed rotation between octaves kills axis alignment.
   291→  float fbm(vec2 p) {
   292→    float v = 0.0;
   293→    float a = 0.5;
   294→    for (int i = 0; i < 5; i++) {
   295→      v += a * vnoise(p);
   296→      p = mat2(0.8, 0.6, -0.6, 0.8) * p * 2.03;
   297→      a *= 0.5;
   298→    }
   299→    return v;
   300→  }
   301→
   302→  void main() {
   303→    float t = uTime * 0.13;
   304→
   305→    // --- cursor vortex --------------------------------------------------
   306→    // Stir the sampling domain around the pointer: rotate the plane-local
   307→    // position around uMouse under a gaussian falloff (~6 world units).
   308→    // Angular displacement tops out at 0.35 rad, so the total shift stays
   309→    // under ~0.7 world units — the composition never turns chaotic.
   310→    vec2 toM = vPos - uMouse;
   311→    float md = length(toM);
   312→    float stir = exp(-(md * md) / 24.0);
   313→    float ang = 0.35 * stir;
   314→    float ca = cos(ang);
   315→    float sa = sin(ang);
   316→    vec2 pos = uMouse + mat2(ca, -sa, sa, ca) * toM;
   317→
   318→    // --- two-level domain warp (Inigo Quilez) ----------------------------
   319→    // q and r are offset fields sampled from the same fbm; f is the fbm
   320→    // sampled through BOTH warps. The two time offsets counter-drift, so
   321→    // the fields shear past each other and the fold pattern keeps
   322→    // re-forming instead of looping.
   323→    vec2 p = pos * 0.08;
   324→    float q = fbm(p + vec2(t * 0.6, -t * 0.4));
   325→    float r = fbm(p * 1.3 + vec2(q * 1.7, q * 1.1) + vec2(-t * 0.5, t * 0.3) + 2.7);
   326→    float f = fbm(p + r * 2.2);
   327→
   328→    // --- window / vignette / energy --------------------------------------
   329→    // Vertical melt at the viewport edges (aspect-safe: the fov is
   330→    // vertical, so ±10.5 world units at the plane distance is the screen
   331→    // height) + radial vignette on the light itself, nudged a touch high
   332→    // so the silk breathes brightest in the upper-middle band behind the
   333→    // headline.
   334→    float vert = smoothstep(-10.5, -5.0, vPos.y) * (1.0 - smoothstep(5.0, 10.5, vPos.y));
   335→    float vig = 1.0 - smoothstep(0.15, 0.42, length(vUv - vec2(0.5, 0.44)));
   336→    float lift = 0.85 + 0.30 * smoothstep(-4.0, 5.0, vPos.y);
   337→    float swell = 1.0 + uScroll * 0.25;
   338→
   339→    // --- masks ------------------------------------------------------------
   340→    float body  = smoothstep(0.36, 0.78, f);               // broad fabric body
   341→    float ridge = smoothstep(0.60, 0.92, f);               // luminous band ridges
   342→    float deep  = 1.0 - smoothstep(0.18, 0.55, f);         // low-energy zones
   343→    float sheen = 0.62 + 0.38 * smoothstep(0.25, 0.85, q); // q shimmers the ridges
   344→
   345→    // Green lives ONLY in thin filaments: where the second warp field r
   346→    // crosses a narrow level band while the body is lit — curved
   347→    // interference hairlines that drift as the warp fields slide.
   348→    float lvl = smoothstep(0.46, 0.55, r) * (1.0 - smoothstep(0.60, 0.70, r));
   349→    float filament = lvl * body * sheen;
   350→
   351→    // --- brand grades (derived from the uniform palette) ------------------
   352→    vec3 midBlue   = mix(uColorC, uColorA, 0.55);   // ≈ #0071E3 band depth
   353→    vec3 lightBlue = mix(uColorA, vec3(1.0), 0.22); // ≈ #60A5FA ridge sheen
   354→
   355→    vec3 col = vec3(0.0); // additive: the CSS stage shows through
   356→
   357→    // deep undertone between the bands
   358→    col += uColorC * deep * deep * 0.42 * vert * vig;
   359→
   360→    // silk body — hue graded by q, lifting into the light-blue sheen on
   361→    // the ridges
   362→    vec3 bandCol = mix(midBlue, uColorA, smoothstep(0.30, 0.70, q));
   363→    col += bandCol * body * (0.26 + 0.20 * ridge) * vert * vig * lift * swell;
   364→    col += lightBlue * body * ridge * sheen * 0.30 * vert * vig * lift * swell;
   365→
   366→    // warp-crossing filaments
   367→    col += uColorB * filament * 0.48 * vert * vig * lift * swell;
   368→
   369→    // broad cursor wash (blue family — green is reserved for filaments)
   370→    col += mix(uColorC, uColorA, 0.75) * exp(-md * md / 220.0) * 0.14 * swell;
   371→
   372→    // intensity governor: hard per-channel cap — the additive layer stays
   373→    // premium-dark even where masks coincidentally stack
   374→    col = min(col, vec3(0.50));
   375→
   376→    gl_FragColor = vec4(col, 1.0);
   377→  }

   380→function Particles({
   381→  mouse,
   382→  count,
   383→  dprMax = 2,
   384→}: {
   385→  mouse: React.MutableRefObject<{ x: number; y: number; tx: number; ty: number }>
   386→  /** Tier-dependent particle budget (desktop 4500 / mobile 1800) — the
   387→   *  parent keys the component on it, so a tier flip rebuilds the buffers. */
   388→  count: number
   389→  /** Tier-dependent upper bound for the uPixelRatio uniform — MUST track the
   390→   *  Canvas dpr prop (desktop 2 / mobile 1.5) or gl_PointSize drifts ~33%
   391→   *  oversize on retina phones (FIX L2-V P3). The parent's key remount on
   392→   *  tier flip re-runs the uniforms memo below. */
   393→  dprMax?: number
   394→}) {
   395→  const pointsRef = useRef<THREE.Points>(null)
   396→  const mouseVec = useRef(new THREE.Vector3(0, 0, 0))
   397→
   398→  // Particle geometry is generated ONCE per mount with Math.random — an impure
   399→  // call that React 19 only permits inside a lazy state initializer (never in
   400→  // render body or useMemo factories). See react-hooks/purity.
   401→  const [buffers] = useState(() => {
   402→    const positions = new Float32Array(count * 3)
   403→    const scales = new Float32Array(count)
   404→    const speeds = new Float32Array(count)
   405→    const colors = new Float32Array(count * 3)
   406→    for (let i = 0; i < count; i++) {
   407→      // Spread in a wide box around the camera
   408→      const radius = 4 + Math.random() * 9
   409→      const theta = Math.random() * Math.PI * 2
   410→      const phi = Math.acos(2 * Math.random() - 1)
   411→      positions[i * 3 + 0] = radius * Math.sin(phi) * Math.cos(theta)
   412→      positions[i * 3 + 1] = radius * Math.sin(phi) * Math.sin(theta) * 0.6
   413→      positions[i * 3 + 2] = radius * Math.cos(phi) * 0.5 - 2
   414→      scales[i] = Math.random() * 1.4 + 0.4
   415→      speeds[i] = Math.random() * 0.6 + 0.2
   416→      const c = COLORS[Math.floor(Math.random() * COLORS.length)]
   417→      if (!c) continue
   418→      colors[i * 3 + 0] = c.r
   419→      colors[i * 3 + 1] = c.g
   420→      colors[i * 3 + 2] = c.b
   421→    }
   422→    return { positions, scales, speeds, colors }
   423→  })
   424→  const { positions, scales, speeds, colors } = buffers
   425→
   426→  const uniforms = useMemo(
   427→    () => ({
   428→      uTime: { value: 0 },
   429→      uMouse: { value: new THREE.Vector3(0, 0, 0) },
   430→      // FIX(L2-V P3): tier-aware — clamped to the SAME ceiling the renderer
   431→      // uses (dprMax), so gl_PointSize matches the actual render scale on
   432→      // every tier. Desktop path is byte-equivalent (min(dpr, 2)).
   433→      uPixelRatio: { value: Math.min(typeof window !== 'undefined' ? window.devicePixelRatio : 1, dprMax) },
   434→      uScroll: { value: 0 },
   435→    }),
   436→    [dprMax]
   437→  )
   438→
   439→  useFrame((state, delta) => {
   440→    if (!pointsRef.current) return
   441→    // Three.js shader uniforms are mutated per-frame — the canonical R3F pattern.
   442→    // eslint-disable-next-line react-hooks/immutability
   443→    uniforms.uTime.value += delta
   444→
   445→    // Lerp mouse target into the actual mouse vector (smoothing)
   446→    mouseVec.current.x += (mouse.current.tx - mouseVec.current.x) * 0.04
   447→    mouseVec.current.y += (mouse.current.ty - mouseVec.current.y) * 0.04
   448→    uniforms.uMouse.value.copy(mouseVec.current)
   449→
   450→    // Scroll bridge (methodology section) → gentle flow swell, lerped.
   451→    const scrollTarget = getHeroScroll()
   452→    uniforms.uScroll.value += (scrollTarget - uniforms.uScroll.value) * 0.08
   453→
   454→    // Gentle group tilt toward mouse
   455→    pointsRef.current.rotation.y += ((mouseVec.current.x * 0.25) - pointsRef.current.rotation.y) * 0.05
   456→    pointsRef.current.rotation.x += ((mouseVec.current.y * 0.15) - pointsRef.current.rotation.x) * 0.05
   457→  })
   458→
   459→  return (
   460→    <points ref={pointsRef} frustumCulled={false}>
   461→      <bufferGeometry>
   462→        <bufferAttribute attach="attributes-position" args={[positions, 3]} />
   463→        <bufferAttribute attach="attributes-aScale" args={[scales, 1]} />
   464→        <bufferAttribute attach="attributes-aSpeed" args={[speeds, 1]} />
   465→        <bufferAttribute attach="attributes-aColor" args={[colors, 3]} />
   466→      </bufferGeometry>
   467→      <shaderMaterial
   468→        transparent
   469→        depthWrite={false}
   470→        blending={THREE.AdditiveBlending}
   471→        uniforms={uniforms}
   472→        vertexShader={PARTICLE_VERTEX}
   473→        fragmentShader={PARTICLE_FRAGMENT}
   474→      />
   475→    </points>
   476→  )
   477→}

   479→function SilkBackdrop({ mouse }: { mouse: React.MutableRefObject<{ x: number; y: number; tx: number; ty: number }> }) {
   480→  const mouseVec = useRef(new THREE.Vector2(0, 0))
   481→
   482→  const uniforms = useMemo(
   483→    () => ({
   484→      uTime: { value: 0 },
   485→      uMouse: { value: new THREE.Vector2(0, 0) },
   486→      uColorA: { value: new THREE.Color('#4285F4') }, // silk body blue
   487→      uColorB: { value: new THREE.Color('#34A853') }, // warp-crossing filaments only
   488→      uColorC: { value: new THREE.Color('#0A2A5E') }, // deep undertone
   489→      uScroll: { value: 0 },
   490→    }),
   491→    []
   492→  )
   493→
   494→  useFrame((state, delta) => {
   495→    // Three.js shader uniforms are mutated per-frame — the canonical R3F pattern.
   496→    // eslint-disable-next-line react-hooks/immutability
   497→    uniforms.uTime.value += delta
   498→
   499→    // Project the pointer onto the silk plane in world space (aspect-
   500→    // correct; numbers only — no per-frame allocations). The CPU lerp
   501→    // smooths the vortex centre, so the stir glides instead of snapping.
   502→    const aspect = state.size.width / state.size.height
   503→    const halfH = Math.tan(CAMERA_FOV_RAD * 0.5) * SILK_PLANE_DIST
   504→    mouseVec.current.x += (mouse.current.tx * halfH * aspect - mouseVec.current.x) * 0.05
   505→    mouseVec.current.y += (mouse.current.ty * halfH - mouseVec.current.y) * 0.05
   506→    uniforms.uMouse.value.copy(mouseVec.current)
   507→
   508→    // Scroll bridge (methodology section) → gentle silk swell, lerped
   509→    // with the same pattern Particles uses.
   510→    const scrollTarget = getHeroScroll()
   511→    uniforms.uScroll.value += (scrollTarget - uniforms.uScroll.value) * 0.08
   512→  })
   513→
   514→  return (
   515→    <mesh position={[0, 0, SILK_PLANE_Z]} renderOrder={-1} frustumCulled={false}>
   516→      {/* 80 wide covers 32:9 ultrawides (FIX(2-b), L1-D P3): the visible
   517→          half-width at the dolly's farthest z (dist 15.8, aspect 3.56) is
   518→          ≈39.4 world units < 40 — one fullscreen 1-segment quad, so the
   519→          extra width is cost-free. Height stays 32 (vertical fov). */}
   520→      <planeGeometry args={[80, 32]} />
   521→      {/* True additive over the PAGE, not just over the framebuffer: the
   522→          canvas is transparent, so plain AdditiveBlending (SRC_ALPHA, ONE)
   523→          would accumulate alpha≈1 across the fullscreen plane and hide the
   524→          CSS stage underneath. CustomBlending with (ONE,ONE / ZERO,ONE)
   525→          adds light to the color buffer while leaving canvas alpha at 0 —
   526→          the deep background, hero-fallback gradient and blueprint
   527→          spotlight grid all stay visible beneath the silk. */}
   528→      <shaderMaterial
   529→        transparent
   530→        depthWrite={false}
   531→        depthTest={false}
   532→        blending={THREE.CustomBlending}
   533→        blendSrc={THREE.OneFactor}
   534→        blendDst={THREE.OneFactor}
   535→        blendSrcAlpha={THREE.ZeroFactor}
   536→        blendDstAlpha={THREE.OneFactor}
   537→        uniforms={uniforms}
   538→        vertexShader={SILK_VERTEX}
   539→        fragmentShader={SILK_FRAGMENT}
   540→      />
   541→    </mesh>
   542→  )
   543→}

   545→/** Batch 3 item 15 — scroll-driven camera dolly. Eases the camera +0.8
   546→ *  units back as the methodology section's scroll progress (bridge)
   547→ *  goes 0→1 — a subtle pull-away, RTL-safe (pure view-axis motion). */
   548→function ScrollDolly() {
   549→  useFrame((state) => {
   550→    const targetZ = BASE_CAMERA_Z + getHeroScroll() * 0.8
   551→    state.camera.position.z += (targetZ - state.camera.position.z) * 0.05
   552→  })
   553→  return null
   554→}

   556→/** Context-loss guard — `preventDefault()` marks the event as handled so
   557→ *  the browser keeps the canvas alive for a possible restore (and stops
   558→ *  the default console error spam); we log once for diagnostics. */
   559→function ContextLossGuard() {
   560→  const gl = useThree((s) => s.gl)
   561→  useEffect(() => {
   562→    const canvas = gl.domElement
   563→    const onLost = (e: Event) => {
   564→      e.preventDefault()
   565→      console.warn('[HeroCanvas] WebGL context lost')
   566→    }
   567→    canvas.addEventListener('webglcontextlost', onLost)
   568→    return () => canvas.removeEventListener('webglcontextlost', onLost)
   569→  }, [gl])
   570→  return null
   571→}
   572→
   573→/* Mobile tier (FIX(2-b), L1-D P2) via useSyncExternalStore — the exact
   574→ * pattern of use-reduced-motion.ts: one module-level MQL singleton
   575→ * (matchMedia allocates a fresh MediaQueryList on every call) + a false
   576→ * server snapshot (desktop tier during SSR; HeroCanvas is a client-only
   577→ * dynamic import anyway). Reactive by design: crossing the boundary
   578→ * mid-session swaps the dpr in place (R3F re-applies the pixel ratio
   579→ * without recreating the canvas/GL context) and remounts Particles via a
   580→ * key so the buffers resize to the new count — the canvas itself survives. */
   581→let tierMql: MediaQueryList | null = null
   582→
   583→function getTierMql(): MediaQueryList | null {
   584→  if (typeof window === 'undefined') return null
   585→  if (!tierMql) {
   586→    tierMql = window.matchMedia('(max-width: 767px), (pointer: coarse)')
   587→  }
   588→  return tierMql
   589→}
   590→
   591→function subscribeTier(onChange: () => void): () => void {
   592→  const mq = getTierMql()
   593→  if (!mq) return () => {}
   594→  mq.addEventListener('change', onChange)
   595→  return () => mq.removeEventListener('change', onChange)
   596→}
   597→
   598→function getTierSnapshot(): boolean {
   599→  return getTierMql()?.matches ?? false
   600→}
   601→
   602→function getTierServerSnapshot(): boolean {
   603→  return false
   604→}
   605→
   606→/** True on coarse-pointer or <768px viewports → lighter particle tier. */
   607→function useMobileTier(): boolean {
   608→  return useSyncExternalStore(subscribeTier, getTierSnapshot, getTierServerSnapshot)
   609→}

   611→interface HeroCanvasProps {
   612→  active: boolean
   613→}
   614→
   615→export function HeroCanvas({ active }: HeroCanvasProps) {
   616→  const mouse = useRef({ x: 0, y: 0, tx: 0, ty: 0 })
   617→  const [glAvailable, setGlAvailable] = useState(true)
   618→  // FIX(2-b), L1-D P2: mobile tier — coarse-pointer/<768px viewports get
   619→  // 1800 particles + dpr ≤ 1.5 (was: full-fat 4500 + dpr ≤ 2 on phones).
   620→  const mobileTier = useMobileTier()
   621→  const particleCount = mobileTier ? PARTICLE_COUNT_MOBILE : PARTICLE_COUNT_DESKTOP
   622→
   623→  useEffect(() => {
   624→    // WebGL feature detection — deferred to rAF so the setState isn't synchronous
   625→    // inside the effect body (hydration-safe + lint-compliant).
   626→    let cancelled = false
   627→    const id = requestAnimationFrame(() => {
   628→      if (cancelled) return
   629→      try {
   630→        const test = document.createElement('canvas').getContext('webgl2') ?? document.createElement('canvas').getContext('webgl')
   631→        if (!test) setGlAvailable(false)
   632→      } catch {
   633→        setGlAvailable(false)
   634→      }
   635→    })
   636→    return () => {
   637→      cancelled = true
   638→      cancelAnimationFrame(id)
   639→    }
   640→  }, [])
   641→
   642→  useEffect(() => {
   643→    const onMove = (e: PointerEvent) => {
   644→      mouse.current.tx = (e.clientX / window.innerWidth) * 2 - 1
   645→      mouse.current.ty = -((e.clientY / window.innerHeight) * 2 - 1)
   646→    }
   647→    window.addEventListener('pointermove', onMove, { passive: true })
   648→    return () => window.removeEventListener('pointermove', onMove)
   649→  }, [])
   650→
   651→  if (!glAvailable) return null
   652→
   653→  return (
   654→    <div className="absolute inset-0 -z-0">
   655→      <Canvas
   656→        dpr={mobileTier ? [1, 1.5] : [1, 2]}
   657→        frameloop={active ? 'always' : 'never'}
   658→        gl={{ antialias: false, alpha: true, powerPreference: 'high-performance' }}
   659→        camera={{ position: [0, 0, BASE_CAMERA_Z], fov: 70 }}
   660→        style={{ background: 'transparent' }}
   661→      >
   662→        <SilkBackdrop mouse={mouse} />
   663→        {/* key: the buffers are sized once per mount in the lazy useState
   664→            initializer, so a tier flip remounts the Points with the new
   665→            count — the Canvas/GL context itself survives the change */}
   666→        <Particles key={particleCount} mouse={mouse} count={particleCount} dprMax={mobileTier ? 1.5 : 2} />
   667→        <ScrollDolly />
   668→        <ContextLossGuard />
   669→      </Canvas>
   670→    </div>
   671→  )
   672→}
