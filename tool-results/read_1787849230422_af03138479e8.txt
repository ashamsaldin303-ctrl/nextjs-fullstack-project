     1→'use client'
     2→
     3→import { useMemo, useRef, useState } from 'react'
     4→import { useLocale, useTranslations } from 'next-intl'
     5→import { AnimatePresence, motion, useReducedMotion } from 'framer-motion'
     6→import { z } from 'zod'
     7→import {
     8→  Globe, Workflow, Boxes, Check, ArrowLeft, ArrowRight,
     9→  Send, RotateCw, AlertCircle,
    10→} from 'lucide-react'
    11→import {
    12→  Slider,
    13→} from '@/components/ui/slider'
    14→import { Input } from '@/components/ui/input'
    15→import { Label } from '@/components/ui/label'
    16→import { Button } from '@/components/ui/button'
    17→import { cn } from '@/lib/utils'
    18→import { SectionHeading } from '@/components/shared/section-heading'
    19→import { leadEmailSchema, leadNameSchema, leadWhatsappSchema } from '@/lib/lead-fields'
    20→import { playSuccess } from '@/lib/sound'
    21→import { toast } from 'sonner'
    22→import { RingGauge } from './ring-gauge'
    23→import {
    24→  computeEstimate, formatMoney,
    25→  type CalculatorInput, type ServiceType, type IntegrationKey,
    26→  type AutomationLevel, type LanguageOption, type ThreeDOption, type Locale,
    27→} from '@/lib/calculator'
    28→
    29→type Step = 0 | 1 | 2
    30→
    31→const SERVICES: { id: ServiceType; icon: typeof Globe }[] = [
    32→  { id: 'website', icon: Globe },
    33→  { id: 'automation', icon: Workflow },
    34→  { id: 'full', icon: Boxes },
    35→]
    36→
    37→const INTEGRATIONS: IntegrationKey[] = ['crm', 'invoicing', 'email', 'telegram', 'sheets', 'ai']
    38→
    39→// Shared lead-field schemas (R2-MED-1 / R6-LOW-3): the SAME rules the
    40→// API enforces — name 2–100, email ≤254, whatsapp 5–30 + phone pattern —
    41→// so a client-side rejection can never diverge from a server-side one.
    42→const leadSchema = z.object({
    43→  name: leadNameSchema,
    44→  email: leadEmailSchema,
    45→  whatsapp: leadWhatsappSchema,
    46→})
    47→
    48→type LeadForm = z.infer<typeof leadSchema>
    49→
    50→const INITIAL_INPUT: CalculatorInput = {
    51→  service: 'website',
    52→  pages: 6,
    53→  languages: 'bilingual',
    54→  threeD: 'no',
    55→  integrations: [],
    56→  automationLevel: 'essential',
    57→}
    58→
    59→export function Calculator() {
    60→  const t = useTranslations('calculator')
    61→  // Whatsapp has no local calculator.errors key — reuse the API's own
    62→  // translated field copy (apiErrors.fields.whatsapp) so client and
    63→  // server rejections read identically (same rule, same message).
    64→  const tApiFields = useTranslations('apiErrors.fields')
    65→  // useLocale() returns a broad `string`; narrow to the routing union
    66→  // (unknown values fall back to `ar`, the site default) so formatMoney's
    67→  // tightened `Locale` param type-checks.
    68→  const localeRaw = useLocale()
    69→  const locale: Locale = localeRaw === 'en' ? 'en' : 'ar'
    70→  const reduced = useReducedMotion()
    71→
    72→  const [step, setStep] = useState<Step>(0)
    73→  const [dir, setDir] = useState<1 | -1>(1)
    74→  const [input, setInput] = useState<CalculatorInput>(INITIAL_INPUT)
    75→  const [form, setForm] = useState<LeadForm>({ name: '', email: '', whatsapp: '' })
    76→  const [errors, setErrors] = useState<{ name?: string; email?: string; whatsapp?: string }>({})
    77→  const [submitting, setSubmitting] = useState(false)
    78→  const [done, setDone] = useState(false)
    79→  const [reference, setReference] = useState<string | null>(null)
    80→  // FIX(2-c/18): honeypot trap — bots autofill hidden "companyWebsite"
    81→  // fields; humans never see it. The value rides along in the JSON body
    82→  // and the API silently discards bot submissions with a fake success.
    83→  const honeypotRef = useRef<HTMLInputElement>(null)
    84→
    85→  const result = useMemo(() => computeEstimate(input), [input])
    86→
    87→  // Breakdown line labels — resolved once for type-safety (guide §4.6).
    88→  const breakdownLabels = {
    89→    base: t('result.base'),
    90→    pages: t('result.pages'),
    91→    bilingual: t('result.bilingual'),
    92→    threeD: t('result.threeD'),
    93→    integrations: t('result.integrations'),
    94→    advanced: t('result.advanced'),
    95→  } as const
    96→
    97→  const goNext = () => { setDir(1); setStep((s) => Math.min(2, s + 1) as Step) }
    98→  const goBack = () => { setDir(-1); setStep((s) => Math.max(0, s - 1) as Step) }
    99→
   100→  const toggleIntegration = (key: IntegrationKey) => {
   101→    setInput((p) => ({
   102→      ...p,
   103→      integrations: p.integrations.includes(key)
   104→        ? p.integrations.filter((k) => k !== key)
   105→        : [...p.integrations, key],
   106→    }))
   107→  }
   108→
   109→  const onSubmit = async (e: React.FormEvent) => {
   110→    e.preventDefault()
   111→    const parsed = leadSchema.safeParse(form)
   112→    if (!parsed.success) {
   113→      const fe: { name?: string; email?: string; whatsapp?: string } = {}
   114→      for (const issue of parsed.error.issues) {
   115→        const path = issue.path[0]
   116→        if (path === 'name') fe.name = t('errors.name')
   117→        if (path === 'email') fe.email = t('errors.email')
   118→        if (path === 'whatsapp') fe.whatsapp = tApiFields('whatsapp')
   119→      }
   120→      setErrors(fe)
   121→      return
   122→    }
   123→    setErrors({})
   124→    setSubmitting(true)
   125→    // Phase 3: real storage — the server recomputes the estimate from the
   126→    // wizard options and returns a reference (guide §2.9). Client numbers
   127→    // are never sent: only the option set travels.
   128→    try {
   129→      const res = await fetch('/api/leads', {
   130→        method: 'POST',
   131→        headers: {
   132→          'Content-Type': 'application/json',
   133→          'x-elyra-locale': locale,
   134→        },
   135→        body: JSON.stringify({
   136→          source: 'calculator',
   137→          companyWebsite: honeypotRef.current?.value ?? '',
   138→          name: parsed.data.name,
   139→          email: parsed.data.email,
   140→          whatsapp: parsed.data.whatsapp || undefined,
   141→          service: input.service,
   142→          pages: input.pages,
   143→          languages: input.languages,
   144→          threeD: input.threeD,
   145→          integrations: input.integrations,
   146→          automationLevel: input.automationLevel,
   147→        }),
   148→      })
   149→
   150→      if (res.status === 201) {
   151→        const data = (await res.json()) as { reference?: string }
   152→        setReference(data.reference ?? null)
   153→        setDone(true)
   154→        playSuccess() // Phase 2 sensory feedback — fires on REAL success only
   155→        return
   156→      }
   157→
   158→      // Server rejected: surface translated server-side messages.
   159→      const data = (await res.json().catch(() => null)) as
   160→        | { message?: string; fields?: Record<string, string> }
   161→        | null
   162→      if (res.status === 400 && data?.fields) {
   163→        const fe: { name?: string; email?: string; whatsapp?: string } = {}
   164→        if (data.fields.name) fe.name = data.fields.name
   165→        if (data.fields.email) fe.email = data.fields.email
   166→        if (data.fields.whatsapp) fe.whatsapp = data.fields.whatsapp
   167→        setErrors(fe)
   168→      }
   169→      toast.error(t('form.errorTitle'), {
   170→        description: data?.message ?? t('form.errorNetwork'),
   171→      })
   172→    } catch {
   173→      // Network failure — data stays in the form for a retry.
   174→      toast.error(t('form.errorTitle'), { description: t('form.errorNetwork') })
   175→    } finally {
   176→      setSubmitting(false)
   177→    }
   178→  }
   179→
   180→  const slideVariants = reduced
   181→    ? undefined
   182→    : {
   183→        enter: (d: number) => ({ opacity: 0, x: d * 30 }),
   184→        center: { opacity: 1, x: 0 },
   185→        exit: (d: number) => ({ opacity: 0, x: -d * 30 }),
   186→      }
   187→
   188→  // L1-C P3 (fix 2-d): aria-label REMOVED — it overrode aria-labelledby
   189→  // and named the landmark by the short kicker instead of the full h2.
   190→  return (
   191→    <section id="calculator" className="bg-background py-20 sm:py-28" aria-labelledby="calc-title">
   192→      <div className="elyra-container max-w-5xl">
   193→        <SectionHeading
   194→          kicker={t('kicker')}
   195→          title={t('title')}
   196→          subtitle={t('subtitle')}
   197→          titleId="calc-title"
   198→        />
   199→
   200→        {/* Progress */}
   201→        <div className="mt-10 flex items-center justify-between gap-4">
   202→          <p className="text-sm text-muted-foreground">
   203→            {t('step', { current: step + 1, total: 3 })}
   204→          </p>
   205→          <div className="flex flex-1 gap-2">
   206→            {[0, 1, 2].map((s) => (
   207→              <div
   208→                key={s}
   209→                className="h-1.5 flex-1 overflow-hidden rounded-full bg-muted"
   210→                aria-hidden="true"
   211→              >
   212→                <motion.div
   213→                  className="h-full rounded-full bg-primary"
   214→                  initial={false}
   215→                  animate={{ width: step >= s ? '100%' : '0%' }}
   216→                  transition={{ duration: 0.4 }}
   217→                />
   218→              </div>
   219→            ))}
   220→          </div>
   221→        </div>
   222→
   223→        <div className="relative mt-8 overflow-hidden rounded-3xl border border-border bg-card p-6 sm:p-10">
   224→          {/* UI-5 (visual-only): gradient hairline accent along the wizard
   225→              card's top edge — decorative, no logic/state/pricing impact. */}
   226→          <span
   227→            aria-hidden="true"
   228→            className="pointer-events-none absolute inset-x-0 top-0 h-px bg-gradient-to-r from-transparent via-primary/40 to-transparent"
   229→          />
   230→          <AnimatePresence mode="wait" custom={dir}>
   231→            <motion.div
   232→              key={step}
   233→              custom={dir}
   234→              variants={slideVariants}
   235→              initial="enter"
   236→              animate="center"
   237→              exit="exit"
   238→              transition={{ duration: 0.4, ease: [0.16, 1, 0.3, 1] }}
   239→            >
   240→              {/* STEP 0 — Service type */}
   241→              {step === 0 ? (
   242→                <div>
   243→                  <h3 className="text-lg font-semibold">{t('steps.service')}</h3>
   244→                  <div className="mt-6 grid gap-4 sm:grid-cols-3">
   245→                    {SERVICES.map(({ id, icon: Icon }) => {
   246→                      const active = input.service === id
   247→                      const isFull = id === 'full'
   248→                      return (
   249→                        <button
   250→                          key={id}
   251→                          type="button"
   252→                          data-cursor="magnet"
   253→                          onClick={() => setInput((p) => ({ ...p, service: id }))}
   254→                          aria-pressed={active}
   255→                          className={cn(
   256→                            'group relative overflow-hidden rounded-2xl border p-5 text-start transition-all',
   257→                            active
   258→                              ? 'border-primary bg-primary/5 shadow-[0_0_0_1px_var(--color-primary)]'
   259→                              : 'border-border hover:border-primary/40 hover:bg-foreground/[0.02]'
   260→                          )}
   261→                        >
   262→                          {isFull ? (
   263→                            <span className="absolute end-3 top-3 rounded-full bg-primary px-2 py-0.5 text-[10px] font-semibold text-primary-foreground">
   264→                              {t('services.full.popular')}
   265→                            </span>
   266→                          ) : null}
   267→                          <div className={cn(
   268→                            'flex size-11 items-center justify-center rounded-xl transition-colors',
   269→                            active ? 'bg-primary text-primary-foreground' : 'bg-primary/10 text-primary'
   270→                          )}>
   271→                            <Icon className="size-5" aria-hidden="true" />
   272→                          </div>
   273→                          <p className="mt-3 font-semibold">{t(`services.${id}.title`)}</p>
   274→                          <p className="mt-1 text-sm text-muted-foreground">{t(`services.${id}.desc`)}</p>
   275→                          {active ? (
   276→                            <span className="absolute -end-2 -top-2 flex size-6 items-center justify-center rounded-full bg-primary text-primary-foreground">
   277→                              <Check className="size-3.5" aria-hidden="true" />
   278→                            </span>
   279→                          ) : null}
   280→                        </button>
   281→                      )
   282→                    })}
   283→                  </div>
   284→                </div>
   285→              ) : null}
   286→
   287→              {/* STEP 1 — Features */}
   288→              {step === 1 ? (
   289→                <div className="space-y-8">
   290→                  <div>
   291→                    <div className="flex items-center justify-between">
   292→                      <h3 className="text-lg font-semibold">{t('pages')}</h3>
   293→                      <span className="text-sm font-medium text-primary tabular-nums">
   294→                        {t('pagesValue', { count: input.pages })}
   295→                      </span>
   296→                    </div>
   297→                    <div className="mt-4 px-1">
   298→                      <Slider
   299→                        value={[input.pages]}
   300→                        onValueChange={(v) => setInput((p) => ({ ...p, pages: v[0] ?? p.pages }))}
   301→                        min={1}
   302→                        max={20}
   303→                        step={1}
   304→                        aria-label={t('pages')}
   305→                      />
   306→                      <div className="mt-2 flex justify-between text-xs text-muted-foreground">
   307→                        <span>1</span><span>20</span>
   308→                      </div>
   309→                    </div>
   310→                  </div>
   311→
   312→                  <div>
   313→                    <h3 className="text-lg font-semibold">{t('languages')}</h3>
   314→                    <div className="mt-3 grid gap-3 sm:grid-cols-2">
   315→                      {(['single', 'bilingual'] as LanguageOption[]).map((opt) => {
   316→                        const active = input.languages === opt
   317→                        return (
   318→                          <button
   319→                            key={opt}
   320→                            type="button"
   321→                            onClick={() => setInput((p) => ({ ...p, languages: opt }))}
   322→                            aria-pressed={active}
   323→                            className={cn(
   324→                              'rounded-xl border px-4 py-3 text-start transition-colors',
   325→                              active ? 'border-primary bg-primary/5' : 'border-border hover:bg-foreground/[0.02]'
   326→                            )}
   327→                          >
   328→                            <span className="flex items-center gap-2">
   329→                              <span className={cn('flex size-5 items-center justify-center rounded-full border', active ? 'border-primary bg-primary text-primary-foreground' : 'border-border')}>
   330→                                {active ? <Check className="size-3" aria-hidden="true" /> : null}
   331→                              </span>
   332→                              <span className="text-sm font-medium">{t(`languagesOptions.${opt}`)}</span>
   333→                            </span>
   334→                          </button>
   335→                        )
   336→                      })}
   337→                    </div>
   338→                  </div>
   339→
   340→                  <div>
   341→                    <h3 className="text-lg font-semibold">{t('threeD')}</h3>
   342→                    <div className="mt-3 grid gap-3 sm:grid-cols-2">
   343→                      {(['yes', 'no'] as ThreeDOption[]).map((opt) => {
   344→                        const active = input.threeD === opt
   345→                        const disabled = input.service === 'automation'
   346→                        return (
   347→                          <button
   348→                            key={opt}
   349→                            type="button"
   350→                            disabled={disabled}
   351→                            onClick={() => setInput((p) => ({ ...p, threeD: opt }))}
   352→                            aria-pressed={active}
   353→                            className={cn(
   354→                              'rounded-xl border px-4 py-3 text-start transition-colors',
   355→                              disabled && 'opacity-50',
   356→                              active ? 'border-primary bg-primary/5' : 'border-border hover:bg-foreground/[0.02]'
   357→                            )}
   358→                          >
   359→                            <span className="flex items-center gap-2">
   360→                              <span className={cn('flex size-5 items-center justify-center rounded-full border', active ? 'border-primary bg-primary text-primary-foreground' : 'border-border')}>
   361→                                {active ? <Check className="size-3" aria-hidden="true" /> : null}
   362→                              </span>
   363→                              <span className="text-sm font-medium">{t(`threeDOptions.${opt}`)}</span>
   364→                            </span>
   365→                          </button>
   366→                        )
   367→                      })}
   368→                    </div>
   369→                  </div>
   370→
   371→                  <div>
   372→                    <h3 className="text-lg font-semibold">{t('integrations')}</h3>
   373→                    <div className="mt-3 flex flex-wrap gap-2">
   374→                      {INTEGRATIONS.map((key) => {
   375→                        const active = input.integrations.includes(key)
   376→                        return (
   377→                          <button
   378→                            key={key}
   379→                            type="button"
   380→                            onClick={() => toggleIntegration(key)}
   381→                            aria-pressed={active}
   382→                            className={cn(
   383→                              'inline-flex min-h-11 items-center gap-1.5 rounded-full border px-3 py-1.5 text-sm transition-colors',
   384→                              active ? 'border-primary bg-primary/10 text-primary' : 'border-border hover:bg-foreground/[0.02]'
   385→                            )}
   386→                          >
   387→                            <span className={cn('flex size-4 items-center justify-center rounded-full border', active ? 'border-primary bg-primary text-primary-foreground' : 'border-border')}>
   388→                              {active ? <Check className="size-2.5" aria-hidden="true" /> : null}
   389→                            </span>
   390→                            {t(`integrationsOptions.${key}`)}
   391→                          </button>
   392→                        )
   393→                      })}
   394→                    </div>
   395→                  </div>
   396→
   397→                  <div>
   398→                    <h3 className="text-lg font-semibold">{t('automationLevel')}</h3>
   399→                    <div className="mt-3 grid gap-3 sm:grid-cols-2">
   400→                      {(['essential', 'advanced'] as AutomationLevel[]).map((opt) => {
   401→                        const active = input.automationLevel === opt
   402→                        const disabled = input.service === 'website'
   403→                        return (
   404→                          <button
   405→                            key={opt}
   406→                            type="button"
   407→                            disabled={disabled}
   408→                            onClick={() => setInput((p) => ({ ...p, automationLevel: opt }))}
   409→                            aria-pressed={active}
   410→                            className={cn(
   411→                              'rounded-xl border px-4 py-3 text-start transition-colors',
   412→                              disabled && 'opacity-50',
   413→                              active ? 'border-primary bg-primary/5' : 'border-border hover:bg-foreground/[0.02]'
   414→                            )}
   415→                          >
   416→                            <span className="flex items-center gap-2">
   417→                              <span className={cn('flex size-5 items-center justify-center rounded-full border', active ? 'border-primary bg-primary text-primary-foreground' : 'border-border')}>
   418→                                {active ? <Check className="size-3" aria-hidden="true" /> : null}
   419→                              </span>
   420→                              <span className="text-sm font-medium">{t(`automationLevelOptions.${opt}`)}</span>
   421→                            </span>
   422→                          </button>
   423→                        )
   424→                      })}
   425→                    </div>
   426→                  </div>
   427→                </div>
   428→              ) : null}
   429→
   430→              {/* STEP 2 — Result */}
   431→              {step === 2 ? (
   432→                <div>
   433→                  {done ? (
   434→                    <div className="flex flex-col items-center py-8 text-center">
   435→                      <div className="flex size-16 items-center justify-center rounded-full bg-g-green/15 text-g-green">
   436→                        <Check className="size-8" aria-hidden="true" />
   437→                      </div>
   438→                      <h3 className="mt-5 text-2xl font-semibold">{t('form.successTitle')}</h3>
   439→                      <p className="mt-2 max-w-md text-muted-foreground">{t('form.successDesc')}</p>
   440→                      {reference ? (
   441→                        <p className="mt-3 rounded-full border border-g-green/30 bg-g-green/5 px-4 py-1.5 font-mono text-sm font-semibold tracking-wide text-g-green">
   442→                          {t('form.successReference', { reference })}
   443→                        </p>
   444→                      ) : null}
   445→                      <button
   446→                        type="button"
   447→                        onClick={() => { setDone(false); setReference(null); setStep(0); setInput(INITIAL_INPUT); setForm({ name: '', email: '', whatsapp: '' }) }}
   448→                        className="mt-6 inline-flex h-11 items-center gap-2 rounded-full border border-border px-5 text-sm font-medium hover:bg-foreground/5"
   449→                      >
   450→                        <RotateCw className="size-4" aria-hidden="true" />
   451→                        {t('form.successRestart')}
   452→                      </button>
   453→                    </div>
   454→                  ) : (
   455→                    <div className="grid gap-8 lg:grid-cols-2">
   456→                      <div>
   457→                        <div className="rounded-2xl border border-border bg-gradient-to-br from-primary/5 to-transparent p-6">
   458→                          {/* WS-4: animated SVG ring gauges */}
   459→                          <div className="mb-6 flex items-center justify-center gap-12">
   460→                            <RingGauge
   461→                              fraction={Math.min(1, result.max / 20000)}
   462→                              value={result.max}
   463→                              formatValue={(n) => formatMoney(Math.round(n), locale)}
   464→                              label={t('result.budget')}
   465→                              color="#0071E3"
   466→                              isRtl={locale === 'ar'}
   467→                            />
   468→                            <RingGauge
   469→                              fraction={Math.min(1, result.weeksMax / 12)}
   470→                              value={result.weeksMax}
   471→                              formatValue={(n) => `${Math.round(n)}`}
   472→                              label={t('result.duration')}
   473→                              color="#34A853"
   474→                              isRtl={locale === 'ar'}
   475→                            />
   476→                          </div>
   477→                          <p className="text-xs font-medium uppercase tracking-wider text-muted-foreground">
   478→                            {t('result.budget')}
   479→                          </p>
   480→                          <p className="mt-2 text-3xl font-bold tracking-tight text-primary sm:text-4xl">
   481→                            {formatMoney(result.min, locale)} – {formatMoney(result.max, locale)}
   482→                          </p>
   483→                          <p className="mt-4 text-xs font-medium uppercase tracking-wider text-muted-foreground">
   484→                            {t('result.duration')}
   485→                          </p>
   486→                          <p className="mt-1 text-xl font-semibold">
   487→                            {t('result.weeks', { min: result.weeksMin, max: result.weeksMax })}
   488→                          </p>
   489→                        </div>
   490→
   491→                        <div className="mt-4">
   492→                          <p className="text-sm font-semibold">{t('result.breakdownTitle')}</p>
   493→                          <ul className="mt-2 space-y-1.5 text-sm">
   494→                            {result.breakdown.map((line) => (
   495→                              <li key={line.labelKey} className="flex items-center justify-between gap-2">
   496→                                <span className="text-muted-foreground">{breakdownLabels[line.labelKey as keyof typeof breakdownLabels]}</span>
   497→                                <span className="tabular-nums">
   498→                                  {formatMoney(line.min, locale)} – {formatMoney(line.max, locale)}
   499→                                </span>
